DECL Shao.OQ3.bounded_support_test_integrable_l1_bound
TYPE ∀ (f : ℝ × ℝ → ℝ),
  Measurable f →
    ∀ (C R : ℝ),
      0 ≤ C →
        0 ≤ R →
          (∀ (z : ℝ × ℝ), |f z| ≤ C) →
            (∀ (z : ℝ × ℝ), R < ‖z‖ → f z = 0) →
              MeasureTheory.Integrable f MeasureTheory.volume ∧ ∫ (z : ℝ × ℝ), |f z| ≤ 4 * C * R ^ 2
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
