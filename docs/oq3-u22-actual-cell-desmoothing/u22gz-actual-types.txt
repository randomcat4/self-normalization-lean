      (rho = MeasureTheory.volume.withDensity fun z => ENNReal.ofReal (k z)) →

          0 ≤ K →
            0 ≤ R →
              0 ≤ C →
                (∀ (z : ℝ × ℝ), k z ≤ K) →
                  ∀ (f : ℝ × ℝ → ℝ),
                    Measurable f →
                      (∀ (z : ℝ × ℝ), 0 ≤ f z) →

                                ∫ (z : ℝ × ℝ), f z ∂rho ≤ 2 * Real.pi * K * Real.exp (R ^ 2) * ∫ (z : ℝ × ℝ), f z ∂Phi
