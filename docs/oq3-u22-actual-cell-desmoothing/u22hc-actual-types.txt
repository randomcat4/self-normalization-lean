DECL Shao.OQ3.compact_open_domain_actual_cell_weighted_desmoothing_small
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure mu] (Theta : Set (ℝ × ℝ)),
  IsCompact Theta →
    Theta ⊆ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)) →
      ∀ (beta c R : ℝ),
        0 ≤ beta →
          beta < 2 / 5 →
            0 < c →
              0 ≤ R →
                ∀ (level a0 stripRadius : ℝ),
                  0 ≤ stripRadius →
                    (∀ p ∈ Theta, 0 < (Shao.OQ3.parabolaCovariance (mu.tilted fun x => p.1 * x + p.2 * x ^ 2)).det) →
                      (∀ p ∈ Theta,
                          ∀ (xi eta : ℝ),
                            R < √(xi ^ 2 + eta ^ 2) →
                              ‖Shao.OQ3.jointCharacteristic (mu.tilted fun x => p.1 * x + p.2 * x ^ 2) xi eta‖ ≤
                                1 - c * (√(xi ^ 2 + eta ^ 2)).rpow (-beta)) →
                        have tilt := fun p => mu.tilted fun x => p.1 * x + p.2 * x ^ 2;
                        have D := fun p => Shao.OQ3.parabolaCovariance (tilt p);
                        have s := fun p => √(D p 0 0 * D p 1 1 - D p 0 1 ^ 2);
                        have t := fun p => √(D p 0 0 + D p 1 1 + 2 * s p);
                        have A := fun p =>
                          ![![(D p 1 1 + s p) / (s p * t p), -D p 0 1 / (s p * t p)],
                            ![-D p 0 1 / (s p * t p), (D p 0 0 + s p) / (s p * t p)]];
                        have means := fun p => ![∫ (x : ℝ), x ∂tilt p, ∫ (x : ℝ), x ^ 2 ∂tilt p];
                        have Y := fun p x => (A p).mulVec (![x, x ^ 2] - means p);
                        have laws := fun p => MeasureTheory.Measure.map (Y p) (tilt p);
                        have Phi := fun p xi eta =>
                          ∫ (y : Fin 2 → ℝ), Complex.exp (↑(xi * y 0 + eta * y 1) * Complex.I) ∂laws p;
                        ∃ lo hi,
                          0 < lo ∧
                            0 < hi ∧
                              (∀ p ∈ Theta,
                                  ∀ (xi eta : ℝ),
                                    lo * (xi ^ 2 + eta ^ 2) ≤
                                        xi ^ 2 * D p 0 0 + 2 * xi * eta * D p 0 1 + eta ^ 2 * D p 1 1 ∧
                                      xi ^ 2 * D p 0 0 + 2 * xi * eta * D p 0 1 + eta ^ 2 * D p 1 1 ≤
                                        hi * (xi ^ 2 + eta ^ 2)) ∧
                                ∃ E,
                                  (∀ (p : ↑Theta),
                                      (∀ (w : ℝ × ℝ),
                                          (E p) w =
                                            (A (↑p) 0 0 * w.1 + A (↑p) 0 1 * w.2,
                                              A (↑p) 1 0 * w.1 + A (↑p) 1 1 * w.2)) ∧
                                        ‖↑(E p)‖ ≤ 2 * (1 / √lo) ∧
                                          ‖↑(E p).symm‖ ≤ 4 * hi * (1 / √lo) ∧
                                            ∀ (x : ℝ),
                                              (E p) ((x, x ^ 2) - (means (↑p) 0, means (↑p) 1)) =
                                                (Y (↑p) x 0, Y (↑p) x 1)) ∧
                                    ∀ (epsilon : ℝ),
                                      0 < epsilon →
                                        ∃ N,
                                          2 ≤ N ∧
                                            ∀ (n : ℕ),
                                              N ≤ n →
                                                ∀ (p : ↑Theta),
                                                  have Qsample := MeasureTheory.Measure.pi fun x => tilt ↑p;
                                                  have M := fun z => ((∑ i, z i) / ↑n, Shao.energy z / ↑n);
                                                  have T := fun z => √↑n • (E p) (M z - (means (↑p) 0, means (↑p) 1));
                                                  have nu := MeasureTheory.Measure.map T Qsample;
                                                  have b := 1 / (2 * ↑n ^ 6);
                                                  have v := (2 * b).toNNReal;
                                                  have gamma :=
                                                    (ProbabilityTheory.gaussianReal 0 v).prod
                                                      (ProbabilityTheory.gaussianReal 0 v);
                                                  have smoothed :=
                                                    MeasureTheory.Measure.map (fun q => q.1 + q.2) (nu.prod gamma);
                                                  ∀ (aL aR qL qR : ℝ),
                                                    (means (↑p) 0, means (↑p) 1) ∈ Set.Icc aL aR ×ˢ Set.Icc qL qR →
                                                      aR - aL ≤ 1 / ↑n →
                                                        qR - qL ≤ 1 / ↑n →
                                                          have cellA :=
                                                            Set.Ioc aL aR ×ˢ Set.Ioc qL qR ∩
                                                              {w |
                                                                |w.1 - a0| ≤ stripRadius ∧
                                                                  0 ≤ w.1 ^ 2 / level ^ 2 - w.2 ∧
                                                                    w.1 ^ 2 / level ^ 2 - w.2 ≤ stripRadius};
                                                          have stdCell :=
                                                            (fun w => √↑n • (E p) (w - (means (↑p) 0, means (↑p) 1))) ''
                                                              cellA;
                                                          have g := fun z =>
                                                            Real.exp
                                                              (-√↑n *
                                                                ((↑p).1 * ((E p).symm z).1 +
                                                                  (↑p).2 * ((E p).symm z).2));
                                                          have f := stdCell.indicator g;
                                                          MeasureTheory.IsProbabilityMeasure nu ∧
                                                            MeasureTheory.IsProbabilityMeasure smoothed ∧
                                                              MeasureTheory.Integrable f nu ∧
                                                                MeasureTheory.Integrable f smoothed ∧
                                                                  ↑n ^ 3 *
                                                                      |∫ (z : ℝ × ℝ), f z ∂smoothed -
                                                                          ∫ (z : ℝ × ℝ), f z ∂nu| <
                                                                    epsilon
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
