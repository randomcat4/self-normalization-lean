DECL Shao.OQ3.boundary_laplace_error_envelope_integrable
TYPE ∀ (d CA CF a0 rho : ℝ),
  0 < d →
    MeasureTheory.Integrable
      (fun p =>
        (CA * (|p.1| + p.2) + |a0| * CF * (|p.1| ^ 3 + |p.1| * p.2 + p.2 ^ 2) + |a0| / rho * (|p.1| + p.2)) *
          Real.exp (-d * (p.2 + p.1 ^ 2)))
      (MeasureTheory.volume.prod (MeasureTheory.volume.restrict (Set.Ici 0)))
AXIOMS [propext, Classical.choice, Quot.sound]
DECL Shao.OQ3.boundary_laplace_monomial_integrable
TYPE ∀ (d : ℝ),
  0 < d →
    ∀ (m n : ℕ),
      MeasureTheory.Integrable (fun p => |p.1| ^ m * p.2 ^ n * Real.exp (-d * (p.2 + p.1 ^ 2)))
        (MeasureTheory.volume.prod (MeasureTheory.volume.restrict (Set.Ici 0)))
AXIOMS [propext, Classical.choice, Quot.sound]
