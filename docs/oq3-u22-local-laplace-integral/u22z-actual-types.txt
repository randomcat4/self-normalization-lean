DECL Shao.OQ3.boundary_laplace_cutoff_integral_error
TYPE ∀ (F A : ℝ × ℝ → ℝ) (lambda Lambda1 rho CF CA c d : ℝ),
  0 ≤ CF →
    0 ≤ CA →
      0 < rho →
        0 < d →
          d ≤ c →
            d ≤ lambda →
              d ≤ Lambda1 / 2 →
                (∀ (r z : ℝ),
                    |r| ≤ rho →
                      0 ≤ z →
                        z ≤ rho →
                          |F (r, z) - F (0, 0) - lambda * z - Lambda1 / 2 * r ^ 2| ≤ CF * (|r| ^ 3 + |r| * z + z ^ 2)) →
                  (∀ (r z : ℝ), |r| ≤ rho → 0 ≤ z → z ≤ rho → c * (z + r ^ 2) ≤ F (r, z) - F (0, 0)) →
                    (∀ (r z : ℝ), |r| ≤ rho → 0 ≤ z → z ≤ rho → |A (r, z) - A (0, 0)| ≤ CA * (|r| + z)) →
                      (∀ (t : ℝ),
                          0 < t →
                            t ≤ 1 →
                              MeasureTheory.AEStronglyMeasurable
                                (fun p =>
                                  if |t * p.1| ≤ rho ∧ t ^ 2 * p.2 ≤ rho then
                                    A (t * p.1, t ^ 2 * p.2) *
                                      Real.exp (-((F (t * p.1, t ^ 2 * p.2) - F (0, 0)) / t ^ 2))
                                  else 0)
                                (MeasureTheory.volume.prod (MeasureTheory.volume.restrict (Set.Ici 0)))) →
                        have mu := MeasureTheory.volume.prod (MeasureTheory.volume.restrict (Set.Ici 0));
                        have K := fun t p =>
                          if |t * p.1| ≤ rho ∧ t ^ 2 * p.2 ≤ rho then
                            A (t * p.1, t ^ 2 * p.2) * Real.exp (-((F (t * p.1, t ^ 2 * p.2) - F (0, 0)) / t ^ 2))
                          else 0;
                        ∃ C,
                          0 < C ∧
                            ∀ (t : ℝ),
                              0 < t →
                                t ≤ 1 →
                                  MeasureTheory.Integrable (K t) mu ∧
                                    |∫ (p : ℝ × ℝ), K t p ∂mu - A (0, 0) * (√(2 * Real.pi / Lambda1) / lambda)| ≤ C * t
AXIOMS [propext, Classical.choice, Quot.sound]
DECL Shao.OQ3.boundary_laplace_leading_integral
TYPE ∀ (lambda Lambda1 a0 : ℝ),
  0 < lambda →
    0 < Lambda1 →
      have mu := MeasureTheory.volume.prod (MeasureTheory.volume.restrict (Set.Ici 0));
      have L := fun p => a0 * Real.exp (-(lambda * p.2 + Lambda1 / 2 * p.1 ^ 2));
      MeasureTheory.Integrable L mu ∧ ∫ (p : ℝ × ℝ), L p ∂mu = a0 * (√(2 * Real.pi / Lambda1) / lambda)
AXIOMS [propext, Classical.choice, Quot.sound]
