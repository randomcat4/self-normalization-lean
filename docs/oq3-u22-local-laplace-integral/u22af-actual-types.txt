DECL Shao.OQ3.open_domain_boundary_laplace_original_strip
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu],
  ∀ theta ∈ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)),
    0 < (Shao.OQ3.parabolaCovariance (mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2)).det →
      ∀ (level a0 : ℝ),
        0 < level →
          0 < a0 →
            theta.2 < 0 →
              (∫ (x : ℝ), x ∂mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2,
                    ∫ (x : ℝ), x ^ 2 ∂mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2) =
                  (a0, a0 ^ 2 / level ^ 2) →
                theta.1 = -2 * a0 * theta.2 / level ^ 2 →
                  have Omega := interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2));
                  have nu := fun p => mu.tilted fun x => p.1 * x + p.2 * x ^ 2;
                  have means := fun p => (∫ (x : ℝ), x ∂nu p, ∫ (x : ℝ), x ^ 2 ∂nu p);
                  have D := Shao.OQ3.parabolaCovariance (nu theta);
                  have pr0 := ContinuousLinearMap.fst ℝ ℝ ℝ;
                  have pr1 := ContinuousLinearMap.snd ℝ ℝ ℝ;
                  have row0 := D 0 0 • pr0 + D 0 1 • pr1;
                  have row1 := D 0 1 • pr0 + D 1 1 • pr1;
                  ∃ L e,
                    ↑L = row0.prod row1 ∧
                      ↑e = means ∧
                        theta ∈ e.source ∧
                          e.source ⊆ Omega ∧
                            have K := fun p => Real.log (∫ (x : ℝ), Real.exp (p.1 * x + p.2 * x ^ 2) ∂mu);
                            have J := fun w => (↑e.symm w).1 * w.1 + (↑e.symm w).2 * w.2 - K (↑e.symm w);
                            have B := fun w => (√(Shao.OQ3.parabolaCovariance (nu (↑e.symm w))).det)⁻¹;
                            have v := (1, 2 * a0 / level ^ 2);
                            have Lambda1 := 2 * theta.2 / level ^ 2 + (L.symm v).1 + 2 * a0 / level ^ 2 * (L.symm v).2;
                            have C0 := (√D.det)⁻¹ * (√(2 * Real.pi / Lambda1) / -theta.2);
                            have J0 := J (a0, a0 ^ 2 / level ^ 2);
                            have m0 := MeasureTheory.volume.prod MeasureTheory.volume;
                            0 < -theta.2 ∧
                              (0 < Lambda1 →
                                0 < C0 ∧
                                  ∃ rho C,
                                    0 < rho ∧
                                      0 < C ∧
                                        have V :=
                                          {w |
                                            |w.1 - a0| ≤ rho ∧
                                              w.1 ^ 2 / level ^ 2 - rho ≤ w.2 ∧ w.2 ≤ w.1 ^ 2 / level ^ 2};
                                        (∀ w ∈ V,
                                            w ∈ e.target ∧
                                              ↑e.symm w ∈ e.source ∧
                                                ↑e.symm w ∈ Omega ∧
                                                  means (↑e.symm w) = w ∧
                                                    0 < (Shao.OQ3.parabolaCovariance (nu (↑e.symm w))).det ∧
                                                      Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2))
                                                          w =
                                                        ↑(J w)) ∧
                                          ∀ (t : ℝ),
                                            0 < t →
                                              t ≤ 1 →
                                                have W := fun w => B w * Real.exp (-J w / t ^ 2);
                                                have Ln := 1 / (2 * Real.pi * t ^ 2) * ∫ (w : ℝ × ℝ) in V, W w ∂m0;
                                                have main := t * Real.exp (-J0 / t ^ 2) * (C0 / (2 * Real.pi));
                                                MeasureTheory.IntegrableOn W V m0 ∧ 0 < main ∧ |Ln / main - 1| ≤ C * t)
AXIOMS [propext, Classical.choice, Quot.sound]
