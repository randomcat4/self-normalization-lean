DECL Shao.OQ3.pair_projection_absolute_moment_bound
TYPE ∀ (law : MeasureTheory.Measure (Fin 2 → ℝ)) (m : ℕ),
  MeasureTheory.Integrable (fun y => √(y 0 ^ 2 + y 1 ^ 2) ^ m) law →
    ∀ (u v : ℝ),
      have f := fun y => |u * y 0 + v * y 1| ^ m;
      MeasureTheory.Integrable f law ∧
        0 ≤ ∫ (y : Fin 2 → ℝ), f y ∂law ∧
          ∫ (y : Fin 2 → ℝ), f y ∂law ≤ √(u ^ 2 + v ^ 2) ^ m * ∫ (y : Fin 2 → ℝ), √(y 0 ^ 2 + y 1 ^ 2) ^ m ∂law
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
