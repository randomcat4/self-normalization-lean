DECL Shao.OQ3.complex_power_difference_bound_on_unit_ball
TYPE ∀ (z w : ℂ), ‖z‖ ≤ 1 → ‖w‖ ≤ 1 → ∀ (j : ℕ), ‖z ^ j - w ^ j‖ ≤ ↑j * ‖z - w‖
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.complex_log_taylor_perturbation_bound
TYPE ∀ (n : ℕ) (z w : ℂ),
  ‖z‖ ≤ 1 / 2 →
    ‖w‖ ≤ 1 → ‖Complex.log (1 + z) - Complex.logTaylor (n + 1) w‖ ≤ 2 * ‖z‖ ^ (n + 1) / ↑(n + 1) + ↑(n + 1) * ‖z - w‖
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.complex_log_taylor_difference_bound_on_unit_ball
TYPE ∀ (n : ℕ) (z w : ℂ), ‖z‖ ≤ 1 → ‖w‖ ≤ 1 → ‖Complex.logTaylor n z - Complex.logTaylor n w‖ ≤ ↑n * ‖z - w‖
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
