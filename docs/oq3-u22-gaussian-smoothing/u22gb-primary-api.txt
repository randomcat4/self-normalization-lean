345: /-! ### The Bochner integral on a product -/
440:   `integrable_prod_iff` can be useful to show that the function in question in integrable.
441:   `MeasureTheory.Integrable.integral_prod_right` is useful to show that the inner integral
442:   of the right-hand side is integrable. -/
443: theorem integral_prod (f : α × β → E) (hf : Integrable f (μ.prod ν)) :
444:     ∫ z, f z ∂μ.prod ν = ∫ x, ∫ y, f (x, y) ∂ν ∂μ := by
445:   by_cases hE : CompleteSpace E; swap; · simp only [integral, dite_eq_right hE]
446:   revert f
447:   apply Integrable.induction
448:   · intro c s hs h2s
449:     simp_rw [integral_indicator hs, ← indicator_comp_right, Function.comp_def,
450:       integral_indicator (measurable_prodMk_left hs), setIntegral_const, integral_smul_const,
451:       measureReal_def,
452:       integral_toReal (measurable_measure_prodMk_left hs).aemeasurable
453:         (ae_measure_lt_top hs h2s.ne)]
454:     rw [Measure.prod_apply hs]
455:   · rintro f g - i_f i_g hf hg
456:     simp_rw [integral_add' i_f i_g, integral_integral_add' i_f i_g, hf, hg]
