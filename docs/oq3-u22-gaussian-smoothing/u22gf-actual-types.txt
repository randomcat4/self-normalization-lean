DECL Shao.OQ3.independent_sum_source_set_le
TYPE ∀ (nu eta : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure nu]
  [MeasureTheory.IsProbabilityMeasure eta] (s t : Set (ℝ × ℝ)),
  MeasurableSet s →
    MeasurableSet t →
      ∀ (eps : ℝ),
        (∀ x ∈ s, ∀ (y : ℝ × ℝ), ‖y‖ ≤ eps → x + y ∈ t) →
          have rho := MeasureTheory.Measure.map (fun q => q.1 + q.2) (nu.prod eta);
          nu s ≤ rho t + eta {y | eps < ‖y‖}
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.independent_sum_image_set_le
TYPE ∀ (nu eta : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure nu]
  [MeasureTheory.IsProbabilityMeasure eta] (s t : Set (ℝ × ℝ)),
  MeasurableSet s →
    MeasurableSet t →
      ∀ (eps : ℝ),
        (∀ (x y : ℝ × ℝ), x + y ∈ s → ‖y‖ ≤ eps → x ∈ t) →
          have rho := MeasureTheory.Measure.map (fun q => q.1 + q.2) (nu.prod eta);
          rho s ≤ nu t + eta {y | eps < ‖y‖}
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
