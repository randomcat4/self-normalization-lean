179: /-- The closed `δ`-thickening `Metric.cthickening δ E` of a subset `E` in a pseudo emetric space
180: consists of those points that are at infimum distance at most `δ` from `E`. -/
181: def cthickening (δ : ℝ) (E : Set α) : Set α :=
182:   { x : α | infEDist x E ≤ ENNReal.ofReal δ }

185: theorem mem_cthickening_iff : x ∈ cthickening δ s ↔ infEDist x s ≤ ENNReal.ofReal δ :=
186:   Iff.rfl

188: /-- An exterior point of a subset `E` (i.e., a point outside the closure of `E`) is not in the
189: closed `δ`-thickening of `E` for small enough positive `δ`. -/
190: lemma eventually_notMem_cthickening_of_infEDist_pos {E : Set α} {x : α} (h : x ∉ closure E) :
191:     ∀ᶠ δ in 𝓝 (0 : ℝ), x ∉ Metric.cthickening δ E := by
192:   obtain ⟨ε, ⟨ε_pos, ε_lt⟩⟩ := exists_real_pos_lt_infEDist_of_notMem_closure h
193:   filter_upwards [eventually_lt_nhds ε_pos] with δ hδ
194:   simp only [cthickening, mem_ofPred_eq, not_le]
195:   exact ((ofReal_lt_ofReal_iff ε_pos).mpr hδ).trans ε_lt

197: theorem mem_cthickening_of_edist_le (x y : α) (δ : ℝ) (E : Set α) (h : y ∈ E)
198:     (h' : edist x y ≤ ENNReal.ofReal δ) : x ∈ cthickening δ E :=
199:   (infEDist_le_edist_of_mem h).trans h'

201: theorem mem_cthickening_of_dist_le {α : Type*} [PseudoMetricSpace α] (x y : α) (δ : ℝ) (E : Set α)
202:     (h : y ∈ E) (h' : dist x y ≤ δ) : x ∈ cthickening δ E := by

297: /-- The closed thickening of a set contains the closure of the set. -/
298: theorem closure_subset_cthickening (δ : ℝ) (E : Set α) : closure E ⊆ cthickening δ E := by
299:   rw [← cthickening_of_nonpos (min_le_right δ 0)]
300:   exact cthickening_mono (min_le_left δ 0) E

302: /-- The (open) thickening of a set contains the closure of the set. -/
303: theorem closure_subset_thickening {δ : ℝ} (δ_pos : 0 < δ) (E : Set α) :
304:     closure E ⊆ thickening δ E := by
305:   rw [← cthickening_zero]
306:   exact cthickening_subset_thickening' δ_pos δ_pos E

308: /-- A set is contained in its own (open) thickening. -/
