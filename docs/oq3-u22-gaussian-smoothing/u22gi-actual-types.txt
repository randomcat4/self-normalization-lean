DECL Shao.OQ3.independent_sum_cthickening_real_sandwich
TYPE ∀ (nu eta : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure nu]
  [MeasureTheory.IsProbabilityMeasure eta] (s : Set (ℝ × ℝ)),
  MeasurableSet s →
    ∀ (eps : ℝ),
      have rho := MeasureTheory.Measure.map (fun q => q.1 + q.2) (nu.prod eta);
      rho.real (Metric.cthickening eps sᶜ)ᶜ - eta.real {y | eps < ‖y‖} ≤ nu.real s ∧
        nu.real s ≤ rho.real (Metric.cthickening eps s) + eta.real {y | eps < ‖y‖}
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.independent_sum_cthickening_bounds
TYPE ∀ (nu eta : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure nu]
  [MeasureTheory.IsProbabilityMeasure eta] (s : Set (ℝ × ℝ)),
  MeasurableSet s →
    ∀ (eps : ℝ),
      have rho := MeasureTheory.Measure.map (fun q => q.1 + q.2) (nu.prod eta);
      nu s ≤ rho (Metric.cthickening eps s) + eta {y | eps < ‖y‖} ∧
        rho (Metric.cthickening eps sᶜ)ᶜ ≤ nu s + eta {y | eps < ‖y‖}
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
