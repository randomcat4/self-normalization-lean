DECL Shao.OQ3.cgf_third_derivative_actual_moments
TYPE ∀ {Ω : Type u_1} [inst : MeasurableSpace Ω] (mu : MeasureTheory.Measure Ω) [MeasureTheory.IsProbabilityMeasure mu]
  (X : Ω → ℝ),
  ∀ v ∈ interior (ProbabilityTheory.integrableExpSet X mu),
    have M := fun k => ∫ (w : Ω), X w ^ k * Real.exp (v * X w) ∂mu;
    iteratedDeriv 3 (ProbabilityTheory.cgf X mu) v = M 3 / M 0 - 3 * M 2 * M 1 / M 0 ^ 2 + 2 * M 1 ^ 3 / M 0 ^ 3
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.cgf_fourth_derivative_actual_moments
TYPE ∀ {Ω : Type u_1} [inst : MeasurableSpace Ω] (mu : MeasureTheory.Measure Ω) [MeasureTheory.IsProbabilityMeasure mu]
  (X : Ω → ℝ),
  ∀ v ∈ interior (ProbabilityTheory.integrableExpSet X mu),
    have M := fun k => ∫ (w : Ω), X w ^ k * Real.exp (v * X w) ∂mu;
    iteratedDeriv 4 (ProbabilityTheory.cgf X mu) v =
      M 4 / M 0 - 4 * M 3 * M 1 / M 0 ^ 2 - 3 * M 2 ^ 2 / M 0 ^ 2 + 12 * M 2 * M 1 ^ 2 / M 0 ^ 3 - 6 * M 1 ^ 4 / M 0 ^ 4
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.centered_cgf_third_fourth_derivatives
TYPE ∀ {Ω : Type u_1} [inst : MeasurableSpace Ω] (mu : MeasureTheory.Measure Ω) [MeasureTheory.IsProbabilityMeasure mu]
  (X : Ω → ℝ),
  0 ∈ interior (ProbabilityTheory.integrableExpSet X mu) →
    ∫ (w : Ω), X w ∂mu = 0 →
      iteratedDeriv 3 (ProbabilityTheory.cgf X mu) 0 = ∫ (w : Ω), X w ^ 3 ∂mu ∧
        iteratedDeriv 4 (ProbabilityTheory.cgf X mu) 0 = ∫ (w : Ω), X w ^ 4 ∂mu - 3 * (∫ (w : Ω), X w ^ 2 ∂mu) ^ 2
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
