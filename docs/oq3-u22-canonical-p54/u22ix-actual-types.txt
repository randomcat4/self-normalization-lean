DECL Shao.OQ3.covariance_row_equivalence_inverse_quadratic
TYPE ∀ (D : Matrix (Fin 2) (Fin 2) ℝ),
  D.det ≠ 0 →
    ∀ (L : (ℝ × ℝ) ≃L[ℝ] ℝ × ℝ),
      (∀ (w : ℝ × ℝ), L w = (D 0 0 * w.1 + D 0 1 * w.2, D 1 0 * w.1 + D 1 1 * w.2)) →
        ∀ (w : ℝ × ℝ),
          D⁻¹.mulVec ![w.1, w.2] = ![(L.symm w).1, (L.symm w).2] ∧
            w.1 * (L.symm w).1 + w.2 * (L.symm w).2 = ![w.1, w.2] ⬝ᵥ D⁻¹.mulVec ![w.1, w.2]
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
