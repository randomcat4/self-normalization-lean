DECL Shao.OQ3.compact_open_domain_six_order_gaussian_hermite_integral
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure mu] (Theta : Set (ℝ × ℝ)),
  IsCompact Theta →
    Theta ⊆ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)) →
      ∀ (beta c R : ℝ),
        0 ≤ beta →
          0 < c →
            0 ≤ R →
              (∀ p ∈ Theta, 0 < (Shao.OQ3.parabolaCovariance (mu.tilted fun x => p.1 * x + p.2 * x ^ 2)).det) →
                (∀ p ∈ Theta,
                    ∀ (xi eta : ℝ),
                      R < √(xi ^ 2 + eta ^ 2) →
                        ‖Shao.OQ3.jointCharacteristic (mu.tilted fun x => p.1 * x + p.2 * x ^ 2) xi eta‖ ≤
                          1 - c * (√(xi ^ 2 + eta ^ 2)).rpow (-beta)) →
                  have tilt := fun p => mu.tilted fun x => p.1 * x + p.2 * x ^ 2;
                  have D := fun p => Shao.OQ3.parabolaCovariance (tilt p);
                  have s := fun p => √(D p 0 0 * D p 1 1 - D p 0 1 ^ 2);
                  have t := fun p => √(D p 0 0 + D p 1 1 + 2 * s p);
                  have A := fun p =>
                    ![![(D p 1 1 + s p) / (s p * t p), -D p 0 1 / (s p * t p)],
                      ![-D p 0 1 / (s p * t p), (D p 0 0 + s p) / (s p * t p)]];
                  have means := fun p => ![∫ (x : ℝ), x ∂tilt p, ∫ (x : ℝ), x ^ 2 ∂tilt p];
                  have Y := fun p x => (A p).mulVec (![x, x ^ 2] - means p);
                  have laws := fun p => MeasureTheory.Measure.map (Y p) (tilt p);
                  have Phi := fun p xi eta =>
                    ∫ (y : Fin 2 → ℝ), Complex.exp (↑(xi * y 0 + eta * y 1) * Complex.I) ∂laws p;
                  ∃ E,
                    (∀ (p : ↑Theta) (w : ℝ × ℝ),
                        (E p) w = (A (↑p) 0 0 * w.1 + A (↑p) 0 1 * w.2, A (↑p) 1 0 * w.1 + A (↑p) 1 1 * w.2)) ∧
                      ∀ (n : ℕ),
                        1 ≤ n →
                          ∀ (p : ↑Theta),
                            have Qsample := MeasureTheory.Measure.pi fun x => tilt ↑p;
                            have M := fun z => ((∑ i, z i) / ↑n, Shao.energy z / ↑n);
                            have T := fun z => √↑n • (E p) (M z - (means (↑p) 0, means (↑p) 1));
                            have rho := MeasureTheory.Measure.map T Qsample;
                            MeasureTheory.IsProbabilityMeasure rho ∧
                              have kappa := fun z j =>
                                iteratedDeriv j (ProbabilityTheory.cgf (fun y => z.1 * y 0 + z.2 * y 1) (laws ↑p)) 0;
                              have a := fun z j =>
                                Complex.I ^ (↑j + 3) / ↑(↑j + 3).factorial * ↑(kappa z (↑j + 3)) / ↑√↑n ^ (↑j + 1);
                              have S := fun z =>
                                ∑ q,
                                  1 / ↑(↑q).factorial *
                                    ∑ sigma, if ∑ i, (↑(sigma i) + 1) ≤ 6 then ∏ i, a z (sigma i) else 0;
                              have S1 := fun z =>
                                ∑ q,
                                  1 / ↑(↑q).factorial *
                                    ∑ sigma, if ∑ i, (↑(sigma i) + 1) = 1 then ∏ i, a z (sigma i) else 0;
                              ∃ P,
                                (∀ (z : ℝ × ℝ),
                                    MvPolynomial.eval₂ Complex.ofRealHom ![Complex.I * ↑z.1, Complex.I * ↑z.2] P =
                                      S z) ∧
                                  (∀ (x : ℝ × ℝ),
                                      MeasureTheory.Integrable
                                          (fun z =>
                                            Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * S z *
                                              Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2))
                                          MeasureTheory.volume ∧
                                        ∫ (z : ℝ × ℝ),
                                            Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * S z *
                                              Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) =
                                          ↑((2 * Real.pi *
                                                ∑ d ∈ P.support,
                                                  P.coeff d *
                                                      Polynomial.eval x.1 (Shao.OQ3.gaussianHermitePolynomial (d 0)) *
                                                    Polynomial.eval x.2 (Shao.OQ3.gaussianHermitePolynomial (d 1))) *
                                              Real.exp (-(x.1 ^ 2 + x.2 ^ 2) / 2))) ∧
                                    MeasureTheory.Integrable (fun z => S1 z * Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2))
                                        MeasureTheory.volume ∧
                                      ∫ (z : ℝ × ℝ), S1 z * Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) = 0
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
