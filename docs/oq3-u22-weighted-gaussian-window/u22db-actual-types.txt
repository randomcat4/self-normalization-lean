DECL Shao.OQ3.pair_characteristic_bounded_frequency_weighted_integral_bound
TYPE ∀ (rho : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure rho] (D : Set (ℝ × ℝ)),
  MeasurableSet D →
    ∀ (K epsilon : ℝ),
      0 ≤ K →
        0 ≤ epsilon →
          (∀ z ∈ D, √(z.1 ^ 2 + z.2 ^ 2) ≤ K) →
            (∀ z ∈ D, ‖∫ (w : ℝ × ℝ), Complex.exp (↑(z.1 * w.1 + z.2 * w.2) * Complex.I) ∂rho‖ ≤ epsilon) →
              ∀ (k : ℕ),
                have f := fun z =>
                  √(z.1 ^ 2 + z.2 ^ 2) ^ k * ‖∫ (w : ℝ × ℝ), Complex.exp (↑(z.1 * w.1 + z.2 * w.2) * Complex.I) ∂rho‖;
                MeasureTheory.IntegrableOn f D MeasureTheory.volume ∧
                  0 ≤ ∫ (z : ℝ × ℝ) in D, f z ∧ ∫ (z : ℝ × ℝ) in D, f z ≤ 4 * K ^ (k + 2) * epsilon
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
