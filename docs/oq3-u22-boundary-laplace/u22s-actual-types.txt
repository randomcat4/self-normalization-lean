DECL Shao.OQ3.exists_positive_boundary_lower_bound_of_contDiffAt
TYPE ∀ (F : ℝ × ℝ → ℝ),
  ContDiffAt ℝ 3 F (0, 0) →
    deriv (fun t => F (t, 0)) 0 = 0 →
      0 < (fderiv ℝ F (0, 0)) (0, 1) →
        0 < deriv (deriv fun t => F (t, 0)) 0 →
          ∃ delta c,
            0 < delta ∧ 0 < c ∧ ∀ (r z : ℝ), |r| ≤ delta → 0 ≤ z → z ≤ delta → c * (z + r ^ 2) ≤ F (r, z) - F (0, 0)
AXIOMS [propext, Classical.choice, Quot.sound]
DECL Shao.OQ3.exists_positive_boundary_lower_bound_of_taylor
TYPE ∀ (F : ℝ × ℝ → ℝ) (lambda Lambda1 rho C : ℝ),
  0 < lambda →
    0 < Lambda1 →
      0 < rho →
        0 < C →
          (∀ (r z : ℝ),
              |r| ≤ rho →
                0 ≤ z →
                  z ≤ rho →
                    |F (r, z) - F (0, 0) - lambda * z - Lambda1 / 2 * r ^ 2| ≤ C * (|r| ^ 3 + |r| * z + z ^ 2)) →
            ∃ delta c,
              0 < delta ∧
                delta ≤ rho ∧
                  0 < c ∧ ∀ (r z : ℝ), |r| ≤ delta → 0 ≤ z → z ≤ delta → c * (z + r ^ 2) ≤ F (r, z) - F (0, 0)
AXIOMS [propext, Classical.choice, Quot.sound]
