DECL Shao.OQ3.exists_uniform_boundary_taylor_bound_at_zero
TYPE ∀ (F : ℝ × ℝ → ℝ),
  ContDiffAt ℝ 3 F (0, 0) →
    ∃ rho C,
      0 < rho ∧
        0 < C ∧
          ∀ (r z : ℝ),
            |r| ≤ rho →
              0 ≤ z →
                z ≤ rho →
                  |F (r, z) - F (0, 0) - deriv (fun t => F (t, 0)) 0 * r - (fderiv ℝ F (0, 0)) (0, 1) * z -
                        deriv (deriv fun t => F (t, 0)) 0 / 2 * r ^ 2| ≤
                    C * (|r| ^ 3 + |r| * z + z ^ 2)
AXIOMS [propext, Classical.choice, Quot.sound]
DECL Shao.OQ3.exists_uniform_normal_increment_bound_at_zero
TYPE ∀ (F : ℝ × ℝ → ℝ),
  ContDiffAt ℝ 2 F (0, 0) →
    ∃ rho C,
      0 < rho ∧
        0 < C ∧
          ∀ (r z : ℝ),
            |r| ≤ rho → 0 ≤ z → z ≤ rho → |F (r, z) - F (r, 0) - (fderiv ℝ F (0, 0)) (0, 1) * z| ≤ C * (|r| * z + z ^ 2)
AXIOMS [propext, Classical.choice, Quot.sound]
