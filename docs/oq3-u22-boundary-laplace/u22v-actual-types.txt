DECL Shao.OQ3.abs_weighted_exp_neg_sub_le
TYPE ∀ (a b w w0 d E Q : ℝ),
  d ≤ a → d ≤ b → |a - b| ≤ E → |w - w0| ≤ Q → |w * Real.exp (-a) - w0 * Real.exp (-b)| ≤ (Q + |w0| * E) * Real.exp (-d)
AXIOMS [propext, Classical.choice, Quot.sound]
DECL Shao.OQ3.abs_exp_neg_sub_exp_neg_le
TYPE ∀ (a b d : ℝ), d ≤ a → d ≤ b → |Real.exp (-a) - Real.exp (-b)| ≤ Real.exp (-d) * |a - b|
AXIOMS [propext, Classical.choice, Quot.sound]
