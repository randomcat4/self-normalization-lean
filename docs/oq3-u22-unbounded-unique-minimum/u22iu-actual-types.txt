DECL Shao.OQ3.exists_self_normalized_high_energy_arbitrary_exponential_bound
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure mu] (level R : ℝ),
  0 < level →
    0 < R →
      ∃ Q,
        0 < Q ∧
          ∀ (n : ℕ),
            1 ≤ n →
              have P := MeasureTheory.Measure.pi fun x => mu;
              have T := {z | level * √↑n ≤ if Shao.energy z = 0 then 0 else (∑ i, z i) / √(Shao.energy z)};
              P.real (T ∩ {z | Q * ↑n ≤ Shao.energy z}) ≤ Real.exp (-R * ↑n)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
