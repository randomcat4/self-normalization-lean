DECL Shao.OQ3.actual_exterior_exponential_bound_of_unique_minimum
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (level a0 rho J0 : ℝ),
  0 < level →
    0 < rho →
      have I := Shao.Probability.bivariateLegendreRate mu fun x => (x, x ^ 2);
      have center := (a0, a0 ^ 2 / level ^ 2);
      have Tail := {w | 0 < w.2 ∧ level * √w.2 ≤ w.1};
      have U := {w | |w.1 - a0| ≤ rho ∧ |w.1 ^ 2 / level ^ 2 - w.2| ≤ rho};
      I center = ↑J0 →
        (∀ w ∈ Tail, I center ≤ I w ∧ (I w = I center ↔ w = center)) →
          ∃ delta,
            0 < delta ∧
              ∃ C,
                0 < C ∧
                  ∀ (n : ℕ),
                    1 ≤ n →
                      have P := MeasureTheory.Measure.pi fun x => mu;
                      have M := fun z => ((∑ i, z i) / ↑n, Shao.energy z / ↑n);
                      P.real (M ⁻¹' (Tail \ U)) ≤ C * Real.exp (-↑n * (J0 + delta))
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.Probability.bivariateLegendreRate.congr_simp
TYPE ∀ {Ω : Type u_1} [inst : MeasurableSpace Ω] (μ μ_1 : MeasureTheory.Measure Ω) (e_μ : μ = μ_1) [inst_1 : NeZero μ]
  (W W_1 : Ω → ℝ × ℝ),
  W = W_1 →
    ∀ (z z_1 : ℝ × ℝ),
      z = z_1 → Shao.Probability.bivariateLegendreRate μ W z = Shao.Probability.bivariateLegendreRate μ_1 W_1 z_1
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
