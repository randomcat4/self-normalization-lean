DECL Shao.OQ3.exists_large_iid_count_arbitrary_exponential_bound
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure mu] (alpha R : ℝ),
  0 < alpha →
    0 < R →
      ∃ B,
        0 < B ∧
          ∀ (n : ℕ),
            have P := MeasureTheory.Measure.pi fun x => mu;
            have LargeCount := fun z => {i | B < |z i|}.card;
            P.real {z | alpha * ↑n ≤ ↑(LargeCount z)} ≤ Real.exp (-R * ↑n)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
