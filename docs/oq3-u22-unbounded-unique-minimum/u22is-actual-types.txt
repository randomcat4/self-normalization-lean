DECL Shao.OQ3.self_normalized_high_energy_many_large
TYPE ∀ (level B : ℝ),
  0 < level →
    0 < B →
      ∀ (n : ℕ),
        1 ≤ n →
          ∀ (z : Shao.Path n),
            (level * √↑n ≤ if Shao.energy z = 0 then 0 else (∑ i, z i) / √(Shao.energy z)) →
              4 * B ^ 2 / level ^ 2 * ↑n ≤ Shao.energy z →
                have Large := {i | B < |z i|};
                level ^ 2 / 4 * ↑n ≤ ↑Large.card
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
