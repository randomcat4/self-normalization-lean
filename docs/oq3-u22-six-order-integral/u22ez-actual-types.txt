DECL Shao.OQ3.pair_directional_moment_binomial_continuous
TYPE ∀ (mu : MeasureTheory.Measure (Fin 2 → ℝ)) (k : ℕ),
  MeasureTheory.Integrable (fun y => √(y 0 ^ 2 + y 1 ^ 2) ^ k) mu →
    have M := fun z => ∫ (y : Fin 2 → ℝ), (z.1 * y 0 + z.2 * y 1) ^ k ∂mu;
    have b := fun j => ∫ (y : Fin 2 → ℝ), y 0 ^ j * y 1 ^ (k - j) ∂mu;
    (∀ j ≤ k, MeasureTheory.Integrable (fun y => y 0 ^ j * y 1 ^ (k - j)) mu) ∧
      (∀ (z : ℝ × ℝ), M z = ∑ j ∈ Finset.range (k + 1), z.1 ^ j * z.2 ^ (k - j) * ↑(k.choose j) * b j) ∧ Continuous M
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
