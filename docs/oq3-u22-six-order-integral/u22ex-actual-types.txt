DECL Shao.OQ3.complex_six_letter_cutoff_word_recurrence
TYPE ∀ (a : Fin 6 → ℂ),
  have W := fun q d => ∑ sigma, if ∑ i, (↑(sigma i) + 1) ≤ d then ∏ i, a (sigma i) else 0;
  (∀ (d : ℕ), W 0 d = 1) ∧ ∀ (q d : ℕ), W (q + 1) d = ∑ j, if ↑j + 1 ≤ d then a j * W q (d - (↑j + 1)) else 0
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem

  have a := fun j => A j * t ^ (↑j + 1);
  have P := fun b => ∑ q, 1 / ↑(↑q).factorial * ∑ sigma, if ∑ i, (↑(sigma i) + 1) ≤ 6 then ∏ i, b (sigma i) else 0;
  have B1 := A 0;
  have B2 := A 1 + A 0 ^ 2 / 2;

          A 0 ^ 2 * A 1 ^ 2 / 4 +
        A 0 ^ 4 * A 1 / 24 +
      A 0 ^ 6 / 720;
  P a = 1 + B1 * t + B2 * t ^ 2 + B3 * t ^ 3 + B4 * t ^ 4 + B5 * t ^ 5 + B6 * t ^ 6
