DECL Shao.OQ3.circle_phase_chord_nearest_lattice
TYPE ∀ (theta phi : ℝ),
  have k := round ((theta - phi) / (2 * Real.pi));
  have d := |theta - phi - 2 * Real.pi * ↑k|;
  d ≤ Real.pi ∧
    (∀ (m : ℤ), d ≤ |theta - phi - 2 * Real.pi * ↑m|) ∧
      2 / Real.pi * d ≤ ‖Complex.exp (↑theta * Complex.I) - Complex.exp (↑phi * Complex.I)‖ ∧
        ‖Complex.exp (↑theta * Complex.I) - Complex.exp (↑phi * Complex.I)‖ ≤ d
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
