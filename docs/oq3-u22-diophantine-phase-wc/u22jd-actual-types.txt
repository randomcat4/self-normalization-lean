DECL Shao.OQ3.symmetric_eight_atom_law_and_phase_gap
TYPE ∀ (a : Fin 8 → ℝ),
  have A := Fin.append a fun i => -a i;
  have mu := MeasureTheory.Measure.sum fun k => ENNReal.ofReal (1 / 16) • MeasureTheory.Measure.dirac (A k);
  have B := ∑ i, |a i|;
  MeasureTheory.IsProbabilityMeasure mu ∧
    MeasureTheory.Integrable (fun x => x) mu ∧
      ∫ (x : ℝ), x ∂mu = 0 ∧
        0 ≤ B ∧
          (∀ᵐ (x : ℝ) ∂mu, |x| ≤ B) ∧
            ∀ (xi eta : ℝ),
              Shao.OQ3.jointCharacteristic mu xi eta =
                  (1 / 16) • (∑ i, Shao.OQ3.jointPhase xi eta (a i) + ∑ i, Shao.OQ3.jointPhase xi eta (-a i)) ∧
                ∀ (i j : Fin (8 + 8)),
                  1 / 1024 * ‖Shao.OQ3.jointPhase xi eta (A i) - Shao.OQ3.jointPhase xi eta (A j)‖ ^ 2 ≤
                    1 - ‖Shao.OQ3.jointCharacteristic mu xi eta‖
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
