    (r : Ω → Ω → ℝ) (x : Fin n → Ω)
    (hn : 4 ≤ n) (hm : 4 ≤ B.card) (j : Fin B.card) :

      (∑ i ∈ B.erase ((e j).val), r (x ((e j).val)) (x i)) /
        ((n : ℝ) - 2)
