theorem originalUpper_scaled_background {n : ℕ} {Ω : Type u}
    (B : Finset (Fin n)) (e : Fin B.card ≃ {i : Fin n // i ∈ B})
    (r : Ω → Ω → ℝ) (x : Fin n → Ω)
    (hsym : ∀ a b : Ω, r a b = r b a) :
    Shao.OQ6.CanonicalDoubleGhostScalarMoments.originalUpper
        B.card (scaledResidual B r) (backgroundSample B e x) =
      backgroundScale B * blockPairSum B
        (fun i j : Fin n => r (x i) (x j))
