theorem physicalStatistic_eq_projection_ratio (n : ℕ) {Ω : Type u}
    (g : Ω → ℝ) (r : Ω → Ω → ℝ) (x : Fin n → Ω)
    (hsym : ∀ a b : Ω, r a b = r b a) (hn : 4 ≤ n) :
    physicalStatistic n g r x =
      (Shao.OQ5.P51.projectionW n g x +
        Shao.OQ5.P51.residualStatisticD n r x) /
        Real.sqrt (Shao.OQ5.P51.centeredSq n g r x)
