theorem numerator_background {n : ℕ} {Ω : Type u}
    (B : Finset (Fin n)) (e : Fin B.card ≃ {i : Fin n // i ∈ B})
    (g : Ω → ℝ) (r : Ω → Ω → ℝ) (x : Fin n → Ω)
    (hsym : ∀ a b : Ω, r a b = r b a)
    (hn : 4 ≤ n) (hm : 4 ≤ B.card) :
    Shao.OQ5.P51.projectionW B.card g (backgroundSample B e x) +
        Shao.OQ5.P51.residualStatisticD
          B.card (scaledResidual B r) (backgroundSample B e x) =
      backgroundNumerator B (fun i : Fin n => g (x i))
        (fun i j : Fin n => r (x i) (x j))
