theorem sqrt_centeredSq_background {n : ℕ} {Ω : Type u}
    (B : Finset (Fin n)) (e : Fin B.card ≃ {i : Fin n // i ∈ B})
    (g : Ω → ℝ) (r : Ω → Ω → ℝ) (x : Fin n → Ω)
    (hsym : ∀ a b : Ω, r a b = r b a)
    (hn : 4 ≤ n) (hm : 4 ≤ B.card) :
    Real.sqrt (Shao.OQ5.P51.centeredSq
        B.card g (scaledResidual B r) (backgroundSample B e x)) =
      backgroundStudentizer B (fun i : Fin n => g (x i))
        (fun i j : Fin n => r (x i) (x j))
