    ∑ j : Fin n, (Shao.OQ6.ResidualDeletion.rowSum
      (Shao.OQ6.ResidualDeletion.eraseIncident (e ζ) i) j) ^ 2
  let K : ℝ := ((n : ℝ) - 2) ^ 2

  let denA : ℝ := (1 + rho) * z * ((n : ℝ) ^ 2 * N)
  let k : ℝ := C0 / (K * Q)
  let kA : ℝ := (4 * C0) / denA

    (4 * C0) * (A + L ζ / denA)
  let BAdelete : (Fin n → Ω) → Fin n → ℝ := fun ζ i =>
    (4 * C0) * (A + Ldelete ζ i / denA)

      (|(|D1 ζ|) - (|residualStatisticDDeleted n r ζ i|)| +
        z * |BA ζ - BAdelete ζ i|)

    dsimp only [k]
    ring

    change C0 * (A + (Ldelete ζ i / K) / Q) = _
    rw [div_div]
    dsimp only [k]
    ring
  have hBAform (ζ : Fin n → Ω) : BA ζ = (4 * C0) * A + kA * L ζ := by
    dsimp only [BA, kA]
    ring
  have hBAdeleteform (ζ : Fin n → Ω) (i : Fin n) :
      BAdelete ζ i = (4 * C0) * A + kA * Ldelete ζ i := by
    dsimp only [BAdelete, kA]
    ring
  have hB0 (ζ : Fin n → Ω) : 0 ≤ B ζ := by
    rw [hBform ζ]
    exact add_nonneg (mul_nonneg hC0 hA0) (mul_nonneg hk0 (hL0 ζ))

    exact add_le_add
      (mul_le_mul_of_nonneg_right (by linarith only [hC0] : C0 ≤ 4 * C0) hA0)
      (mul_le_mul_of_nonneg_right hcoeff (hL0 ζ))
  -- The shared deterministic terms cancel BEFORE taking absolute values.
  have hBdiff (ζ : Fin n → Ω) (i : Fin n) :
      |B ζ - Bdelete ζ i| = k * |L ζ - Ldelete ζ i| := by
    rw [hBform ζ, hBdeleteform ζ i]
    rw [show C0 * A + k * L ζ - (C0 * A + k * Ldelete ζ i) =
      k * (L ζ - Ldelete ζ i) by ring]
    rw [abs_mul, abs_of_nonneg hk0]
  have hBAdiff (ζ : Fin n → Ω) (i : Fin n) :
      |BA ζ - BAdelete ζ i| = kA * |L ζ - Ldelete ζ i| := by
    rw [hBAform ζ, hBAdeleteform ζ i]
    rw [show (4 * C0) * A + kA * L ζ - ((4 * C0) * A + kA * Ldelete ζ i) =
      kA * (L ζ - Ldelete ζ i) by ring]
    rw [abs_mul, abs_of_nonneg hkA0]
  have hBdiffdom (ζ : Fin n → Ω) (i : Fin n) :
      |B ζ - Bdelete ζ i| ≤ |BA ζ - BAdelete ζ i| := by
    rw [hBdiff ζ i, hBAdiff ζ i]

  -- This is the accepted analytic producer, instantiated once at coefficient 4*C0.
