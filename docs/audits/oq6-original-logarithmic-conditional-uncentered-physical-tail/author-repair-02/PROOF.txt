      _ = Real.rpow 2 (1 / 6 : ℝ) * Real.rpow (m : ℝ) (1 / 6 : ℝ) := by
        simpa only [Real.rpow_eq_pow] using
          Real.mul_rpow (x := (2 : ℝ)) (y := (m : ℝ)) (z := (1 / 6 : ℝ))
            (by norm_num : (0 : ℝ) ≤ 2) hmpos.le
      _ ≤ 2 * Real.rpow (m : ℝ) (1 / 6 : ℝ) := by
        apply mul_le_mul_of_nonneg_right _ (Real.rpow_nonneg hmpos.le _)
        convert Real.rpow_le_rpow_of_exponent_le (by norm_num : (1 : ℝ) ≤ 2)
          (by norm_num : (1 / 6 : ℝ) ≤ 1) using 1 <;> norm_num
  have hWindowR : z ≤ cR * Real.rpow (m : ℝ) (1 / 6 : ℝ) := by
    calc
      _ ≤ c * Real.rpow (n : ℝ) qpow := hzWindow
      _ ≤ c * (2 * Real.rpow (m : ℝ) (1 / 6 : ℝ)) :=
        mul_le_mul_of_nonneg_left hNmSixth hc.le
      _ ≤ cR * Real.rpow (m : ℝ) (1 / 6 : ℝ) := by
        have hh := mul_le_mul_of_nonneg_right hcRBound (Real.rpow_nonneg hmpos.le (1 / 6 : ℝ))
        simp only [Real.rpow_eq_pow] at hh ⊢
        nlinarith only [hh]
  have hWindowA : z ≤ cA * Real.rpow (m : ℝ) (1 / 6 : ℝ) := by
    calc
      _ ≤ c * Real.rpow (n : ℝ) qpow := hzWindow
      _ ≤ c * (2 * Real.rpow (m : ℝ) (1 / 6 : ℝ)) :=
        mul_le_mul_of_nonneg_left hNmSixth hc.le
      _ ≤ cA * Real.rpow (m : ℝ) (1 / 6 : ℝ) := by
        have hh := mul_le_mul_of_nonneg_right hcABound (Real.rpow_nonneg hmpos.le (1 / 6 : ℝ))
        simp only [Real.rpow_eq_pow] at hh ⊢
        nlinarith only [hh]
  have hGood := hGoodN m hmGood νn gstd hgstdm hgstd2 hgstdMean
    hgstdSecond hgstdp hβstdBound rstd wstd hrstdm hrstd1 hrstdCan hrstdCanL
    hwstd0 hwHeight hwstdSq hwstdSqBound hrstdEnv hrstdsym
    0 0 z (le_refl 0) (le_refl 0) hz1 hWindowR
  rcases hGood with ⟨hHmeas, hPoint, hZeroOutside, hPayment⟩
  change MeasurableSet Hgood at hHmeas
  change P.real Hgoodᶜ ≤ payment at hPayment
  have hPhysical : ∀ ζ : Fin m → Ω, ζ ∈ Hgood →
      (1 / 2 : ℝ) ≤ RH ζ ∧ |Real.sqrt (m : ℝ) * κ / RH ζ| ≤ δn ∧
        |Tuncenter ζ - Tcenter ζ| ≤ δn := by
    intro ζ hζ
    rcases hPoint ζ hζ with
      ⟨hVpos, hEnergy, hD8, hD4, hzDelta, hDelta, hDeltaLower,
        heabs, hzeabs, hze, he32, hClipPos, hspos, hFactor⟩
    change sstd ζ = projectionV m gstd ζ *
      Real.sqrt (1 + centeredDelta m gstd rstd ζ) at hFactor
    have hdLower : (31 / 32 : ℝ) ≤ 1 + centeredDelta m gstd rstd ζ := by
      linarith only [(abs_le.mp hDelta).1]
    have hd0 : 0 ≤ 1 + centeredDelta m gstd rstd ζ := by linarith only [hdLower]
    have hSquare : (sstd ζ) ^ 2 = projectionEnergy m gstd ζ *
        (1 + centeredDelta m gstd rstd ζ) := by
      rw [hFactor, mul_pow, projectionV_sq, Real.sq_sqrt hd0]
    have hProduct := mul_le_mul hEnergy hdLower (by norm_num : (0 : ℝ) ≤ 31 / 32)
      (by linarith only [hEnergy] : 0 ≤ projectionEnergy m gstd ζ)
    have hs0 : 0 ≤ sstd ζ := Real.sqrt_nonneg _
    have hsHalf : (1 / 2 : ℝ) ≤ sstd ζ := by
      apply (sq_le_sq₀ (by norm_num : (0 : ℝ) ≤ 1 / 2) hs0).1
      nlinarith only [hSquare, hProduct]
    have hσs := mul_le_mul hσlow hsHalf (by norm_num : (0 : ℝ) ≤ 1 / 2) hσpos.le
    have hRH : (1 / 2 : ℝ) ≤ RH ζ := by nlinarith only [(hStats ζ).1, hσs]
    have hRHpos : 0 < RH ζ := by linarith only [hRH]
    have hShift : |Real.sqrt (m : ℝ) * κ / RH ζ| ≤ δn := by
      rw [abs_div, abs_mul, abs_of_nonneg (Real.sqrt_nonneg _), abs_of_pos hRHpos]
      calc
        _ ≤ 2 * (Real.sqrt (m : ℝ) * |κ|) := by
          apply (div_le_iff₀ hRHpos).2
          have hNumerator : 0 ≤ 2 * (Real.sqrt (m : ℝ) * |κ|) :=
            mul_nonneg (by norm_num : (0 : ℝ) ≤ 2)
              (mul_nonneg (Real.sqrt_nonneg (m : ℝ)) (abs_nonneg κ))
          calc
            Real.sqrt (m : ℝ) * |κ| =
                (2 * (Real.sqrt (m : ℝ) * |κ|)) * (1 / 2 : ℝ) := by ring
            _ ≤ (2 * (Real.sqrt (m : ℝ) * |κ|)) * RH ζ :=
              mul_le_mul_of_nonneg_left hRH hNumerator
        _ ≤ 2 * (Kmean * Real.rpow (n : ℝ) (-J)) :=
          mul_le_mul_of_nonneg_left hMean (by norm_num)
        _ = δn := by dsimp only [δn]; ring
    have hDifference : Tuncenter ζ - Tcenter ζ = Real.sqrt (m : ℝ) * κ / RH ζ := by
      dsimp only [Tuncenter, Tcenter]
      simp only [if_neg (ne_of_gt hRHpos)]
      ring
    refine ⟨hRH, hShift, ?_⟩
    rw [hDifference]
    exact hShift
  -- The complement is the actual 3+1+4 payment; the last exponential is
  -- bounded using z >= 1 before applying the uniform absorption theorem.
  have hExpZ : Real.exp (-(z * Real.sqrt (m : ℝ)) / 2) ≤
      Real.exp (-(Real.sqrt (m : ℝ)) / 2) := by
    apply Real.exp_le_exp.mpr
    nlinarith only [mul_nonneg (sub_nonneg.mpr hz1) (Real.sqrt_nonneg (m : ℝ))]
  have hAbsM := (hNabs m hmAbs).2.2 z hz1 hWindowA
  change 6 * Real.exp (-qm) + 2 * Real.exp (-cg * (m : ℝ)) +
    8 * Real.exp (-(Real.sqrt (m : ℝ)) / 2) ≤
      Real.exp (-(t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ)) *
        Shao.standardNormalTail z at hAbsM
  have hBad0 : 0 ≤ P.real Hgoodᶜ := measureReal_nonneg
  have hDoublePayment : 2 * P.real Hgoodᶜ ≤
      Real.exp (-(t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ)) *
        Shao.standardNormalTail z := by
    dsimp only [payment] at hPayment
    nlinarith only [hPayment, hExpZ, hAbsM]
  have hBadRelative : P.real Hgoodᶜ / Shao.standardNormalTail z ≤
      Real.exp (-(t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ)) := by
    apply (div_le_iff₀ hΨ).2
    nlinarith only [hDoublePayment, hBad0]
  -- Closed-event inclusions need no continuity of either statistic's law.
  have hLowerInclusion : {ζ : Fin m → Ω | z + δn ≤ Tcenter ζ} ⊆
      {ζ : Fin m → Ω | z ≤ Tuncenter ζ} ∪ Hgoodᶜ := by
    intro ζ hζ
    by_cases hhgood : ζ ∈ Hgood
    · apply Or.inl
      have hh := (abs_le.mp (hPhysical ζ hhgood).2.2).1
      change z + δn ≤ Tcenter ζ at hζ
      change z ≤ Tuncenter ζ
      linarith only [hh, hζ]
    · exact Or.inr hhgood
  have hUpperInclusion : {ζ : Fin m → Ω | z ≤ Tuncenter ζ} ⊆
      {ζ : Fin m → Ω | z - δn ≤ Tcenter ζ} ∪ Hgoodᶜ := by
    intro ζ hζ
    by_cases hhgood : ζ ∈ Hgood
    · apply Or.inl
      have hh := (abs_le.mp (hPhysical ζ hhgood).2.2).2
      change z ≤ Tuncenter ζ at hζ
      change z - δn ≤ Tcenter ζ
      linarith only [hh, hζ]
    · exact Or.inr hhgood
  have hLowerPay : P.real {ζ : Fin m → Ω | z + δn ≤ Tcenter ζ} ≤
      P.real {ζ : Fin m → Ω | z ≤ Tuncenter ζ} + P.real Hgoodᶜ :=
    (measureReal_mono hLowerInclusion).trans (measureReal_union_le _ _)
  have hLower : P.real {ζ : Fin m → Ω | z + δn ≤ Tcenter ζ} - P.real Hgoodᶜ ≤
      P.real {ζ : Fin m → Ω | z ≤ Tuncenter ζ} := by linarith only [hLowerPay]
  have hUpper : P.real {ζ : Fin m → Ω | z ≤ Tuncenter ζ} ≤
      P.real {ζ : Fin m → Ω | z - δn ≤ Tcenter ζ} + P.real Hgoodᶜ :=
    (measureReal_mono hUpperInclusion).trans (measureReal_union_le _ _)
  let X : ℝ := β * Real.rpow (1 + z) p * Real.rpow (n : ℝ) (-((p - 2) / 2))
  let Y : ℝ := (1 + z) ^ 3 / Real.sqrt (n : ℝ)
  let E : ℝ := X + Y
  have hX0 : 0 ≤ X := by
    exact mul_nonneg
      (mul_nonneg hβ0 (Real.rpow_nonneg (by linarith only [hz0] : 0 ≤ 1 + z) p))
      (Real.rpow_nonneg hnpos.le (-((p - 2) / 2)))
  have hY0 : 0 ≤ Y := by dsimp only [Y]; positivity
  have hE0 : 0 ≤ E := add_nonneg hX0 hY0
  have hLinearCube : 1 + z ≤ (1 + z) ^ 3 := by
    nlinarith only [hz0, sq_nonneg z, mul_nonneg hz0 (sq_nonneg z)]
  have hCubeOne : 1 ≤ (1 + z) ^ 3 := by linarith only [hLinearCube, hz0]
  have hRootComparison : Real.sqrt (n : ℝ) ≤ 2 * Real.sqrt (m : ℝ) := by
    calc
      _ ≤ Real.sqrt (4 * (m : ℝ)) := Real.sqrt_le_sqrt (by
        exact_mod_cast (by omega : n ≤ 4 * m))
      _ = 2 * Real.sqrt (m : ℝ) := by
        rw [Real.sqrt_mul (by norm_num : (0 : ℝ) ≤ 4)]
        rw [show (4 : ℝ) = (2 : ℝ) ^ 2 by norm_num,
          Real.sqrt_sq (by norm_num : (0 : ℝ) ≤ 2)]
  have hExpSmall := hNexp m hmExp
  have hBadSmall : P.real Hgoodᶜ / Shao.standardNormalTail z ≤ 2 * Y := by
    calc
      _ ≤ Real.exp (-(t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ)) := hBadRelative
      _ ≤ 2 / Real.sqrt (n : ℝ) := by
        apply (le_div_iff₀ hSqrtN).2
        have hh := mul_le_mul_of_nonneg_right hRootComparison
          (Real.exp_pos (-(t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ))).le
        nlinarith only [hh, hExpSmall]
      _ ≤ 2 * Y := by
        change 2 / Real.sqrt (n : ℝ) ≤ 2 * ((1 + z) ^ 3 / Real.sqrt (n : ℝ))
        have hh := mul_le_mul_of_nonneg_left
          (div_le_div_of_nonneg_right hCubeOne hSqrtN.le)
          (by norm_num : (0 : ℝ) ≤ 2)
        convert hh using 1 <;> ring
  have hGaussianPaid : 2 * Real.exp 2 * (1 + z) * δn ≤ Lmean * Y := by
    calc
      _ ≤ 2 * Real.exp 2 * (1 + z) * (2 * Kmean / Real.sqrt (n : ℝ)) :=
        mul_le_mul_of_nonneg_left hδSqrt (by positivity)
      _ = Lmean * ((1 + z) / Real.sqrt (n : ℝ)) := by dsimp only [Lmean]; ring
      _ ≤ Lmean * Y := mul_le_mul_of_nonneg_left
        (div_le_div_of_nonneg_right hLinearCube hSqrtN.le) hLmean
  have hTwoWindow : 2 * z ≤ c0 * Real.rpow (n : ℝ) qpow := by
    have hh := mul_le_mul_of_nonneg_right hc0Bound hzpow
    have hz2 := mul_le_mul_of_nonneg_left hzWindow (by norm_num : (0 : ℝ) ≤ 2)
    nlinarith only [hh, hz2]
  have hPlus1 : 1 ≤ z + δn := by linarith only [hz, hδ0]
  have hMinus1 : 1 ≤ z - δn := by linarith only [hz, hδSmall]
  have hPlusWindow : z + δn ≤ c0 * Real.rpow (n : ℝ) qpow :=
    (show z + δn ≤ 2 * z by linarith only [hz, hδSmall]).trans hTwoWindow
  have hMinusWindow : z - δn ≤ c0 * Real.rpow (n : ℝ) qpow :=
    (show z - δn ≤ 2 * z by linarith only [hz, hδ0]).trans hTwoWindow
  have hPlusDistance : |z + δn - z| ≤ δn := by
    rw [show z + δn - z = δn by ring, abs_of_nonneg hδ0]
  have hMinusDistance : |z - δn - z| ≤ δn := by
    rw [show z - δn - z = -δn by ring, abs_neg, abs_of_nonneg hδ0]
  -- Polynomial comparison and Gaussian increment are uniform for both legal
  -- shifted levels. This uses J >= 1/2 only through the common shift budget.
  have hShifted : ∀ v : ℝ, 1 ≤ v →
      v ≤ c0 * Real.rpow (n : ℝ) qpow → |v - z| ≤ δn →
      |P.real {ζ : Fin m → Ω | v ≤ Tcenter ζ} / Shao.standardNormalTail z - 1| ≤
        (8 * Real.exp 2 * K0) * E + Lmean * Y := by
    intro v hv1 hvWindow hvDistance
    have hv0 : 0 ≤ v := by linarith only [hv1]

      linarith only [(abs_le.mp hvDistance).2, hδSmall, hz]

      exact hh

              (by linarith only [hz0] : 0 ≤ 1 + z)

          mul_le_mul_of_nonneg_right hTwoP (Real.rpow_nonneg (by linarith only [hz0]) p)

      calc
        _ ≤ (2 * (1 + z)) ^ 3 :=
          pow_le_pow_left₀ (by linarith only [hv0]) hbase 3
        _ = _ := by ring
    have hFirst : β * Real.rpow (1 + v) p * Real.rpow (n : ℝ) (-((p - 2) / 2)) ≤ 8 * X := by
      have hh := mul_le_mul_of_nonneg_right
        (mul_le_mul_of_nonneg_left hPowerP hβ0)
        (Real.rpow_nonneg hnpos.le (-((p - 2) / 2)))
      dsimp only [X]
      simp only [Real.rpow_eq_pow] at hh ⊢
      convert hh using 1 <;> ring
    have hSecond : (1 + v) ^ 3 / Real.sqrt (n : ℝ) ≤ 8 * Y := by

      convert hh using 1 <;> dsimp only [Y] <;> ring

        |P.real {ζ : Fin m → Ω | v ≤ Tcenter ζ} / Shao.standardNormalTail v - 1| ≤
          (8 * K0) * E := by
      apply hCenterBound.trans
      have hh := mul_le_mul_of_nonneg_left (add_le_add hFirst hSecond) hK0.le
      convert hh using 1 <;> dsimp only [E] <;> ring
    rcases gaussianIncrement z δn v hz hδ0 hδNeighborhood hvDistance hv0 with
      ⟨hRatioBound, hIncrement⟩
    have hRatio0 : 0 ≤ Shao.standardNormalTail v / Shao.standardNormalTail z :=
      div_nonneg ((Shao.standardNormalTail_basic.2.2.1 v).1).le hΨ.le
    have hΨv : 0 < Shao.standardNormalTail v := (Shao.standardNormalTail_basic.2.2.1 v).1
    have hIdentity :
        P.real {ζ : Fin m → Ω | v ≤ Tcenter ζ} / Shao.standardNormalTail z - 1 =
          (P.real {ζ : Fin m → Ω | v ≤ Tcenter ζ} / Shao.standardNormalTail v - 1) *
            (Shao.standardNormalTail v / Shao.standardNormalTail z) +
          (Shao.standardNormalTail v / Shao.standardNormalTail z - 1) := by
      field_simp [ne_of_gt hΨ, ne_of_gt hΨv] <;> ring
    rw [hIdentity]
    calc
      _ ≤ |(P.real {ζ : Fin m → Ω | v ≤ Tcenter ζ} / Shao.standardNormalTail v - 1) *
            (Shao.standardNormalTail v / Shao.standardNormalTail z)| +
          |Shao.standardNormalTail v / Shao.standardNormalTail z - 1| := abs_add_le _ _
      _ = |P.real {ζ : Fin m → Ω | v ≤ Tcenter ζ} / Shao.standardNormalTail v - 1| *
            (Shao.standardNormalTail v / Shao.standardNormalTail z) +
          |Shao.standardNormalTail v / Shao.standardNormalTail z - 1| := by
        rw [abs_mul, abs_of_nonneg hRatio0]
      _ ≤ ((8 * K0) * E) * Real.exp 2 + 2 * Real.exp 2 * (1 + z) * δn :=
        add_le_add (mul_le_mul hCenterPaid hRatioBound hRatio0 (by positivity)) hIncrement
      _ ≤ ((8 * K0) * E) * Real.exp 2 + Lmean * Y :=
        add_le_add (le_refl (((8 * K0) * E) * Real.exp 2)) hGaussianPaid
      _ = _ := by ring

  have hMinus := hShifted (z - δn) hMinus1 hMinusWindow hMinusDistance
  have hLowerDiv := div_le_div_of_nonneg_right hLower hΨ.le
  have hUpperDiv := div_le_div_of_nonneg_right hUpper hΨ.le
  rw [sub_div] at hLowerDiv
  rw [add_div] at hUpperDiv
  have hRatioSandwich :
      |P.real {ζ : Fin m → Ω | z ≤ Tuncenter ζ} / Shao.standardNormalTail z - 1| ≤
        (8 * Real.exp 2 * K0) * E + Lmean * Y + 2 * Y := by
    apply abs_le.mpr
    constructor
    · have hh := (abs_le.mp hPlus).1
      linarith only [hLowerDiv, hh, hBadSmall]
    · have hh := (abs_le.mp hMinus).2
      linarith only [hUpperDiv, hh, hBadSmall]
  have hFinal :
      |P.real {ζ : Fin m → Ω | z ≤ Tuncenter ζ} / Shao.standardNormalTail z - 1| ≤
        K * E := by
    apply hRatioSandwich.trans
    have hh := mul_nonneg hLmean hX0
    dsimp only [K, E]
    nlinarith only [hh, hX0]
  refine ⟨hHmeas, hPhysical, hPayment, hBadRelative, hLower, hUpper, hFinal, ?_⟩
  have hEvent : MeasurableSet {ζ : Fin m → Ω | z ≤ Tuncenter ζ} :=
    measurableSet_le measurable_const hTuMeas
  rw [← hLawReal _ hEvent]
