theorem split_joinSample {n : ℕ} {Ω : Type u}
    (B : Finset (Fin n))
    (eB : Fin B.card ≃ {i : Fin n // i ∈ B})
    (eA : Fin (Bᶜ : Finset (Fin n)).card ≃
      {i : Fin n // i ∈ (Bᶜ : Finset (Fin n))})
    (y : (Fin B.card → Ω) × (Fin (Bᶜ : Finset (Fin n)).card → Ω)) :
    splitSample B eB eA (joinSample B eB eA y) = y
