private theorem finite_projection_residual_numerator
    {n : ℕ} (g : Fin n → ℝ) (r : Fin n → Fin n → ℝ)
    (hn : 3 ≤ n) :
    (∑ i : Fin n, g i) / Real.sqrt (n : ℝ) +
      Shao.OQ5.P35.unorderedPairSum r /
        (Real.sqrt (n : ℝ) * ((n : ℝ) - 1)) =
      Real.sqrt (n : ℝ) *
        Shao.OQ6.FiniteRows.uStatistic (fun i j => g i + g j + r i j) / 2
