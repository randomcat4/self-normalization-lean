private theorem centered_energy_partition
    (n : ℕ) (b : Fin n → ℝ) (B : Finset (Fin n))
    (hn : 0 < n) (hB : B.Nonempty) :
    let A : Finset (Fin n) := Finset.univ \ B
    let M : ℝ := (B.card : ℝ)
    let K : ℝ := (A.card : ℝ)
    let muB : ℝ := (∑ i ∈ B, b i) / M
    let muA : ℝ := (∑ i ∈ A, b i) / K
    (∑ i : Fin n, (Shao.OQ5.P35.center b i) ^ (2 : ℕ)) =
      (∑ i ∈ B, (b i - muB) ^ (2 : ℕ)) +
      (∑ i ∈ A, (b i - muA) ^ (2 : ℕ)) +
      (M * K / (n : ℝ)) * (muA - muB) ^ (2 : ℕ)
private theorem positive_part_block_cauchy
    (N M K q a y : ℝ)
    (hN : 0 < N) (hM : 0 < M) (hK : 0 ≤ K) (hq : 0 < q) :
    (max (N * a + K * y) 0) ^ (2 : ℕ) ≤
