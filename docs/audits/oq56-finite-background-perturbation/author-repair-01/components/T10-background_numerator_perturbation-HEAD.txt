theorem background_numerator_perturbation {n : ℕ} (B : Finset (Fin n))
    (g : Fin n → ℝ) (r : Fin n → Fin n → ℝ)
    (hsym : ∀ i j : Fin n, r i j = r j i)
    (hn : 4 ≤ n) (hm : 3 ≤ B.card) :
    Real.sqrt (n : ℝ) * backgroundAnchor B g r =
      Real.sqrt ((n : ℝ) / (B.card : ℝ)) * backgroundNumerator B g r +
        numeratorShift B r
