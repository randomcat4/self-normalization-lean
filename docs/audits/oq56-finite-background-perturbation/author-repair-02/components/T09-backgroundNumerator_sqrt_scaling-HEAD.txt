theorem backgroundNumerator_sqrt_scaling {n : ℕ} (B : Finset (Fin n))
    (g : Fin n → ℝ) (r : Fin n → Fin n → ℝ)
    (hn : 4 ≤ n) (hm : 3 ≤ B.card) :
    Real.sqrt ((n : ℝ) / (B.card : ℝ)) * backgroundNumerator B g r =
      Real.sqrt (n : ℝ) / (B.card : ℝ) *
        ((∑ i ∈ B, g i) + ((B.card : ℝ) - 2) * blockPairSum B r /
          (((n : ℝ) - 2) * ((B.card : ℝ) - 1)))
