:= by
  classical
  change MeasurableSet
    ({x : Fin n → Ω | ∀ i : Fin n, 0 ≤ w (x i)} ∩
      {x : Fin n → Ω | ∀ i j : Fin n,
        i ≠ j → |r (x i) (x j)| ≤ w (x i) * w (x j)})
  apply MeasurableSet.inter
  · simp only [Set.ofPred_forall]
    exact MeasurableSet.iInter fun i =>
      measurableSet_le measurable_const (hw.comp (measurable_pi_apply i))
  · simp only [Set.ofPred_forall]
    exact MeasurableSet.iInter fun i => MeasurableSet.iInter fun j =>
      MeasurableSet.iInter fun _ =>
        measurableSet_le
          ((hr.comp ((measurable_pi_apply i).prodMk (measurable_pi_apply j))).abs)
          ((hw.comp (measurable_pi_apply i)).mul (hw.comp (measurable_pi_apply j)))
