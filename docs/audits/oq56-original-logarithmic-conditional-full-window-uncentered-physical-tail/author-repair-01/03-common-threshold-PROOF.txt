:= by
  let q : ℝ := (p - 2) / (2 * p)
  let C : ℝ := 216 / t0 ^ 3
  have hp0 : 0 < p := lt_trans (by norm_num) hp2
  have hq : 0 < q := div_pos (sub_pos.mpr hp2) (mul_pos (by norm_num) hp0)
  have hC : 0 < C := by dsimp only [C]; positivity
  have hnat : Tendsto (fun n : ℕ => (n : ℝ)) atTop atTop :=
    tendsto_natCast_atTop_atTop
  have hPower : Tendsto (fun n : ℕ => Real.rpow (n : ℝ) q) atTop atTop :=
    (tendsto_rpow_atTop hq).comp hnat
  obtain ⟨Nw, hNw⟩ := Filter.eventually_atTop.1
    (hPower.eventually_ge_atTop (3 / ccenter))
  obtain ⟨Ns, hNs⟩ := Filter.eventually_atTop.1 (hnat.eventually_ge_atTop (C ^ 2))
  let N : ℕ := max 8 (max Nw Ns)
  refine ⟨N, le_max_left _ _, ?_⟩
  intro n hn
  have hn8 : 8 ≤ n := (le_max_left _ _).trans hn
  have hnRest : max Nw Ns ≤ n := (le_max_right _ _).trans hn
  have hnW : Nw ≤ n := (le_max_left _ _).trans hnRest
  have hnS : Ns ≤ n := (le_max_right _ _).trans hnRest
  have hnpos : 0 < (n : ℝ) := by exact_mod_cast (by omega : 0 < n)
  have hRootN : 0 < Real.sqrt (n : ℝ) := Real.sqrt_pos.mpr hnpos
  have hCroot : C ≤ Real.sqrt (n : ℝ) := by
    apply (sq_le_sq₀ hC.le (Real.sqrt_nonneg (n : ℝ))).mp
    rw [Real.sq_sqrt hnpos.le]
    exact hNs n hnS
  refine ⟨?_, ?_⟩
  · change (3 : ℝ) ≤ ccenter * Real.rpow (n : ℝ) q
    have hh := (div_le_iff₀ hccenter).mp (hNw n hnW)
    simpa only [mul_comm] using hh
  · intro m hnm
    have hmpos : 0 < (m : ℝ) := by exact_mod_cast (by omega : 0 < m)
    have hnHalf : (n : ℝ) ≤ 2 * (m : ℝ) := by exact_mod_cast hnm
    let x : ℝ := (t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ)
    have hx : 0 < x := mul_pos (div_pos ht0 (by norm_num)) (Real.rpow_pos_of_pos hmpos _)
    have hRootCube : (Real.rpow (m : ℝ) (1 / 3 : ℝ)) ^ 3 = (m : ℝ) := by
      simp only [Real.rpow_eq_pow]
      rw [← Real.rpow_mul_natCast hmpos.le (1 / 3 : ℝ) (3 : ℕ)]
      norm_num
    have hxCube : x ^ 3 = t0 ^ 3 * (m : ℝ) / 8 := by
      dsimp only [x]
      rw [mul_pow, div_pow, hRootCube]
      ring
    have he : x / 3 ≤ Real.exp (x / 3) := by
      linarith only [Real.add_one_le_exp (x / 3)]
    have heCube := pow_le_pow_left₀ (div_nonneg hx.le (by norm_num)) he 3
    have hExpCube : (Real.exp (x / 3)) ^ 3 = Real.exp x := by
      rw [← Real.exp_nat_mul]
      congr 1
      ring
    rw [hExpCube] at heCube
    have hExpLower : t0 ^ 3 * (m : ℝ) / 216 ≤ Real.exp x := by
      nlinarith only [heCube, hxCube]
    have hDenM : 0 < t0 ^ 3 * (m : ℝ) := mul_pos (pow_pos ht0 _) hmpos
    have hDenN : 0 < t0 ^ 3 * (n : ℝ) := mul_pos (pow_pos ht0 _) hnpos
    have hExpInv : Real.exp (-x) ≤ 216 / (t0 ^ 3 * (m : ℝ)) := by
      rw [Real.exp_neg, ← one_div]
      apply (div_le_div_iff₀ (Real.exp_pos x) hDenM).2
      nlinarith only [hExpLower]
    rw [show -(t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ) = -x by
      dsimp only [x]; ring]
    calc
      _ ≤ 216 / (t0 ^ 3 * (m : ℝ)) := hExpInv
      _ ≤ 432 / (t0 ^ 3 * (n : ℝ)) := by
        apply (div_le_div_iff₀ hDenM hDenN).2
        have hh := mul_le_mul_of_nonneg_left hnHalf (pow_nonneg ht0.le 3)
        nlinarith only [hh]
      _ = 2 * C / (n : ℝ) := by
        dsimp only [C]
        field_simp [ne_of_gt ht0, ne_of_gt hnpos]
        <;> ring
      _ ≤ 2 / Real.sqrt (n : ℝ) := by
        apply (div_le_div_iff₀ hnpos hRootN).2
        have hh := mul_le_mul_of_nonneg_right hCroot (Real.sqrt_nonneg (n : ℝ))
        nlinarith only [hh, Real.sq_sqrt hnpos.le]
