private theorem half_sample_and_mean_rate_bounds
    (p J Kmean : ℝ) (hp : 2 < p ∧ p ≤ 3)
    (hJ : (1 / 2 : ℝ) ≤ J) (hKmean : 0 ≤ Kmean)
    (n m : ℕ) (hn : 1 ≤ n) (hnm : n ≤ 2 * m) :
    let a : ℝ := (p - 2) / 2
    Real.rpow (m : ℝ) (-a) ≤ 2 * Real.rpow (n : ℝ) (-a) ∧
      1 / Real.sqrt (m : ℝ) ≤ 2 / Real.sqrt (n : ℝ) ∧
      2 * Kmean * Real.rpow (n : ℝ) (-J) ≤
        2 * Kmean / Real.sqrt (n : ℝ)
