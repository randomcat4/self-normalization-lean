private theorem eventual_near_window_and_exception_rate
    (p t0 ccenter : ℝ) (hp2 : 2 < p)
    (ht0 : 0 < t0) (hccenter : 0 < ccenter) :
    ∃ N : ℕ, 8 ≤ N ∧
      ∀ n : ℕ, N ≤ n →
        (3 : ℝ) ≤ ccenter * Real.rpow (n : ℝ) ((p - 2) / (2 * p)) ∧
          ∀ m : ℕ, n ≤ 2 * m →
            Real.exp (-(t0 / 2) * Real.rpow (m : ℝ) (1 / 3 : ℝ)) ≤
              2 / Real.sqrt (n : ℝ)
