DECL Shao.OQ3.finite_word_monomial_encoding
TYPE ∀ {ι : Type u_1} {κ : Type u_2} [inst : Fintype ι] [inst_1 : Fintype κ] [inst_2 : DecidableEq κ] (sigma : ι → κ),
  have p := fun j => Fintype.card { i // sigma i = j };
  ∑ j, p j = Fintype.card ι ∧
    (∀ (w : κ → ℕ), ∑ j, w j * p j = ∑ i, w (sigma i)) ∧ ∀ (a : κ → ℂ), ∏ j, a j ^ p j = ∏ i, a (sigma i)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
