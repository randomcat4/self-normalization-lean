DECL Shao.OQ3.complex_gaussian_six_order_exponential_polynomial_error
TYPE ∀ (A : Fin 6 → ℝ),
  (∀ (j : Fin 6), 0 ≤ A j) →
    have P := fun a => ∑ q, 1 / ↑(↑q).factorial * ∑ sigma, if ∑ i, (↑(sigma i) + 1) ≤ 6 then ∏ i, a (sigma i) else 0;
    ∃ K,
      0 < K ∧
        ∀ (t r : ℝ),
          0 ≤ t →
            t ≤ 1 →
              0 ≤ r →
                ∀ (a : Fin 6 → ℂ),
                  (∀ (j : Fin 6), ‖a j‖ ≤ A j * t ^ (↑j + 1) * r ^ (↑j + 3)) →
                    ‖∑ j, a j‖ ≤ r ^ 2 / 4 →
                      ‖Complex.exp (-↑(r ^ 2) / 2) * (Complex.exp (∑ j, a j) - P a)‖ ≤
                        K * t ^ 7 * (r ^ 9 + r ^ 56) * Real.exp (-r ^ 2 / 4)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
