DECL Shao.OQ3.norm_complex_exp_add_error_of_re_le
TYPE ∀ (z e : ℂ) (a b : ℝ),
  z.re ≤ -a → ‖e‖ ≤ b → b ≤ a / 2 → ‖Complex.exp (z + e) - Complex.exp z‖ ≤ ‖e‖ * Real.exp (-a / 2)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.norm_complex_exp_add_error
TYPE ∀ (z e : ℂ), ‖Complex.exp (z + e) - Complex.exp z‖ ≤ ‖e‖ * Real.exp (z.re + ‖e‖)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
