DECL Shao.OQ3.quadratic_whitening_pair_equiv
TYPE ∀ (D : Matrix (Fin 2) (Fin 2) ℝ),
  D 1 0 = D 0 1 →
    ∀ (lo hi : ℝ),
      0 < lo →
        0 < hi →
          (∀ (xi eta : ℝ),
              lo * (xi ^ 2 + eta ^ 2) ≤ xi ^ 2 * D 0 0 + 2 * xi * eta * D 0 1 + eta ^ 2 * D 1 1 ∧
                xi ^ 2 * D 0 0 + 2 * xi * eta * D 0 1 + eta ^ 2 * D 1 1 ≤ hi * (xi ^ 2 + eta ^ 2)) →
            have s := √(D 0 0 * D 1 1 - D 0 1 ^ 2);
            have t := √(D 0 0 + D 1 1 + 2 * s);
            have A := ![![(D 1 1 + s) / (s * t), -D 0 1 / (s * t)], ![-D 0 1 / (s * t), (D 0 0 + s) / (s * t)]];
            have B := D * A;
            A * D * A = 1 ∧
              ∃ E,
                (∀ (w : ℝ × ℝ), E w = (A 0 0 * w.1 + A 0 1 * w.2, A 1 0 * w.1 + A 1 1 * w.2)) ∧
                  (∀ (w : ℝ × ℝ), E.symm w = (B 0 0 * w.1 + B 0 1 * w.2, B 1 0 * w.1 + B 1 1 * w.2)) ∧
                    ‖↑E‖ ≤ 2 * (1 / √lo) ∧ ‖↑E.symm‖ ≤ 4 * hi * (1 / √lo)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.matrix_pair_continuous_linear_equiv_of_inverse
TYPE ∀ (A B : Matrix (Fin 2) (Fin 2) ℝ),
  A * B = 1 →
    B * A = 1 →
      have F := fun M w => (M 0 0 * w.1 + M 0 1 * w.2, M 1 0 * w.1 + M 1 1 * w.2);
      ∃ E,
        ⇑E = F A ∧
          ⇑E.symm = F B ∧
            (∀ (c : ℝ), 0 ≤ c → (∀ (i j : Fin 2), |A i j| ≤ c) → ‖↑E‖ ≤ 2 * c) ∧
              ∀ (c : ℝ), 0 ≤ c → (∀ (i j : Fin 2), |B i j| ≤ c) → ‖↑E.symm‖ ≤ 2 * c
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
