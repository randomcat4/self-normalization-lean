DECL Shao.OQ3.bounded_compact_actual_cell_physical_gaussian_approximation
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure mu] (supportRadius : ℝ),
  0 ≤ supportRadius →
    (∀ᵐ (x : ℝ) ∂mu, |x| ≤ supportRadius) →
      ∀ (theta0 : ℝ × ℝ) (Theta : Set (ℝ × ℝ)),
        IsCompact Theta →
          Theta ⊆ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)) →
            ∀ (beta c R : ℝ),
              0 < beta →
                beta < 2 / 5 →
                  0 < c →
                    0 ≤ R →
                      ∀ (level a0 stripRadius : ℝ),
                        0 ≤ stripRadius →
                          (∀ p ∈ Theta,
                              0 < (Shao.OQ3.parabolaCovariance (mu.tilted fun x => p.1 * x + p.2 * x ^ 2)).det) →
                            (∀ (xi eta : ℝ),
                                R < √(xi ^ 2 + eta ^ 2) →
                                  ‖Shao.OQ3.jointCharacteristic (mu.tilted fun x => theta0.1 * x + theta0.2 * x ^ 2) xi
                                        eta‖ ≤
                                    1 - c * (√(xi ^ 2 + eta ^ 2)).rpow (-beta)) →
                              have tilt := fun p => mu.tilted fun x => p.1 * x + p.2 * x ^ 2;
                              have D := fun p => Shao.OQ3.parabolaCovariance (tilt p);
                              have s := fun p => √(D p 0 0 * D p 1 1 - D p 0 1 ^ 2);
                              have t := fun p => √(D p 0 0 + D p 1 1 + 2 * s p);
                              have A := fun p =>
                                ![![(D p 1 1 + s p) / (s p * t p), -D p 0 1 / (s p * t p)],
                                  ![-D p 0 1 / (s p * t p), (D p 0 0 + s p) / (s p * t p)]];
                              have means := fun p => ![∫ (x : ℝ), x ∂tilt p, ∫ (x : ℝ), x ^ 2 ∂tilt p];
                              have Y := fun p x => (A p).mulVec (![x, x ^ 2] - means p);
                              have laws := fun p => MeasureTheory.Measure.map (Y p) (tilt p);
                              have Phi := fun p xi eta =>
                                ∫ (y : Fin 2 → ℝ), Complex.exp (↑(xi * y 0 + eta * y 1) * Complex.I) ∂laws p;
                              have phi := fun x => 1 / (2 * Real.pi) * Real.exp (-(x.1 ^ 2 + x.2 ^ 2) / 2);
                              have Klog := fun p => Real.log (∫ (x : ℝ), Real.exp (p.1 * x + p.2 * x ^ 2) ∂mu);
                              ∃ lo hi,
                                0 < lo ∧
                                  0 < hi ∧
                                    (∀ p ∈ Theta,
                                        ∀ (xi eta : ℝ),
                                          lo * (xi ^ 2 + eta ^ 2) ≤
                                              xi ^ 2 * D p 0 0 + 2 * xi * eta * D p 0 1 + eta ^ 2 * D p 1 1 ∧
                                            xi ^ 2 * D p 0 0 + 2 * xi * eta * D p 0 1 + eta ^ 2 * D p 1 1 ≤
                                              hi * (xi ^ 2 + eta ^ 2)) ∧
                                      ∃ E,
                                        (∀ (p : ↑Theta),
                                            (∀ (w : ℝ × ℝ),
                                                (E p) w =
                                                  (A (↑p) 0 0 * w.1 + A (↑p) 0 1 * w.2,
                                                    A (↑p) 1 0 * w.1 + A (↑p) 1 1 * w.2)) ∧
                                              ‖↑(E p)‖ ≤ 2 * (1 / √lo) ∧
                                                ‖↑(E p).symm‖ ≤ 4 * hi * (1 / √lo) ∧
                                                  ∀ (x : ℝ),
                                                    (E p) ((x, x ^ 2) - (means (↑p) 0, means (↑p) 1)) =
                                                      (Y (↑p) x 0, Y (↑p) x 1)) ∧
                                          ∃ C,
                                            0 < C ∧
                                              ∀ (epsilon : ℝ),
                                                0 < epsilon →
                                                  ∃ N,
                                                    2 ≤ N ∧
                                                      ∀ (n : ℕ),
                                                        N ≤ n →
                                                          ∀ (p : ↑Theta),
                                                            have Qsample := MeasureTheory.Measure.pi fun x => tilt ↑p;
                                                            have M := fun z => ((∑ i, z i) / ↑n, Shao.energy z / ↑n);
                                                            have T := fun z =>
                                                              √↑n • (E p) (M z - (means (↑p) 0, means (↑p) 1));
                                                            have nu := MeasureTheory.Measure.map T Qsample;
                                                            have P := MeasureTheory.Measure.pi fun x => mu;
                                                            have rateAtMean :=
                                                              (↑p).1 * means (↑p) 0 + (↑p).2 * means (↑p) 1 - Klog ↑p;
                                                            MeasureTheory.IsProbabilityMeasure nu ∧
                                                              ∀ (aL aR qL qR : ℝ),
                                                                (means (↑p) 0, means (↑p) 1) ∈
                                                                    Set.Icc aL aR ×ˢ Set.Icc qL qR →
                                                                  aR - aL ≤ 1 / ↑n →
                                                                    qR - qL ≤ 1 / ↑n →
                                                                      have cellA :=
                                                                        Set.Ioc aL aR ×ˢ Set.Ioc qL qR ∩
                                                                          {w |
                                                                            |w.1 - a0| ≤ stripRadius ∧
                                                                              0 ≤ w.1 ^ 2 / level ^ 2 - w.2 ∧
                                                                                w.1 ^ 2 / level ^ 2 - w.2 ≤
                                                                                  stripRadius};
                                                                      have stdCell :=
                                                                        (fun w =>
                                                                            √↑n •
                                                                              (E p)
                                                                                (w - (means (↑p) 0, means (↑p) 1))) ''
                                                                          cellA;
                                                                      have g := fun z =>
                                                                        Real.exp
                                                                          (-√↑n *
                                                                            ((↑p).1 * ((E p).symm z).1 +
                                                                              (↑p).2 * ((E p).symm z).2));
                                                                      have f := stdCell.indicator g;
                                                                      have J0 := ∫ (z : ℝ × ℝ), f z * phi z;
                                                                      Measurable f ∧
                                                                        MeasureTheory.Integrable f
                                                                            MeasureTheory.volume ∧
                                                                          MeasureTheory.Integrable f nu ∧
                                                                            MeasureTheory.Integrable
                                                                                (fun z => f z * phi z)
                                                                                MeasureTheory.volume ∧
                                                                              0 ≤ J0 ∧
                                                                                P.real (M ⁻¹' cellA) =
                                                                                    Real.exp (-↑n * rateAtMean) *
                                                                                      ∫ (z : ℝ × ℝ), f z ∂nu ∧
                                                                                  |Real.exp (↑n * rateAtMean) *
                                                                                          P.real (M ⁻¹' cellA) -
                                                                                        J0| <
                                                                                    C / ↑n * J0 + epsilon / ↑n ^ 3
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
