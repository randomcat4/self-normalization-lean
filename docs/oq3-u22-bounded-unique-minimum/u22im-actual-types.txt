DECL Shao.OQ3.exists_small_nonzero_iid_count_arbitrary_exponential_bound
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure mu] (alpha R : ℝ),
  0 < alpha →
    0 < R →
      ∃ eps,
        0 < eps ∧
          ∀ (n : ℕ),
            have P := MeasureTheory.Measure.pi fun x => mu;
            have SmallCount := fun z => {i | z i ≠ 0 ∧ |z i| ≤ eps}.card;
            P.real {z | alpha * ↑n ≤ ↑(SmallCount z)} ≤ Real.exp (-R * ↑n)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
