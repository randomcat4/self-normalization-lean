DECL Shao.OQ3.actual_high_energy_exterior_rate_gap_of_unique_minimum
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (level a0 B eta rho J0 : ℝ),
  0 < eta →
    0 < rho →
      have I := Shao.Probability.bivariateLegendreRate mu fun x => (x, x ^ 2);
      have center := (a0, a0 ^ 2 / level ^ 2);
      have Tail := {w | 0 < w.2 ∧ level * √w.2 ≤ w.1};
      have U := {w | |w.1 - a0| ≤ rho ∧ |w.1 ^ 2 / level ^ 2 - w.2| ≤ rho};
      have Carrier := Set.Icc (-B) B ×ˢ Set.Icc 0 (B ^ 2);
      have A := Tail \ U ∩ {w | eta ≤ w.2};
      I center = ↑J0 →
        (∀ w ∈ Tail, I center ≤ I w ∧ (I w = I center ↔ w = center)) →
          ∃ delta, 0 < delta ∧ ∀ w ∈ Carrier ∩ closure A, ↑(J0 + delta) < I w
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
