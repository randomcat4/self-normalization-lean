DECL Shao.OQ3.self_normalized_low_energy_many_small_nonzero
TYPE ∀ (level eps : ℝ),
  0 < level →
    0 < eps →
      ∀ (n : ℕ),
        1 ≤ n →
          ∀ (z : Shao.Path n),
            (level * √↑n ≤ if Shao.energy z = 0 then 0 else (∑ i, z i) / √(Shao.energy z)) →
              Shao.energy z ≤ level ^ 2 * eps ^ 2 / 2 * ↑n →
                have Small := {i | z i ≠ 0 ∧ |z i| ≤ eps};
                level ^ 2 / 2 * ↑n ≤ ↑Small.card
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
