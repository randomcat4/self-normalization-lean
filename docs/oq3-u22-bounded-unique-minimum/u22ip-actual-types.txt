DECL Shao.OQ3.bounded_actual_exterior_exponential_bound_of_unique_minimum
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (B : ℝ),
  0 ≤ B →
    (∀ᵐ (x : ℝ) ∂mu, |x| ≤ B) →
      ∀ (level a0 rho J0 : ℝ),
        0 < level →
          0 < rho →
            have I := Shao.Probability.bivariateLegendreRate mu fun x => (x, x ^ 2);
            have center := (a0, a0 ^ 2 / level ^ 2);
            have Tail := {w | 0 < w.2 ∧ level * √w.2 ≤ w.1};
            have U := {w | |w.1 - a0| ≤ rho ∧ |w.1 ^ 2 / level ^ 2 - w.2| ≤ rho};
            I center = ↑J0 →
              (∀ w ∈ Tail, I center ≤ I w ∧ (I w = I center ↔ w = center)) →
                ∃ delta,
                  0 < delta ∧
                    ∃ C,
                      0 < C ∧
                        ∀ (n : ℕ),
                          1 ≤ n →
                            have P := MeasureTheory.Measure.pi fun x => mu;
                            have M := fun z => ((∑ i, z i) / ↑n, Shao.energy z / ↑n);
                            P.real (M ⁻¹' (Tail \ U)) ≤ C * Real.exp (-↑n * (J0 + delta))
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
