      ∀ s : ℝ≥0, s ≤ T → |y + B s ω| < a := by

    (hB : IsBrownianReal B P) (a y z : ℝ) (T : ℝ≥0) (hT : T ≠ 0) :
    HasDerivAt (fun v : ℝ => P.real {ω |
      (∀ s : ℝ≥0, s ≤ T → |y + B s ω| < a) ∧ y + B T ω ≤ v})
      (gaussianPDFReal y T z * intervalCompletion (bridgeLaw B P) a y (T : ℝ) z) z := by
