27:   axs.foldl (init := s) fun acc ax => acc.insert ax

29: /--
30: Collect axioms reachable from constant `c`, using `extFind?` to look up pre-computed axioms
31: for imported declarations. Results are cached in `State.seen`.
32: 
33: When processing a constant not found in `extFind?` or the cache, the function temporarily
34: clears the axiom accumulator, recurses into the constant's dependencies, caches the result
35: in `seen`, and merges the collected axioms back.
36: -/
37: partial def collect
38:     (extFind? : Environment → Name → Option (Array Name))
39:     (c : Name) : M Unit := do
40:   let env ← read
41:   -- Check extension for pre-computed axioms (imported declarations)
42:   if let some axs := extFind? env c then
43:     modify fun s => { s with axioms := insertArray s.axioms axs, seen := s.seen.insert c axs }
44:     return
45:   -- Check local cache
46:   let s ← get
47:   if let some axs := s.seen.find? c then
48:     modify fun s => { s with axioms := insertArray s.axioms axs }
49:     return
50:   -- Recurse: temporarily clear axioms to isolate this constant's contribution.
51:   -- Insert sentinel to prevent infinite recursion (e.g., inductives ↔ constructors).
52:   let savedAxioms := s.axioms
53:   modify fun s => { s with axioms := {}, seen := s.seen.insert c #[] }
54:   let collectExpr (e : Expr) : M Unit := e.getUsedConstants.forM (collect extFind?)
55:   -- Take constants from the kernel env, which may differ from the elab env for (async) errors.
56:   match env.checked.get.find? c with
57:   | some (.axiomInfo v)  =>
58:       modify fun s => { s with axioms := s.axioms.insert c }
59:       collectExpr v.type
60:   | some (.defnInfo v)   => collectExpr v.type *> collectExpr v.value
61:   | some (.thmInfo v)    => collectExpr v.type *> collectExpr v.value
62:   | some (.opaqueInfo v) => collectExpr v.type *> collectExpr v.value
63:   | some (.quotInfo _)   => pure ()
64:   | some (.ctorInfo v)   => collectExpr v.type
65:   | some (.recInfo v)    => collectExpr v.type
66:   | some (.inductInfo v) => collectExpr v.type *> v.ctors.forM (collect extFind?)
67:   | none                 => pure ()
68:   -- Cache result (sorted for canonical order) and merge back into saved axioms
69:   let collected := (← get).axioms
70:   let result := collected.toArray.qsort Name.lt
71:   modify fun s => { s with
72:     seen   := s.seen.insert c result
73:     axioms := insertArray savedAxioms result
74:   }

86: /--

88: pre-computed axiom data.
89: 
90: We use `registerPersistentEnvExtension` with manual lookup instead of `MapDeclarationExtension`
91: because `exportEntriesFnEx` needs to call `collect`, which needs the extension's `find?`, but

93: thus cannot reference it. This state replicates `MapDeclarationExtension.find?`'s per-module

99: instance : Inhabited ExportedAxiomsState := ⟨{}⟩

101: /-- Look up pre-computed axioms for an imported declaration. -/
102: def ExportedAxiomsState.find? (s : ExportedAxiomsState) (env : Environment)
103:     (c : Name) : Option (Array Name) :=
104:   match env.getModuleIdxFor? c with
105:   | some modIdx =>
106:     if h : modIdx.toNat < s.importedModuleEntries.size then
107:       match s.importedModuleEntries[modIdx].binSearch (c, #[]) (fun a b => Name.quickLt a.1 b.1) with
108:       | some entry => some entry.2
109:       | none       => none
110:     else none
111:   | none => none

113: /--
114: Environment extension that records axiom dependencies for all declarations in a module.
115: Entries are computed once by `beforeExportFn` when the olean is serialized, not during
116: elaboration. During elaboration, `collectAxioms` walks bodies directly. Downstream modules
117: look up pre-computed entries for imported declarations, so axiom collection never crosses
118: module boundaries.
119: -/
120: builtin_initialize exportedAxiomsExt :
121:     PersistentEnvExtension (Name × Array Name) (Name × Array Name) ExportedAxiomsState ←
122:   registerPersistentEnvExtension {
123:     mkInitial     := pure {}
124:     addImportedFn := fun importedEntries => pure { importedModuleEntries := importedEntries }
125:     addEntryFn    := fun s _ => s
126:     exportEntriesFnEx := fun env s =>
127:       let exportedEnv := env.setExporting true
128:       let privateEnv := env.setExporting false
129:       -- Collect current-module declarations visible in the exported view.
130:       -- By pre-computing axiom data for every exported declaration, downstream modules can
131:       -- look up any imported declaration without walking its body, keeping collection
132:       -- module-local.
133:       let allNames := env.checked.get.constants.foldStage2
134:         (fun names name _ =>
135:           if (exportedEnv.find? name).isSome then names.push name
136:           else names) #[]
137:       -- Compute axioms within a shared state (for caching across declarations).
138:       -- Use `privateEnv` so that `collect` can see all constant bodies.
139:       let entries := CollectAxioms.runM privateEnv do
140:         allNames.mapM fun name =>
141:           return (name, ← CollectAxioms.collectAndGet s.find? name)
142:       -- Sort by name for binary search at import time.
143:       let entries := entries.qsort fun a b => Name.quickLt a.1 b.1
144:       .uniform entries
145:     asyncMode     := .mainOnly
146:   }
