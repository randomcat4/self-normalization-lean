DECL Shao.OQ3.bounded_parabola_tail_global_minimum_of_finite_test
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (M : ℝ),
  0 ≤ M →
    (∀ᵐ (x : ℝ) ∂mu, |x| ≤ M) →
      ∀ (lower upper : ℝ),
        0 < lower →
          (∀ᵐ (x : ℝ) ∂mu, lower ≤ x ^ 2 ∧ x ^ 2 ≤ upper) →
            ∀ (level L : ℝ) (w : ℝ × ℝ),
              level * √w.2 ≤ w.1 →
                Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) w ≤ ↑L →
                  ∃ z,
                    0 < z.2 ∧
                      level * √z.2 ≤ z.1 ∧
                        Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z ≤ ↑L ∧
                          ∀ (y : ℝ × ℝ),
                            0 < y.2 →
                              level * √y.2 ≤ y.1 →
                                Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z ≤
                                  Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) y
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
