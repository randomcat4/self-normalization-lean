DECL Shao.OQ3.bounded_parabola_legendre_compact_sublevels
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (M : ℝ),
  0 ≤ M →
    (∀ᵐ (x : ℝ) ∂mu, |x| ≤ M) →
      ∀ (lower upper : ℝ),
        (∀ᵐ (x : ℝ) ∂mu, lower ≤ x ^ 2 ∧ x ^ 2 ≤ upper) →
          (∀ (z : ℝ × ℝ),
              z.1 < -M ∨ M < z.1 ∨ z.2 < lower ∨ upper < z.2 →
                Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z = ⊤) ∧
            ∀ (L : ℝ), IsCompact {z | Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z ≤ ↑L}
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
