DECL Shao.OQ3.bounded_parabola_legendre_gap_away_mean
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (M : ℝ),
  0 ≤ M →
    (∀ᵐ (x : ℝ) ∂mu, |x| ≤ M) →
      ∀ (r : ℝ),
        0 < r →
          ∃ gap,
            0 < gap ∧
              ∀ (z : ℝ × ℝ),
                r ≤ dist z (∫ (x : ℝ), x ∂mu, ∫ (x : ℝ), x ^ 2 ∂mu) →
                  ↑gap ≤ Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.bounded_parabola_low_rate_localization
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (M : ℝ),
  0 ≤ M →
    (∀ᵐ (x : ℝ) ∂mu, |x| ≤ M) →
      ∀ (r : ℝ),
        0 < r →
          ∃ delta,
            0 < delta ∧
              ∀ (z : ℝ × ℝ),
                Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z ≤ ↑delta →
                  dist z (∫ (x : ℝ), x ∂mu, ∫ (x : ℝ), x ^ 2 ∂mu) < r
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
