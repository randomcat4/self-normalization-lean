DECL Shao.OQ3.bounded_centered_parabola_global_minima_localize
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu] (M : ℝ),
  0 ≤ M →
    (∀ᵐ (x : ℝ) ∂mu, |x| ≤ M) →
      ∀ (lower upper : ℝ),
        0 < lower →
          (∀ᵐ (x : ℝ) ∂mu, lower ≤ x ^ 2 ∧ x ^ 2 ≤ upper) →
            ∫ (x : ℝ), x ∂mu = 0 →
              (Shao.OQ3.parabolaCovariance mu).det ≠ 0 →
                ∀ (r : ℝ),
                  0 < r →
                    ∃ epsilon,
                      0 < epsilon ∧
                        epsilon ≤ 1 ∧
                          ∀ (level : ℝ),
                            0 < level →
                              level < epsilon →
                                ∃ z,
                                  0 < z.2 ∧
                                    level * √z.2 ≤ z.1 ∧
                                      0 < Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z ∧
                                        Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z < ⊤ ∧
                                          (∀ (y : ℝ × ℝ),
                                              0 < y.2 →
                                                level * √y.2 ≤ y.1 →
                                                  Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z ≤
                                                    Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) y) ∧
                                            ∀ (y : ℝ × ℝ),
                                              Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) y ≤
                                                  Shao.Probability.bivariateLegendreRate mu (fun x => (x, x ^ 2)) z →
                                                dist y (0, ∫ (x : ℝ), x ^ 2 ∂mu) < r
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
