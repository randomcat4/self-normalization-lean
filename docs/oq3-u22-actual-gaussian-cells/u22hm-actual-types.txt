DECL Shao.OQ3.nonnegative_weighted_relative_reference_error
TYPE ∀ (phi reference f : ℝ × ℝ → ℝ),
  Measurable phi →
    Measurable reference →
      Measurable f →
        MeasureTheory.Integrable f MeasureTheory.volume →
          (∀ (z : ℝ × ℝ), 0 ≤ f z) →
            (∀ (z : ℝ × ℝ), 0 ≤ phi z) →
              ∀ (B delta : ℝ),
                0 ≤ B →
                  0 ≤ delta →
                    (∀ (z : ℝ × ℝ), phi z ≤ B) →
                      (∀ (z : ℝ × ℝ), f z ≠ 0 → |reference z - phi z| ≤ delta * phi z) →
                        MeasureTheory.Integrable (fun z => f z * phi z) MeasureTheory.volume ∧
                          MeasureTheory.Integrable (fun z => f z * reference z) MeasureTheory.volume ∧
                            0 ≤ ∫ (z : ℝ × ℝ), f z * phi z ∧
                              |(∫ (z : ℝ × ℝ), f z * reference z) - ∫ (z : ℝ × ℝ), f z * phi z| ≤
                                delta * ∫ (z : ℝ × ℝ), f z * phi z
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
