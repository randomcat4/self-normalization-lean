DECL Shao.OQ3.actual_mean_chart_compact_laplace_tilt_windows
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [inst : MeasureTheory.IsProbabilityMeasure mu],
  ∀ theta ∈ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)),
    0 < (Shao.OQ3.parabolaCovariance (mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2)).det →
      ∀ (level a0 : ℝ),
        0 < level →
          0 < a0 →
            theta.2 < 0 →
              (∫ (x : ℝ), x ∂mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2,
                    ∫ (x : ℝ), x ^ 2 ∂mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2) =
                  (a0, a0 ^ 2 / level ^ 2) →
                theta.1 = -2 * a0 * theta.2 / level ^ 2 →
                  ∀ (e : OpenPartialHomeomorph (ℝ × ℝ) (ℝ × ℝ)),
                    (↑e = fun p =>
                        (∫ (x : ℝ), x ∂mu.tilted fun x => p.1 * x + p.2 * x ^ 2,
                          ∫ (x : ℝ), x ^ 2 ∂mu.tilted fun x => p.1 * x + p.2 * x ^ 2)) →
                      theta ∈ e.source →
                        e.source ⊆ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)) →
                          ∀ (rmax : ℝ),
                            0 < rmax →
                              have Omega := interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2));
                              have nu := fun p => mu.tilted fun x => p.1 * x + p.2 * x ^ 2;
                              have means := fun p => (∫ (x : ℝ), x ∂nu p, ∫ (x : ℝ), x ^ 2 ∂nu p);
                              have K := fun p => Real.log (∫ (x : ℝ), Real.exp (p.1 * x + p.2 * x ^ 2) ∂mu);
                              have D := Shao.OQ3.parabolaCovariance (nu theta);
                              have pr0 := ContinuousLinearMap.fst ℝ ℝ ℝ;
                              have pr1 := ContinuousLinearMap.snd ℝ ℝ ℝ;
                              have row0 := D 0 0 • pr0 + D 0 1 • pr1;
                              have row1 := D 0 1 • pr0 + D 1 1 • pr1;
                              ∀ (L : (ℝ × ℝ) ≃L[ℝ] ℝ × ℝ),
                                ↑L = row0.prod row1 →
                                  have J := fun w => (↑e.symm w).1 * w.1 + (↑e.symm w).2 * w.2 - K (↑e.symm w);
                                  have B := fun w => (√(Shao.OQ3.parabolaCovariance (nu (↑e.symm w))).det)⁻¹;
                                  have v := (1, 2 * a0 / level ^ 2);
                                  have Lambda1 :=
                                    2 * theta.2 / level ^ 2 + (L.symm v).1 + 2 * a0 / level ^ 2 * (L.symm v).2;
                                  have C0 := (√D.det)⁻¹ * (√(2 * Real.pi / Lambda1) / -theta.2);
                                  have J0 := J (a0, a0 ^ 2 / level ^ 2);
                                  have m0 := MeasureTheory.volume.prod MeasureTheory.volume;
                                  0 < -theta.2 ∧
                                    (0 < Lambda1 →
                                      ∃ cap,
                                        0 < cap ∧
                                          cap ≤ rmax ∧
                                            0 < C0 ∧
                                              ∀ (rho : ℝ),
                                                0 < rho →
                                                  rho ≤ cap →
                                                    have U :=
                                                      {w | |w.1 - a0| ≤ rho ∧ |w.1 ^ 2 / level ^ 2 - w.2| ≤ rho};
                                                    have Tail := {w | 0 < w.2 ∧ level * √w.2 ≤ w.1};
                                                    have V :=
                                                      {w |
                                                        |w.1 - a0| ≤ rho ∧
                                                          0 ≤ w.1 ^ 2 / level ^ 2 - w.2 ∧
                                                            w.1 ^ 2 / level ^ 2 - w.2 ≤ rho};
                                                    have Theta := ↑e.symm '' V;
                                                    IsCompact Theta ∧
                                                      theta ∈ Theta ∧
                                                        Theta ⊆ Omega ∧
                                                          (∀ p ∈ Theta,
                                                              p ∈ e.source ∧
                                                                means p ∈ V ∧
                                                                  0 < (Shao.OQ3.parabolaCovariance (nu p)).det) ∧
                                                            (∀ w ∈ V, ↑e.symm w ∈ Theta) ∧
                                                              V = U ∩ Tail ∧
                                                                IsCompact V ∧
                                                                  (∀ w ∈ V,
                                                                      0 < w.1 ∧
                                                                        0 < w.2 ∧
                                                                          w ∈ e.target ∧
                                                                            ↑e.symm w ∈ e.source ∧
                                                                              ↑e.symm w ∈ Omega ∧
                                                                                means (↑e.symm w) = w ∧
                                                                                  0 <
                                                                                      (Shao.OQ3.parabolaCovariance
                                                                                          (nu (↑e.symm w))).det ∧
                                                                                    Shao.Probability.bivariateLegendreRate
                                                                                        mu (fun x => (x, x ^ 2)) w =
                                                                                      ↑(J w)) ∧
                                                                    (∀ w ∈ V, J0 ≤ J w) ∧
                                                                      (∀ (aL aR qL qR : ℝ),
                                                                          have A := Set.Ioc aL aR ×ˢ Set.Ioc qL qR ∩ V;
                                                                          have Q := Set.Icc aL aR ×ˢ Set.Icc qL qR;
                                                                          A.Nonempty →
                                                                            ∃ y ∈ closure A,
                                                                              y ∈ Q ∩ V ∧
                                                                                0 < y.1 ∧
                                                                                  0 < y.2 ∧
                                                                                    y ∈ e.target ∧
                                                                                      ↑e.symm y ∈ e.source ∧
                                                                                        ↑e.symm y ∈ Omega ∧
                                                                                          means (↑e.symm y) = y ∧
                                                                                            0 <
                                                                                                (Shao.OQ3.parabolaCovariance
                                                                                                    (nu
                                                                                                      (↑e.symm
                                                                                                        y))).det ∧
                                                                                              Shao.Probability.bivariateLegendreRate
                                                                                                    mu
                                                                                                    (fun x =>
                                                                                                      (x, x ^ 2))
                                                                                                    y =
                                                                                                  ↑(J y) ∧
                                                                                                ∀ w ∈ closure A,
                                                                                                  Shao.Probability.bivariateLegendreRate
                                                                                                        mu
                                                                                                        (fun x =>
                                                                                                          (x, x ^ 2))
                                                                                                        y ≤
                                                                                                      Shao.Probability.bivariateLegendreRate
                                                                                                        mu
                                                                                                        (fun x =>
                                                                                                          (x, x ^ 2))
                                                                                                        w ∧
                                                                                                    J y ≤ J w) ∧
                                                                        ∃ C,
                                                                          0 < C ∧
                                                                            ∀ (n : ℕ),
                                                                              1 ≤ n →
                                                                                have W := fun w =>
                                                                                  B w * Real.exp (-↑n * J w);
                                                                                have T := fun w =>
                                                                                  B w * Real.exp (-↑n * (J w - J0));
                                                                                have Ln :=
                                                                                  ↑n / (2 * Real.pi) *
                                                                                    ∫ (w : ℝ × ℝ) in V, T w ∂m0;
                                                                                have main :=
                                                                                  1 /
                                                                                    (√(2 * Real.pi * ↑n) * -theta.2 *
                                                                                        √D.det *
                                                                                      √Lambda1);
                                                                                MeasureTheory.IntegrableOn W V m0 ∧
                                                                                  MeasureTheory.IntegrableOn T V m0 ∧
                                                                                    0 ≤ Ln ∧
                                                                                      0 < main ∧
                                                                                        |Ln / main - 1| ≤ C / √↑n)
