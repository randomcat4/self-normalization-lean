theorem torus_simultaneous_bad_set_volume (m : ℕ) (q : ℤ) (hq : q ≠ 0) (eps : ℝ) :
    volume {x : Fin m → UnitAddCircle | ∀ j, ‖q • x j‖ ≤ eps} =
        (ENNReal.ofReal (min 1 (2 * eps))) ^ m ∧
      volume {x : Fin m → UnitAddCircle | ∀ j, ‖q • x j‖ ≤ eps} ≤
        (ENNReal.ofReal (2 * eps)) ^ m
