DECL Shao.OQ3.uniform_standardized_pair_quartic_log_remainder
TYPE ∀ {ι : Type u_1} (law : ι → MeasureTheory.Measure (Fin 2 → ℝ)),
  (∀ (i : ι), MeasureTheory.IsProbabilityMeasure (law i)) →
    (∀ (i : ι) (j : Fin 2), MeasureTheory.MemLp (fun y => y j) 2 (law i)) →
      (∀ (i : ι) (j : Fin 2), ∫ (y : Fin 2 → ℝ), y j ∂law i = 0) →
        (∀ (i : ι) (j k : Fin 2),
            ProbabilityTheory.covariance (fun y => y j) (fun y => y k) (law i) = if j = k then 1 else 0) →
          ∀ (M3 M4 M5 : ℝ),
            (∀ (i : ι), MeasureTheory.Integrable (fun y => √(y 0 ^ 2 + y 1 ^ 2) ^ 3) (law i)) →
              (∀ (i : ι), ∫ (y : Fin 2 → ℝ), √(y 0 ^ 2 + y 1 ^ 2) ^ 3 ∂law i ≤ M3) →
                (∀ (i : ι), MeasureTheory.Integrable (fun y => √(y 0 ^ 2 + y 1 ^ 2) ^ 4) (law i)) →
                  (∀ (i : ι), ∫ (y : Fin 2 → ℝ), √(y 0 ^ 2 + y 1 ^ 2) ^ 4 ∂law i ≤ M4) →
                    (∀ (i : ι), MeasureTheory.Integrable (fun y => √(y 0 ^ 2 + y 1 ^ 2) ^ 5) (law i)) →
                      (∀ (i : ι), ∫ (y : Fin 2 → ℝ), √(y 0 ^ 2 + y 1 ^ 2) ^ 5 ∂law i ≤ M5) →
                        have Phi := fun i u v =>
                          ∫ (y : Fin 2 → ℝ), Complex.exp (↑(u * y 0 + v * y 1) * Complex.I) ∂law i;
                        have Q := fun i u v =>
                          -↑(u ^ 2 + v ^ 2) / 2 - Complex.I * ↑(∫ (y : Fin 2 → ℝ), (u * y 0 + v * y 1) ^ 3 ∂law i) / 6 +
                            ↑(∫ (y : Fin 2 → ℝ), (u * y 0 + v * y 1) ^ 4 ∂law i - 3 * (u ^ 2 + v ^ 2) ^ 2) / 24;
                        ∃ C delta,
                          0 < C ∧
                            0 < delta ∧
                              delta ≤ 1 ∧
                                ∀ (i : ι) (u v : ℝ),
                                  √(u ^ 2 + v ^ 2) ≤ delta →
                                    ‖Complex.log (Phi i u v) - Q i u v‖ ≤ C * √(u ^ 2 + v ^ 2) ^ 5
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
