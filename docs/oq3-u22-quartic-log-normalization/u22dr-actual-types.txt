DECL Shao.OQ3.complex_log_taylor_quartic_correction_bound
TYPE ∀ (a b c : ℂ) (r A B C : ℝ),
  0 ≤ r →
    r ≤ 1 →
      0 ≤ A →
        0 ≤ B →
          0 ≤ C →
            ‖a‖ ≤ A * r ^ 2 →
              ‖b‖ ≤ B * r ^ 3 →
                ‖c‖ ≤ C * r ^ 4 →
                  ‖Complex.logTaylor 3 (a + b + c) - (a + b + c - a ^ 2 / 2)‖ ≤ (A * (B + C) + (B + C) ^ 2) * r ^ 5
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
