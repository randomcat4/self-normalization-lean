585: /-! ### Sub-Gaussian moment-generating function -/
586: 
587: variable {Ω : Type*} {m mΩ : MeasurableSpace Ω} {μ : Measure Ω} {X : Ω → ℝ} {c : ℝ≥0}
588: 
589: /-- A random variable `X` has a sub-Gaussian moment-generating function with parameter `c`
590: with respect to a measure `μ` if for all `t : ℝ`, `exp (t * X)` is `μ`-integrable and
591: the moment-generating function of `X` is bounded by `exp (c * t ^ 2 / 2)` for all `t : ℝ`.
592: This implies in particular that `X` has expectation 0.
593: 
594: This is equivalent to `Kernel.HasSubgaussianMGF X c (Kernel.const Unit μ) (Measure.dirac ())`,
595: as proved in `HasSubgaussianMGF_iff_kernel`.
596: Properties about sub-Gaussian moment-generating functions should be proved first for
597: `Kernel.HasSubgaussianMGF` when possible. -/
598: structure HasSubgaussianMGF (X : Ω → ℝ) (c : ℝ≥0) (μ : Measure Ω := by volume_tac) : Prop where
599:   integrable_exp_mul : ∀ t : ℝ, Integrable (fun ω ↦ exp (t * X ω)) μ
600:   mgf_le : ∀ t : ℝ, mgf X μ t ≤ exp (c * t ^ 2 / 2)
601:

603:     HasSubgaussianMGF X c μ

605:   ⟨fun ⟨h1, h2⟩ ↦ ⟨by simpa, by simpa⟩, fun ⟨h1, h2⟩ ↦ ⟨by simpa using h1, by simpa using h2⟩⟩
