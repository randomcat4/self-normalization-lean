theorem eventually_original_boundedPolylog_clipped_comparison_payments
    {Ω : Type*} [mΩ : MeasurableSpace Ω]
    (μ : Measure Ω) [hμ : IsProbabilityMeasure μ]
    (g : Ω → ℝ) (hg : Measurable g) (hg2 : MemLp g 2 μ)
    (hgmean : (∫ o : Ω, g o ∂μ) = 0)
    (hgsecond : (∫ o : Ω, (g o) ^ (2 : ℕ) ∂μ) = 1)
    (C a t0 M rho : ℝ) (hC : 0 ≤ C) (ha : 0 ≤ a)
    (ht0 : 0 < t0) (hM0 : 0 ≤ M) (hrho : 0 ≤ rho) :
    ∃ c : ℝ, 0 < c ∧ ∀ᶠ n : ℕ in Filter.atTop,
      ∀ (r : Ω → Ω → ℝ) (w : Ω → ℝ),
      Measurable (Function.uncurry r) →
      Integrable (Function.uncurry r) (μ.prod μ) →
      (∀ᵐ a ∂μ, (∫ b : Ω, r a b ∂μ) = 0) →
      (∀ᵐ b ∂μ, (∫ a : Ω, r a b ∂μ) = 0) →
      (∀ᵐ o ∂μ, 0 ≤ w o) →
      (∀ᵐ o ∂μ, w o ≤ Shao.OQ5.P51.boundedPolylogHeight C a n) →
      Integrable (fun o : Ω => (w o) ^ (2 : ℕ)) μ →
      (∫ o : Ω, (w o) ^ (2 : ℕ) ∂μ) ≤ M →
      (∀ᵐ z ∂μ.prod μ, |r z.1 z.2| ≤ w z.1 * w z.2) →
      (∀ a b : Ω, r a b = r b a) →
      ∀ y : ℝ, (1 : ℝ) ≤ y →
      y ≤ Shao.OQ5.P51.remainderWindowScale
        (Shao.OQ6.BoundedProductEnvelopeTails.tailScale M) t0 *
          Real.rpow (n : ℝ) ((1 : ℝ) / 6) →
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real
        {x : Fin n → Ω |
          Shao.OQ5.P51.clippedCenteredStatistic n g r 512 rho y x ≠
            (if Real.sqrt (Shao.OQ5.P51.centeredSq n g r x) = 0 then 0 else
              (Shao.OQ5.P51.projectionW n g x + Shao.OQ5.P51.residualStatisticD n r x) /
                Real.sqrt (Shao.OQ5.P51.centeredSq n g r x))} +
        (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.numeratorRemainderBad n g r y) +
        (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.denominatorRemainderBad n g r y) ≤
        Real.exp (-c * (n : ℝ)) +
          (4 : ℝ) * Real.exp (-(y * Real.sqrt (n : ℝ)) / (2 : ℝ)) +
          (9 : ℝ) * Real.exp (-Shao.OQ5.P51.boundedPolylogMomentOrder t0 n) := by
  have hThirdTendsto :
      Filter.Tendsto (fun n : ℕ => Real.rpow (n : ℝ) ((1 : ℝ) / 3)) Filter.atTop Filter.atTop := by
    simpa only [Real.rpow_eq_pow] using
      ((tendsto_rpow_atTop (by norm_num : (0 : ℝ) < (1 : ℝ) / 3)).comp
        (tendsto_natCast_atTop_atTop :
          Filter.Tendsto (fun n : ℕ => (n : ℝ)) Filter.atTop Filter.atTop))
  have hRhoEventually :
      ∀ᶠ n : ℕ in Filter.atTop, 1 + rho ≤ Real.rpow (n : ℝ) ((1 : ℝ) / 3) := by
    exact hThirdTendsto.eventually_ge_atTop (1 + rho)
  have hScaleLeOne :
      Shao.OQ5.P51.remainderWindowScale (Shao.OQ6.BoundedProductEnvelopeTails.tailScale M) t0 ≤ (1 : ℝ) := by
    unfold Shao.OQ5.P51.remainderWindowScale
    exact min_le_left _ _
  have hSigmaNonneg :
      (0 : ℝ) ≤ 1 + rho := by
    linarith only [hrho]
  have hNEventually :
      ∀ᶠ n : ℕ in Filter.atTop, (4 : ℕ) ≤ n := by
    exact Filter.eventually_ge_atTop (4 : ℕ)
  have hWindowEventually :
      ∀ᶠ n : ℕ in Filter.atTop,
      ∀ y : ℝ, y ≤ Shao.OQ5.P51.remainderWindowScale (Shao.OQ6.BoundedProductEnvelopeTails.tailScale M) t0 * Real.rpow (n : ℝ) ((1 : ℝ) / 6) →
      (1 + rho) * y ≤ Real.sqrt (n : ℝ) := by
    filter_upwards [hNEventually, hRhoEventually] with n hn4 hRhoPower
    have hnNatPos :
        (0 : ℕ) < n := by
      exact lt_of_lt_of_le (by decide : (0 : ℕ) < 4) hn4
    have hnRealPos :
        (0 : ℝ) < (n : ℝ) := by
      exact_mod_cast hnNatPos
    have hSixthNonneg :
        (0 : ℝ) ≤ Real.rpow (n : ℝ) ((1 : ℝ) / 6) := by
      exact Real.rpow_nonneg (Nat.cast_nonneg n) ((1 : ℝ) / 6)
    have hPowSum :
        Real.rpow (n : ℝ) ((1 : ℝ) / 3) * Real.rpow (n : ℝ) ((1 : ℝ) / 6) = Real.sqrt (n : ℝ) := by
      calc
        Real.rpow (n : ℝ) ((1 : ℝ) / 3) * Real.rpow (n : ℝ) ((1 : ℝ) / 6) =
            Real.rpow (n : ℝ) (((1 : ℝ) / 3) + ((1 : ℝ) / 6)) :=
          (Real.rpow_add hnRealPos ((1 : ℝ) / 3) ((1 : ℝ) / 6)).symm
        _ = Real.rpow (n : ℝ) ((1 : ℝ) / 2) := by congr 1 <;> norm_num
        _ = Real.sqrt (n : ℝ) := (Real.sqrt_eq_rpow (n : ℝ)).symm
    intro y hyWindow
    have hySixth :
        y ≤ Real.rpow (n : ℝ) ((1 : ℝ) / 6) := by
      exact hyWindow.trans (by
        simpa only [one_mul] using
          mul_le_mul_of_nonneg_right hScaleLeOne hSixthNonneg)
    have hFinalWindow :
        (1 + rho) * y ≤ Real.sqrt (n : ℝ) := by
      calc
        (1 + rho) * y ≤ (1 + rho) * Real.rpow (n : ℝ) ((1 : ℝ) / 6) :=
          mul_le_mul_of_nonneg_left hySixth hSigmaNonneg
        _ ≤ Real.rpow (n : ℝ) ((1 : ℝ) / 3) * Real.rpow (n : ℝ) ((1 : ℝ) / 6) :=
          mul_le_mul_of_nonneg_right hRhoPower hSixthNonneg
        _ = Real.sqrt (n : ℝ) := hPowSum
    exact hFinalWindow
  obtain ⟨c, hc, hFourEventually⟩ :=
    Shao.OQ6.OriginalBoundedFourPayments.eventually_original_boundedPolylog_four_payments
      μ g hg hg2 hgmean hgsecond C a t0 M hC ha ht0 hM0
  refine ⟨c, hc, ?_⟩
  filter_upwards [hFourEventually, hWindowEventually, hNEventually] with n hFourAtN hWindowAtN hn4
  haveI hSampleProbability :
      IsProbabilityMeasure (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ) :=
    inferInstanceAs (IsProbabilityMeasure (Measure.pi (fun _ : Fin n => μ)))
  intro r w hr hr1 hcR hcL hW hWR hSq hSqBound hEnv hsym y hy hyWindow
  have hwindow :
      (1 + rho) * y ≤ Real.sqrt (n : ℝ) := by
    exact hWindowAtN y hyWindow
  have hSubset :
      {x : Fin n → Ω |
      Shao.OQ5.P51.clippedCenteredStatistic n g r 512 rho y x ≠
      (if Real.sqrt (Shao.OQ5.P51.centeredSq n g r x) = 0 then 0 else
      (Shao.OQ5.P51.projectionW n g x + Shao.OQ5.P51.residualStatisticD n r x) /
      Real.sqrt (Shao.OQ5.P51.centeredSq n g r x))} ⊆ (Shao.OQ5.P51.projectionGood n g y)ᶜ ∪ Shao.OQ5.P51.lowerStudentizerBad n g r y := by
    exact Shao.OQ5.P51.literal_clippedCentered_comparison_exception_subset
      n g r rho y hn4 hy hsym hrho hwindow
  have hUnionFinite :
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ) ((Shao.OQ5.P51.projectionGood n g y)ᶜ ∪ Shao.OQ5.P51.lowerStudentizerBad n g r y) ≠ ⊤ := by
    finiteness
  have hComparisonProbability :
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real {x : Fin n → Ω |
      Shao.OQ5.P51.clippedCenteredStatistic n g r 512 rho y x ≠
      (if Real.sqrt (Shao.OQ5.P51.centeredSq n g r x) = 0 then 0 else
      (Shao.OQ5.P51.projectionW n g x + Shao.OQ5.P51.residualStatisticD n r x) /
      Real.sqrt (Shao.OQ5.P51.centeredSq n g r x))} ≤ (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.projectionGood n g y)ᶜ + (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.lowerStudentizerBad n g r y) := by
    exact (MeasureTheory.measureReal_mono hSubset hUnionFinite).trans
      (MeasureTheory.measureReal_union_le
        (μ := Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ)
        (Shao.OQ5.P51.projectionGood n g y)ᶜ
        (Shao.OQ5.P51.lowerStudentizerBad n g r y))
  have hFourPaymentBound :
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.projectionGood n g y)ᶜ +
      Shao.OQ5.P51.remainderPayment n (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ) g r y ≤
      Real.exp (-c * (n : ℝ)) +
      (4 : ℝ) * Real.exp (-(y * Real.sqrt (n : ℝ)) / (2 : ℝ)) +
      (9 : ℝ) * Real.exp (-Shao.OQ5.P51.boundedPolylogMomentOrder t0 n) := by
    exact hFourAtN r w hr hr1 hcR hcL hW hWR hSq hSqBound hEnv hsym
      y hy hyWindow
  have hPaymentComparison :
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real
      {x : Fin n → Ω |
      Shao.OQ5.P51.clippedCenteredStatistic n g r 512 rho y x ≠
      (if Real.sqrt (Shao.OQ5.P51.centeredSq n g r x) = 0 then 0 else
      (Shao.OQ5.P51.projectionW n g x + Shao.OQ5.P51.residualStatisticD n r x) /
      Real.sqrt (Shao.OQ5.P51.centeredSq n g r x))} +
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.numeratorRemainderBad n g r y) +
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.denominatorRemainderBad n g r y) ≤
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.projectionGood n g y)ᶜ +
      Shao.OQ5.P51.remainderPayment n (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ) g r y := by
    simpa only [Shao.OQ5.P51.remainderPayment, add_assoc] using
      add_le_add_right
        (add_le_add_right hComparisonProbability
          ((Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real
            (Shao.OQ5.P51.numeratorRemainderBad n g r y)))
        ((Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real
          (Shao.OQ5.P51.denominatorRemainderBad n g r y))
  have hFinalPayments :
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real
      {x : Fin n → Ω |
      Shao.OQ5.P51.clippedCenteredStatistic n g r 512 rho y x ≠
      (if Real.sqrt (Shao.OQ5.P51.centeredSq n g r x) = 0 then 0 else
      (Shao.OQ5.P51.projectionW n g x + Shao.OQ5.P51.residualStatisticD n r x) /
      Real.sqrt (Shao.OQ5.P51.centeredSq n g r x))} +
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.numeratorRemainderBad n g r y) +
      (Shao.OQ6.CanonicalDoubleGhostRowMoments.sampleLaw n μ).real (Shao.OQ5.P51.denominatorRemainderBad n g r y) ≤
      Real.exp (-c * (n : ℝ)) +
      (4 : ℝ) * Real.exp (-(y * Real.sqrt (n : ℝ)) / (2 : ℝ)) +
      (9 : ℝ) * Real.exp (-Shao.OQ5.P51.boundedPolylogMomentOrder t0 n) := by
    exact hPaymentComparison.trans hFourPaymentBound
  exact hFinalPayments

end Shao.OQ6.OriginalClippedComparisonPayments
