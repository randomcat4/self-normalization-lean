DECL Shao.OQ3.pair_standard_gaussian_hermite_integral
TYPE ∀ (m n : ℕ) (x : ℝ × ℝ),
  MeasureTheory.Integrable
      (fun z =>
        Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * (Complex.I * ↑z.1) ^ m * (Complex.I * ↑z.2) ^ n *
          Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2))
      MeasureTheory.volume ∧
    ∫ (z : ℝ × ℝ),
        Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * (Complex.I * ↑z.1) ^ m * (Complex.I * ↑z.2) ^ n *
          Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) =
      ↑(2 * Real.pi * Polynomial.eval x.1 (Shao.OQ3.gaussianHermitePolynomial m) *
            Polynomial.eval x.2 (Shao.OQ3.gaussianHermitePolynomial n) *
          Real.exp (-(x.1 ^ 2 + x.2 ^ 2) / 2))
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
