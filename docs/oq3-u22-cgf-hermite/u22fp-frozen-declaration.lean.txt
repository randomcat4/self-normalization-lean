theorem gaussian_hermite_scaled_complex_iterated_derivative
    (c a : ℝ) (n : ℕ) (x : ℝ) :
    iteratedDeriv n (fun y : ℝ =>
      (c : ℂ) * Complex.exp (((-(a * y) ^ 2 / 2 : ℝ) : ℂ))) x =
        ((c * a ^ n *
          ((-1 : ℝ) ^ n * (gaussianHermitePolynomial n).eval (a * x) *
            Real.exp (-(a * x) ^ 2 / 2)) : ℝ) : ℂ)
