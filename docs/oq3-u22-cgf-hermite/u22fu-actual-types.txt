TYPE ∀ (mu : MeasureTheory.Measure (Fin 2 → ℝ)) [MeasureTheory.IsProbabilityMeasure mu],
  (∀ (u v : ℝ), 0 ∈ interior (ProbabilityTheory.integrableExpSet (fun y => u * y 0 + v * y 1) mu)) →
    (∀ (k : ℕ), MeasureTheory.Integrable (fun y => √(y 0 ^ 2 + y 1 ^ 2) ^ k) mu) →
      ∀ (n : ℕ),
        1 ≤ n →
          have kappa := fun z j => iteratedDeriv j (ProbabilityTheory.cgf (fun y => z.1 * y 0 + z.2 * y 1) mu) 0;
          have a := fun z j => Complex.I ^ (↑j + 3) / ↑(↑j + 3).factorial * ↑(kappa z (↑j + 3)) / ↑√↑n ^ (↑j + 1);
          have S := fun z =>
            ∑ q, 1 / ↑(↑q).factorial * ∑ sigma, if ∑ i, (↑(sigma i) + 1) ≤ 6 then ∏ i, a z (sigma i) else 0;
          ∃ P,
            (∀ (z : ℝ × ℝ), MvPolynomial.eval₂ Complex.ofRealHom ![Complex.I * ↑z.1, Complex.I * ↑z.2] P = S z) ∧
              ∀ (x : ℝ × ℝ),
                MeasureTheory.Integrable
                    (fun z =>
                      Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * S z *
                        Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2))
                    MeasureTheory.volume ∧
                  ∫ (z : ℝ × ℝ),
                      Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * S z *
                        Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) =
                    ↑((2 * Real.pi *
                          ∑ d ∈ P.support,
                            P.coeff d * Polynomial.eval x.1 (Shao.OQ3.gaussianHermitePolynomial (d 0)) *
                              Polynomial.eval x.2 (Shao.OQ3.gaussianHermitePolynomial (d 1))) *
                        Real.exp (-(x.1 ^ 2 + x.2 ^ 2) / 2))
