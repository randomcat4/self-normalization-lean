DECL Shao.OQ3.normalized_iid_pair_compact_middle_frequency_bound
TYPE ∀ (law : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure law] (beta c R : ℝ),
  0 ≤ beta →
    0 < c →
      0 ≤ R →
        (∀ (u v : ℝ),
            R < √(u ^ 2 + v ^ 2) →
              ‖∫ (w : ℝ × ℝ), Complex.exp (↑(u * w.1 + v * w.2) * Complex.I) ∂law‖ ≤
                1 - c * (√(u ^ 2 + v ^ 2)).rpow (-beta)) →
          ∀ (K : Set (ℝ × ℝ)),
            IsCompact K →
              (∀ a ∈ K, a.1 ≠ 0 ∨ a.2 ≠ 0) →
                ∃ gamma,
                  0 < gamma ∧
                    gamma < 1 ∧
                      ∀ (n : ℕ),
                        1 ≤ n →
                          ∀ (xi eta : ℝ),
                            (xi / √↑n, eta / √↑n) ∈ K →
                              have Q := MeasureTheory.Measure.pi fun x => law;
                              have S := fun z => (√↑n)⁻¹ • ∑ i, z i;
                              have rho := MeasureTheory.Measure.map S Q;
                              ‖∫ (w : ℝ × ℝ), Complex.exp (↑(xi * w.1 + eta * w.2) * Complex.I) ∂rho‖ ≤ gamma ^ n
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
