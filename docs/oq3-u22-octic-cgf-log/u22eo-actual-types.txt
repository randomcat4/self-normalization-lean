DECL Shao.OQ3.actual_cgf_iterated_deriv_scaling
TYPE ∀ {Ω : Type u_1} [inst : MeasurableSpace Ω] (mu : MeasureTheory.Measure Ω) [MeasureTheory.IsProbabilityMeasure mu]
  (X : Ω → ℝ) (a : ℝ) (k : ℕ) (t : ℝ),
  iteratedDeriv k (ProbabilityTheory.cgf (fun w => a * X w) mu) t =
    a ^ k * iteratedDeriv k (ProbabilityTheory.cgf X mu) (a * t)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
