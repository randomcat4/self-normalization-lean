DECL Shao.OQ3.complex_log_taylor_octic_correction_bound
TYPE ∀ (A B C D E F G : ℝ),
  0 ≤ A →
    0 ≤ B →
      0 ≤ C →
        0 ≤ D →
          0 ≤ E →
            0 ≤ F →
              0 ≤ G →
                ∃ K,
                  0 < K ∧
                    ∀ (r : ℝ),
                      0 ≤ r →
                        r ≤ 1 →
                          ∀ (a b c d e f g : ℂ),
                            ‖a‖ ≤ A * r ^ 2 →
                              ‖b‖ ≤ B * r ^ 3 →
                                ‖c‖ ≤ C * r ^ 4 →
                                  ‖d‖ ≤ D * r ^ 5 →
                                    ‖e‖ ≤ E * r ^ 6 →
                                      ‖f‖ ≤ F * r ^ 7 →
                                        ‖g‖ ≤ G * r ^ 8 →
                                          have P := a + b + c + d + e + f + g;
                                          have Q :=
                                            a + b + c + d + e + f + g -
                                                  (a ^ 2 + 2 * a * b + 2 * a * c + b ^ 2 + 2 * a * d + 2 * b * c +
                                                          2 * a * e +
                                                        2 * b * d +
                                                      c ^ 2) /
                                                    2 +
                                                (a ^ 3 + 3 * a ^ 2 * b + 3 * a ^ 2 * c + 3 * a * b ^ 2) / 3 -
                                              a ^ 4 / 4;
                                          ‖Complex.logTaylor 5 P - Q‖ ≤ K * r ^ 9
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
