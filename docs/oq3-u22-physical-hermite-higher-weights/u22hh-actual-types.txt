DECL Shao.OQ3.complex_six_order_higher_weight_growth_bound
TYPE ∀ (A : Fin 6 → ℝ),
  (∀ (j : Fin 6), 0 ≤ A j) →
    have W := fun a =>
      ∑ q,
        1 / ↑(↑q).factorial *
          ∑ sigma, if 2 ≤ ∑ l, (↑(sigma l) + 1) ∧ ∑ l, (↑(sigma l) + 1) ≤ 6 then ∏ l, a (sigma l) else 0;
    ∃ K,
      0 < K ∧
        ∀ (t r : ℝ),
          0 ≤ t →
            t ≤ 1 →
              0 ≤ r →
                ∀ (a : Fin 6 → ℂ),
                  (∀ (j : Fin 6), ‖a j‖ ≤ A j * t ^ (↑j + 1) * r ^ (↑j + 3)) → ‖W a‖ ≤ K * t ^ 2 * (1 + r ^ 18)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
