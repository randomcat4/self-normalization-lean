theorem actual_double_stopped_telescope
    {Ω : Type*} [MeasurableSpace Ω]
    (B : ℝ≥0 → Ω → ℝ) (PB : Measure Ω) [IsProbabilityMeasure PB]
    (hB : IsBrownianReal B PB)
    (n : ℕ) (P : Measure (Shao.Path n)) [IsProbabilityMeasure P]
    (x : ℝ) (hx : 0 < x) (H : ℝ) (hH : 0 < H)
    (nu : Shao.Path n → LabelPerm n → ℕ)
    (hnum : ∀ sigma, Measurable (fun a => nu a sigma))
    (hnu : ∀ a, IsRevealStoppingRank (nu a))
    (hcap : ∀ a sigma, nu a sigma ≤ energyCutoff a H sigma)
    (m : Shao.Path n → ℝ) (hmm : Measurable m) (hmb : ∀ a, |m a| ≤ 1)
    (hminv : PermutationInvariant m) :
    let Q : Measure BridgeNoise := bridgeLaw B PB
    let Z : (Shao.Path n × LabelPerm n) → ℝ := fun p =>
      m p.1 * (actualDoubleValue Q x p.1 (nu p.1 p.2) p.2 -
        Shao.C2.InitialDoubleCompletionBudget.initialDoubleObservable Q x p.1)
    let L : (Shao.Path n × LabelPerm n) → ℝ := fun p =>
      m p.1 * ∑ r ∈ Finset.range n, actualDoubleRetained Q x p.1 (nu p.1) r p.2
    let O : (Shao.Path n × LabelPerm n) → ℝ := fun p =>
      m p.1 * ∑ r ∈ Finset.range n, actualDoubleSeparated Q x p.1 (nu p.1) r p.2
    (∀ a sigma r, 0 ≤ actualDoubleSeparated Q x a (nu a) r sigma) ∧
    Integrable Z (revealLaw P) ∧ Integrable L (revealLaw P) ∧ Integrable O (revealLaw P) ∧
    (∫ p, Z p ∂revealLaw P) = (∫ p, L p ∂revealLaw P) - (∫ p, O p ∂revealLaw P) ∧
    ((∀ a, 0 ≤ m a) → (∫ p, Z p ∂revealLaw P) ≤ (∫ p, L p ∂revealLaw P))

    (nu : Measure BridgeNoise) (x : ℝ) (a : Shao.Path n) : ℝ :=
  if Shao.energy a = 0 then 1

def brownianInitialDouble {Ω : Type*}
    (B : ℝ≥0 → Ω → ℝ) (x : ℝ) : Set Ω :=
  {omega | (∃ t : BridgeTime, x ≤ B ⟨(t : ℝ), t.property.1⟩ omega) ∧

DECLARATION original_finite_second_initial_double_completion_budget SOURCE_LINE 559
theorem original_finite_second_initial_double_completion_budget :
    ∃ C : ℝ, 0 < C ∧
      ∀ (rho : Measure ℝ) [IsProbabilityMeasure rho]
        (n : ℕ) (hn : 0 < n),
        Integrable (fun z : ℝ => z ^ 2) rho →
        (∫ z : ℝ, z ∂rho) = 0 →
        (∫ z : ℝ, z ^ 2 ∂rho) = 1 →
        ∀ (Ω : Type*) [MeasurableSpace Ω]
          (B : ℝ≥0 → Ω → ℝ) (PB : Measure Ω) [IsProbabilityMeasure PB],
          IsBrownianReal B PB →
          ∀ x : ℝ, 1 ≤ x →
            let normalizedLaw : Measure ℝ :=
              Measure.map (fun z : ℝ => z / Real.sqrt (n : ℝ)) rho
            let b : ℝ := Shao.truncationBudget normalizedLaw n x
            let P : Measure (Shao.Path n) := Measure.pi (fun _ : Fin n => rho)
            let W : Shao.Path n → ℝ := Shao.C1.normalizedEndpoint
            let Z : Shao.Path n → ℝ := initialDoubleObservable (bridgeLaw B PB) x
            b ≤ 1 / 2 →
              Integrable Z P ∧
              |(∫ a : Shao.Path n, Z a ∂P) - PB.real (brownianInitialDouble B x)| ≤
                C * b * Shao.standardNormalTail x ∧
              ∀ v : ℝ,
                Integrable (fun a : Shao.Path n => if W a ≤ v then Z a else 0) P ∧
                |(∫ a : Shao.Path n, (if W a ≤ v then Z a else 0) ∂P) -
                  PB.real (brownianInitialDouble B x ∩ {omega : Ω | B 1 omega ≤ v})| ≤
                    C * b * Shao.standardNormalTail x

  nu.real ((strictUpper x 0 1 w)ᶜ ∩ (strictLower x 0 1 w)ᶜ)

theorem actual_initial_double_completion_kernel
    {Ω : Type*} [MeasurableSpace Ω]
    (B : ℝ≥0 → Ω → ℝ) (P : Measure Ω) [IsProbabilityMeasure P]
