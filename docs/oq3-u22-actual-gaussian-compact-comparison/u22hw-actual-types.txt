DECL Shao.OQ3.compact_plane_uniform_quadratic_taylor_and_amplitude
TYPE ∀ (f A : ℝ × ℝ → ℝ) (U K : Set (ℝ × ℝ)),
  IsOpen U →
    IsCompact K →
      K ⊆ U →
        (∀ x ∈ U, ContDiffAt ℝ 2 f x) →
          (∀ x ∈ U, ContDiffAt ℝ 1 A x) →
            ∃ delta C,
              0 < delta ∧
                0 < C ∧
                  ∀ x ∈ K,
                    ∀ (y : ℝ × ℝ),
                      ‖y - x‖ ≤ delta →
                        y ∈ U ∧ |f y - f x - (fderiv ℝ f x) (y - x)| ≤ C * ‖y - x‖ ^ 2 ∧ |A y - A x| ≤ C * ‖y - x‖
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
