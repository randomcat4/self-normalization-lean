  (↑e = fun p =>
      (∫ (x : ℝ), x ∂mu.tilted fun x => p.1 * x + p.2 * x ^ 2,
        ∫ (x : ℝ), x ^ 2 ∂mu.tilted fun x => p.1 * x + p.2 * x ^ 2)) →
    e.source ⊆ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)) →
      ∀ (V : Set (ℝ × ℝ)),
        IsCompact V →
          V ⊆ e.target →
            (∀ y ∈ V,
                0 < (Shao.OQ3.parabolaCovariance (mu.tilted fun x => (↑e.symm y).1 * x + (↑e.symm y).2 * x ^ 2)).det) →
              have nu := fun p => mu.tilted fun x => p.1 * x + p.2 * x ^ 2;
              have D := fun y => Shao.OQ3.parabolaCovariance (nu (↑e.symm y));
              have K := fun p => Real.log (∫ (x : ℝ), Real.exp (p.1 * x + p.2 * x ^ 2) ∂mu);
              have J := fun y => (↑e.symm y).1 * y.1 + (↑e.symm y).2 * y.2 - K (↑e.symm y);
              have A := fun y => (√(D y).det)⁻¹;
              have Q := fun y v => (D y 1 1 * v.1 ^ 2 - 2 * D y 0 1 * v.1 * v.2 + D y 0 0 * v.2 ^ 2) / (D y).det;

                0 < C ∧

                        N ≤ n →

                                          Real.exp

                                              (J y + ((↑e.symm y).1 * (w - y).1 + (↑e.symm y).2 * (w - y).2) +
                                                  Q y (w - y) / 2 -
                                                J0)) -
                                        A w * Real.exp (-↑n * (J w - J0))| ≤
                                    C / ↑n * (A w * Real.exp (-↑n * (J w - J0)))
