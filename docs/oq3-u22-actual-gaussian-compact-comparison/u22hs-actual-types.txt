DECL Shao.OQ3.standardized_exponential_gaussian_cell_pullback
TYPE ∀ (n : ℕ),
  1 ≤ n →
    ∀ (E : (ℝ × ℝ) ≃L[ℝ] ℝ × ℝ) (y theta : ℝ × ℝ) (A : Set (ℝ × ℝ)),
      MeasurableSet A →
        have F := fun w => √↑n • E (w - y);
        have phi := fun z => 1 / (2 * Real.pi) * Real.exp (-(z.1 ^ 2 + z.2 ^ 2) / 2);
        have g := fun z => Real.exp (-√↑n * (theta.1 * (E.symm z).1 + theta.2 * (E.symm z).2));
        MeasurableSet (F '' A) ∧
          ∫ (z : ℝ × ℝ), (F '' A).indicator g z * phi z =
            ↑n * |LinearMap.det ↑↑E| / (2 * Real.pi) *
              ∫ (w : ℝ × ℝ) in A,
                Real.exp
                  (-↑n * (theta.1 * (w - y).1 + theta.2 * (w - y).2) - ↑n / 2 * ((E (w - y)).1 ^ 2 + (E (w - y)).2 ^ 2))
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.standardized_plane_affine_jacobian
TYPE ∀ (n : ℕ) (E : (ℝ × ℝ) ≃L[ℝ] ℝ × ℝ), |LinearMap.det ↑(√↑n • ↑E)| = ↑n * |LinearMap.det ↑↑E|
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
