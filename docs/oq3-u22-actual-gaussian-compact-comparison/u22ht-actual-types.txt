DECL Shao.OQ3.canonical_pair_whitening_determinant_quadratic
TYPE ∀ (a b d : ℝ),
  0 < a →
    0 < a * d - b ^ 2 →
      ∀ (E : (ℝ × ℝ) ≃L[ℝ] ℝ × ℝ),
        have s := √(a * d - b ^ 2);
        have t := √(a + d + 2 * s);
        (∀ (w : ℝ × ℝ),
            E w = ((d + s) / (s * t) * w.1 + -b / (s * t) * w.2, -b / (s * t) * w.1 + (a + s) / (s * t) * w.2)) →
          LinearMap.det ↑↑E = 1 / s ∧
            |LinearMap.det ↑↑E| = 1 / s ∧
              ∀ (w : ℝ × ℝ),
                (E w).1 ^ 2 + (E w).2 ^ 2 = (d * w.1 ^ 2 - 2 * b * w.1 * w.2 + a * w.2 ^ 2) / (a * d - b ^ 2)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
