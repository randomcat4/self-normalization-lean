DECL Shao.OQ3.imaginary_exponential_quadratic_remainder_bound
TYPE ∀ (t : ℝ), ‖Complex.exp (↑t * Complex.I) - (1 + ↑t * Complex.I - ↑t ^ 2 / 2)‖ ≤ (Real.exp 1 + 4) * |t| ^ 3
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.probability_characteristic_quadratic_remainder_bound
TYPE ∀ (nu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure nu],
  MeasureTheory.Integrable (fun x => x) nu →
    MeasureTheory.Integrable (fun x => x ^ 2) nu →
      MeasureTheory.Integrable (fun x => |x| ^ 3) nu →
        ∫ (x : ℝ), x ∂nu = 0 →
          ∀ (t : ℝ),
            ‖∫ (x : ℝ), Complex.exp (↑(t * x) * Complex.I) ∂nu - (1 - ↑t ^ 2 * ↑(∫ (x : ℝ), x ^ 2 ∂nu) / 2)‖ ≤
              (Real.exp 1 + 4) * |t| ^ 3 * ∫ (x : ℝ), |x| ^ 3 ∂nu
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
