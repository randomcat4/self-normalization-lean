DECL Shao.OQ3.exists_positive_parabola_window
TYPE ∀ (level a0 cap : ℝ),
  0 < level →
    0 < a0 →
      0 < cap →
        ∃ rho,
          0 < rho ∧
            rho ≤ cap ∧
              ∀ (r z : ℝ),
                |r| ≤ rho →
                  |z| ≤ rho →
                    0 < a0 + r ∧
                      0 < (a0 + r) ^ 2 / level ^ 2 - z ∧ (level * √((a0 + r) ^ 2 / level ^ 2 - z) ≤ a0 + r ↔ 0 ≤ z)
AXIOMS [propext, Classical.choice, Quot.sound]
DECL Shao.OQ3.self_normalized_boundary_event_iff
TYPE ∀ (level a q : ℝ), 0 < level → 0 < q → (level * √q ≤ a ↔ 0 < a ∧ q ≤ a ^ 2 / level ^ 2)
AXIOMS [propext, Classical.choice, Quot.sound]
