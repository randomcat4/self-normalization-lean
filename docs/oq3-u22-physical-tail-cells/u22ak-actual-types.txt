DECL Shao.OQ3.positive_parabola_window_event_geometry
TYPE ∀ (level a0 rho : ℝ),
  0 < level →
    0 < rho →
      (∀ (r z : ℝ), |r| ≤ rho → |z| ≤ rho → 0 < a0 + r ∧ 0 < (a0 + r) ^ 2 / level ^ 2 - z) →
        have U := {w | |w.1 - a0| ≤ rho ∧ |w.1 ^ 2 / level ^ 2 - w.2| ≤ rho};
        have V := {w | |w.1 - a0| ≤ rho ∧ w.1 ^ 2 / level ^ 2 - rho ≤ w.2 ∧ w.2 ≤ w.1 ^ 2 / level ^ 2};
        have D := {w | 0 < w.2 ∧ level * √w.2 ≤ w.1};
        IsClosed U ∧ (a0, a0 ^ 2 / level ^ 2) ∈ interior U ∧ (∀ w ∈ U, 0 < w.1 ∧ 0 < w.2) ∧ V = U ∩ D
AXIOMS [propext, Classical.choice, Quot.sound]
