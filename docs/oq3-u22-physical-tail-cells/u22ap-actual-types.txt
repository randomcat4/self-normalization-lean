DECL Shao.OQ3.standardized_cell_exponential_bounds
TYPE ∀ (n : ℕ),
  1 ≤ n →
    ∀ (theta : ℝ × ℝ) (E : (ℝ × ℝ) ≃L[ℝ] ℝ × ℝ) (R : ℝ),
      0 ≤ R →
        have B := (|theta.1| + |theta.2|) * ‖↑E.symm‖;
        have g := fun v => Real.exp (-√↑n * (theta.1 * (E.symm v).1 + theta.2 * (E.symm v).2));
        (∀ (v : ℝ × ℝ), ‖v‖ ≤ R / √↑n → 0 < g v ∧ |g v| ≤ Real.exp (B * R)) ∧
          ∀ (v w : ℝ × ℝ), ‖v‖ ≤ R / √↑n → ‖w‖ ≤ R / √↑n → |g v - g w| ≤ √↑n * B * Real.exp (B * R) * ‖v - w‖
AXIOMS [propext, Classical.choice, Quot.sound]
