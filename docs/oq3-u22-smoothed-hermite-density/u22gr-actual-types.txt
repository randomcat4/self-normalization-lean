DECL Shao.OQ3.compact_open_domain_six_order_log_cubic_damped_window
TYPE ∀ (mu : MeasureTheory.Measure ℝ) [MeasureTheory.IsProbabilityMeasure mu] (Theta : Set (ℝ × ℝ)),
  IsCompact Theta →
    Theta ⊆ interior (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)) →
      ∀ (beta c R : ℝ),
        0 ≤ beta →
          beta < 2 / 5 →
            0 < c →
              0 ≤ R →
                (∀ p ∈ Theta, 0 < (Shao.OQ3.parabolaCovariance (mu.tilted fun x => p.1 * x + p.2 * x ^ 2)).det) →
                  (∀ p ∈ Theta,
                      ∀ (xi eta : ℝ),
                        R < √(xi ^ 2 + eta ^ 2) →
                          ‖Shao.OQ3.jointCharacteristic (mu.tilted fun x => p.1 * x + p.2 * x ^ 2) xi eta‖ ≤
                            1 - c * (√(xi ^ 2 + eta ^ 2)).rpow (-beta)) →
                    have tilt := fun p => mu.tilted fun x => p.1 * x + p.2 * x ^ 2;
                    have D := fun p => Shao.OQ3.parabolaCovariance (tilt p);
                    have s := fun p => √(D p 0 0 * D p 1 1 - D p 0 1 ^ 2);
                    have t := fun p => √(D p 0 0 + D p 1 1 + 2 * s p);
                    have A := fun p =>
                      ![![(D p 1 1 + s p) / (s p * t p), -D p 0 1 / (s p * t p)],
                        ![-D p 0 1 / (s p * t p), (D p 0 0 + s p) / (s p * t p)]];
                    have means := fun p => ![∫ (x : ℝ), x ∂tilt p, ∫ (x : ℝ), x ^ 2 ∂tilt p];
                    have Y := fun p x => (A p).mulVec (![x, x ^ 2] - means p);
                    have laws := fun p => MeasureTheory.Measure.map (Y p) (tilt p);
                    have Phi := fun p xi eta =>
                      ∫ (y : Fin 2 → ℝ), Complex.exp (↑(xi * y 0 + eta * y 1) * Complex.I) ∂laws p;
                    ∃ lo hi,
                      0 < lo ∧
                        0 < hi ∧
                          (∀ p ∈ Theta,
                              ∀ (xi eta : ℝ),
                                lo * (xi ^ 2 + eta ^ 2) ≤
                                    xi ^ 2 * D p 0 0 + 2 * xi * eta * D p 0 1 + eta ^ 2 * D p 1 1 ∧
                                  xi ^ 2 * D p 0 0 + 2 * xi * eta * D p 0 1 + eta ^ 2 * D p 1 1 ≤
                                    hi * (xi ^ 2 + eta ^ 2)) ∧
                            ∃ E,
                              (∀ (p : ↑Theta),
                                  (∀ (w : ℝ × ℝ),
                                      (E p) w =
                                        (A (↑p) 0 0 * w.1 + A (↑p) 0 1 * w.2, A (↑p) 1 0 * w.1 + A (↑p) 1 1 * w.2)) ∧
                                    ‖↑(E p)‖ ≤ 2 * (1 / √lo) ∧
                                      ‖↑(E p).symm‖ ≤ 4 * hi * (1 / √lo) ∧
                                        ∀ (x : ℝ),
                                          (E p) ((x, x ^ 2) - (means (↑p) 0, means (↑p) 1)) =
                                            (Y (↑p) x 0, Y (↑p) x 1)) ∧
                                ∀ (k : ℕ),
                                  ∃ C,
                                    0 < C ∧
                                      ∃ N,
                                        2 ≤ N ∧
                                          (∀ (n : ℕ),
                                              N ≤ n →
                                                ∀ (p : ↑Theta) (b : ℝ),
                                                  0 ≤ b →
                                                    have Qsample := MeasureTheory.Measure.pi fun x => tilt ↑p;
                                                    have M := fun z => ((∑ i, z i) / ↑n, Shao.energy z / ↑n);
                                                    have T := fun z => √↑n • (E p) (M z - (means (↑p) 0, means (↑p) 1));
                                                    have rho := MeasureTheory.Measure.map T Qsample;
                                                    have X := fun z y => z.1 * y 0 + z.2 * y 1;
                                                    have kappa := fun j z =>
                                                      iteratedDeriv j (ProbabilityTheory.cgf (X z) (laws ↑p)) 0;
                                                    have a := fun z j =>
                                                      Complex.I ^ (↑j + 3) / ↑(↑j + 3).factorial * ↑(kappa (↑j + 3) z) /
                                                        ↑√↑n ^ (↑j + 1);
                                                    have Pn := fun z =>
                                                      Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) *
                                                        ∑ q,
                                                          1 / ↑(↑q).factorial *
                                                            ∑ sigma,
                                                              if ∑ i, (↑(sigma i) + 1) ≤ 6 then ∏ i, a z (sigma i)
                                                              else 0;
                                                    have window := {z | √(z.1 ^ 2 + z.2 ^ 2) ≤ ↑n ^ 3 * Real.log ↑n};
                                                    have weight := fun z => √(z.1 ^ 2 + z.2 ^ 2) ^ k;
                                                    have error := fun z =>
                                                      ∫ (w : ℝ × ℝ),
                                                          Complex.exp (↑(z.1 * w.1 + z.2 * w.2) * Complex.I) ∂rho -
                                                        Pn z;
                                                    have damping := fun z => Real.exp (-b * (z.1 ^ 2 + z.2 ^ 2));
                                                    have kernel := fun x z =>
                                                      Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2));
                                                    MeasureTheory.IsProbabilityMeasure rho ∧
                                                      Continuous Pn ∧
                                                        Continuous error ∧
                                                          MeasureTheory.IntegrableOn (fun z => weight z * ‖error z‖)
                                                              window MeasureTheory.volume ∧
                                                            0 ≤ ∫ (z : ℝ × ℝ) in window, weight z * ‖error z‖ ∧
                                                              ∫ (z : ℝ × ℝ) in window, weight z * ‖error z‖ ≤
                                                                  C / (↑n ^ 3 * √↑n) ∧
                                                                MeasureTheory.IntegrableOn
                                                                    (fun z => weight z * ‖↑(damping z) * error z‖)
                                                                    window MeasureTheory.volume ∧
                                                                  ∫ (z : ℝ × ℝ) in window,
                                                                        weight z * ‖↑(damping z) * error z‖ ≤
                                                                      C / (↑n ^ 3 * √↑n) ∧
                                                                    ∀ (x : ℝ × ℝ),
                                                                      MeasureTheory.IntegrableOn
                                                                          (fun z =>
                                                                            kernel x z * ↑(weight z) *
                                                                              (↑(damping z) * error z))
                                                                          window MeasureTheory.volume ∧
                                                                        ‖∫ (z : ℝ × ℝ) in window,
                                                                              kernel x z * ↑(weight z) *
                                                                                (↑(damping z) * error z)‖ ≤
                                                                          C / (↑n ^ 3 * √↑n)) ∧
                                            ∀ (epsilon : ℝ),
                                              0 < epsilon →
                                                ∃ Ne,
                                                  N ≤ Ne ∧
                                                    ∀ (n : ℕ),
                                                      Ne ≤ n →
                                                        ∀ (p : ↑Theta) (b : ℝ),
                                                          0 ≤ b →
                                                            have Qsample := MeasureTheory.Measure.pi fun x => tilt ↑p;
                                                            have M := fun z => ((∑ i, z i) / ↑n, Shao.energy z / ↑n);
                                                            have T := fun z =>
                                                              √↑n • (E p) (M z - (means (↑p) 0, means (↑p) 1));
                                                            have rho := MeasureTheory.Measure.map T Qsample;
                                                            have X := fun z y => z.1 * y 0 + z.2 * y 1;
                                                            have kappa := fun j z =>
                                                              iteratedDeriv j (ProbabilityTheory.cgf (X z) (laws ↑p)) 0;
                                                            have a := fun z j =>
                                                              Complex.I ^ (↑j + 3) / ↑(↑j + 3).factorial *
                                                                  ↑(kappa (↑j + 3) z) /
                                                                ↑√↑n ^ (↑j + 1);
                                                            have Pn := fun z =>
                                                              Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) *
                                                                ∑ q,
                                                                  1 / ↑(↑q).factorial *
                                                                    ∑ sigma,
                                                                      if ∑ i, (↑(sigma i) + 1) ≤ 6 then
                                                                        ∏ i, a z (sigma i)
                                                                      else 0;
                                                            have window :=
                                                              {z | √(z.1 ^ 2 + z.2 ^ 2) ≤ ↑n ^ 3 * Real.log ↑n};
                                                            have weight := fun z => √(z.1 ^ 2 + z.2 ^ 2) ^ k;
                                                            have error := fun z =>
                                                              ∫ (w : ℝ × ℝ),
                                                                  Complex.exp
                                                                    (↑(z.1 * w.1 + z.2 * w.2) * Complex.I) ∂rho -
                                                                Pn z;
                                                            have damping := fun z =>
                                                              Real.exp (-b * (z.1 ^ 2 + z.2 ^ 2));
                                                            have kernel := fun x z =>
                                                              Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2));
                                                            ↑n ^ 3 *
                                                                  ∫ (z : ℝ × ℝ) in window,
                                                                    weight z * ‖↑(damping z) * error z‖ <
                                                                epsilon ∧
                                                              ∀ (x : ℝ × ℝ),
                                                                ↑n ^ 3 *
                                                                    ‖∫ (z : ℝ × ℝ) in window,
                                                                        kernel x z * ↑(weight z) *
                                                                          (↑(damping z) * error z)‖ <
                                                                  epsilon
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
