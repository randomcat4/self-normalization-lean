DECL Shao.OQ3.pair_gaussian_oscillatory_integral
TYPE ∀ (b : ℝ),
  0 < b →
    ∀ (x : ℝ × ℝ),
      MeasureTheory.Integrable
          (fun z => Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * Complex.exp (-↑b * ↑(z.1 ^ 2 + z.2 ^ 2)))
          MeasureTheory.volume ∧
        ∫ (z : ℝ × ℝ), Complex.exp (-Complex.I * ↑(x.1 * z.1 + x.2 * z.2)) * Complex.exp (-↑b * ↑(z.1 ^ 2 + z.2 ^ 2)) =
          ↑(Real.pi / b) * Complex.exp (-↑((x.1 ^ 2 + x.2 ^ 2) / (4 * b)))
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem

DECL Shao.OQ3.real_gaussian_oscillatory_integral
TYPE ∀ (b : ℝ),
  0 < b →
    ∀ (x : ℝ),
      MeasureTheory.Integrable (fun t => Complex.exp (-Complex.I * ↑x * ↑t) * Complex.exp (-↑b * ↑t ^ 2))
          MeasureTheory.volume ∧
        ∫ (t : ℝ), Complex.exp (-Complex.I * ↑x * ↑t) * Complex.exp (-↑b * ↑t ^ 2) =
          ↑√(Real.pi / b) * Complex.exp (-↑(x ^ 2 / (4 * b)))
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
