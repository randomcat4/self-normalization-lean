DECL Shao.OQ3.uniform_actual_cgf_six_order_gaussian_polynomial_bound
TYPE ∀ {ι : Type u_1} (law : ι → MeasureTheory.Measure (Fin 2 → ℝ)) (B : Fin 6 → ℝ),
  (∀ (j : Fin 6), 0 ≤ B j) →
    (∀ (j : Fin 6) (i : ι) (u v : ℝ),
        |iteratedDeriv (↑j + 3) (ProbabilityTheory.cgf (fun y => u * y 0 + v * y 1) (law i)) 0| ≤
          B j * √(u ^ 2 + v ^ 2) ^ (↑j + 3)) →
      have X := fun i z y => z.1 * y 0 + z.2 * y 1;
      have kappa := fun i j z => iteratedDeriv j (ProbabilityTheory.cgf (X i z) (law i)) 0;
      have a := fun i n z j => Complex.I ^ (↑j + 3) / ↑(↑j + 3).factorial * ↑(kappa i (↑j + 3) z) / ↑√↑n ^ (↑j + 1);
      have Pn := fun i n z =>
        Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) *
          ∑ q, 1 / ↑(↑q).factorial * ∑ sigma, if ∑ l, (↑(sigma l) + 1) ≤ 6 then ∏ l, a i n z (sigma l) else 0;
      ∃ C,
        0 < C ∧
          ∀ (n : ℕ),
            1 ≤ n →
              ∀ (i : ι) (u v : ℝ), ‖Pn i n (u, v)‖ ≤ C * (1 + √(u ^ 2 + v ^ 2) ^ 18) * Real.exp (-(u ^ 2 + v ^ 2) / 2)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.uniform_actual_cgf_six_order_gaussian_polynomial_tail
TYPE ∀ {ι : Type u_1} (law : ι → MeasureTheory.Measure (Fin 2 → ℝ)),
  (∀ (i : ι), MeasureTheory.IsProbabilityMeasure (law i)) →
    (∀ (i : ι) (u v : ℝ), 0 ∈ interior (ProbabilityTheory.integrableExpSet (fun y => u * y 0 + v * y 1) (law i))) →
      (∀ (i : ι) (k : ℕ), MeasureTheory.Integrable (fun y => √(y 0 ^ 2 + y 1 ^ 2) ^ k) (law i)) →
        ∀ (B : Fin 6 → ℝ),
          (∀ (j : Fin 6), 0 ≤ B j) →
            (∀ (j : Fin 6) (i : ι) (u v : ℝ),
                |iteratedDeriv (↑j + 3) (ProbabilityTheory.cgf (fun y => u * y 0 + v * y 1) (law i)) 0| ≤
                  B j * √(u ^ 2 + v ^ 2) ^ (↑j + 3)) →
              have X := fun i z y => z.1 * y 0 + z.2 * y 1;
              have kappa := fun i j z => iteratedDeriv j (ProbabilityTheory.cgf (X i z) (law i)) 0;
              have a := fun i n z j =>
                Complex.I ^ (↑j + 3) / ↑(↑j + 3).factorial * ↑(kappa i (↑j + 3) z) / ↑√↑n ^ (↑j + 1);
              have Pn := fun i n z =>
                Complex.exp (-↑(z.1 ^ 2 + z.2 ^ 2) / 2) *
                  ∑ q, 1 / ↑(↑q).factorial * ∑ sigma, if ∑ l, (↑(sigma l) + 1) ≤ 6 then ∏ l, a i n z (sigma l) else 0;
              ∀ (delta : ℝ),
                0 < delta →
                  ∀ (k : ℕ) (q epsilon : ℝ),
                    0 < epsilon →
                      ∃ N,
                        1 ≤ N ∧
                          ∀ (n : ℕ),
                            N ≤ n →
                              ∀ (i : ι),
                                have tail := {z | delta * √↑n ≤ √(z.1 ^ 2 + z.2 ^ 2)};
                                have f := fun z => √(z.1 ^ 2 + z.2 ^ 2) ^ k * ‖Pn i n z‖;
                                MeasureTheory.IntegrableOn f tail MeasureTheory.volume ∧
                                  0 ≤ ∫ (z : ℝ × ℝ) in tail, f z ∧ (↑n).rpow q * ∫ (z : ℝ × ℝ) in tail, f z < epsilon
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
