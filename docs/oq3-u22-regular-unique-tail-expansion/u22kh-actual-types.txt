DECL Shao.OQ3.joint_energy_positive_curvature_neighborhood
TYPE ∀ (F : ℝ × ℝ → ℝ) (q0 : ℝ),
  ContDiffAt ℝ 3 F (0, q0) →
    0 < deriv (deriv fun q => F (0, q)) q0 →
      ∃ eta rho,
        0 < eta ∧
          0 < rho ∧
            ∀ (level q : ℝ),
              |level| ≤ eta →
                |q - q0| ≤ rho →
                  ContDiffAt ℝ 2 (fun energy => F (level, energy)) q ∧
                    0 < deriv (deriv fun energy => F (level, energy)) q
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
