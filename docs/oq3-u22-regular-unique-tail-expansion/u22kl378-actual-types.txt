DECL Shao.OQ3.scaled_tangent_positive_curvature_neighborhood
TYPE ∀ (J : ℝ × ℝ → ℝ) (P : ℝ × ℝ → ℝ × ℝ) (q0 : ℝ),
  0 < q0 →
    ContDiffAt ℝ 3 J (0, q0) →
      ContinuousAt P (0, q0) →
        P (0, q0) = (0, 0) →
          0 < ((fderiv ℝ (fun z => fderiv ℝ J z) (0, q0)) (0, 1)) (0, 1) →
            ∃ eta rho,
              0 < eta ∧
                0 < rho ∧
                  rho < q0 ∧
                    ∀ (level : ℝ) (z : ℝ × ℝ),
                      |level| ≤ eta →
                        dist z (0, q0) ≤ rho →
                          0 < 2 * (P z).2 + ((fderiv ℝ (fun w => fderiv ℝ J w) z) (level, 2 * √z.2)) (level, 2 * √z.2)
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
