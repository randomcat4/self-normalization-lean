DECL Shao.OQ3.parabola_energy_positive_curvature_neighborhood
TYPE ∀ (J : ℝ × ℝ → ℝ) (q0 : ℝ),
  0 < q0 →
    ContDiffAt ℝ 3 J (0, q0) →
      0 < ((fderiv ℝ (fun z => fderiv ℝ J z) (0, q0)) (0, 1)) (0, 1) →
        ∃ eta rho,
          0 < eta ∧
            0 < rho ∧
              rho < q0 ∧
                ∀ (level q : ℝ),
                  |level| ≤ eta →
                    |q - q0| ≤ rho →
                      ContDiffAt ℝ 2 (fun energy => J (level * √energy, energy)) q ∧
                        0 < deriv (deriv fun energy => J (level * √energy, energy)) q
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
