DECL Shao.OQ3.exists_symmetric_eight_atom_regular_unique_saddle_witness
TYPE ∃ a,
  a 0 = 1 ∧
    StrictMono a ∧
      (∀ (i : Fin 8), 0 < a i) ∧
        let A := Fin.append a fun i => -a i;
        let mu := MeasureTheory.Measure.sum fun k => ENNReal.ofReal (1 / 16) • MeasureTheory.Measure.dirac (A k);
        Function.Injective A ∧
          MeasureTheory.IsProbabilityMeasure mu ∧
            MeasureTheory.Integrable (fun x => x) mu ∧
              ∫ (x : ℝ), x ∂mu = 0 ∧
                (∃ B, 0 < B ∧ (∀ᵐ (x : ℝ) ∂mu, 1 ≤ |x| ∧ |x| ≤ B) ∧ ∀ᵐ (x : ℝ) ∂mu, 1 ≤ x ^ 2 ∧ x ^ 2 ≤ B ^ 2) ∧
                  0 < mu (Set.Ioi 0) ∧
                    (∀ (xi eta : ℝ), xi ≠ 0 ∨ eta ≠ 0 → ‖Shao.OQ3.jointCharacteristic mu xi eta‖ < 1) ∧
                      (∃ c,
                          0 < c ∧
                            ∃ R,
                              0 ≤ R ∧
                                (∀ (xi eta : ℝ),
                                    R < √(xi ^ 2 + eta ^ 2) →
                                      ‖Shao.OQ3.jointCharacteristic mu xi eta‖ ≤
                                        1 - c * (√(xi ^ 2 + eta ^ 2)).rpow (-3 / 8)) ∧
                                  ∀ (K : Set (ℝ × ℝ)),
                                    IsCompact K →
                                      ∃ m,
                                        0 < m ∧
                                          m ≤ 1 ∧
                                            0 < m ^ 2 * c / 2 ∧

                                                  (∀ (xi eta : ℝ),
                                                      m ^ 2 * (1 - ‖Shao.OQ3.jointCharacteristic mu xi eta‖ ^ 2) ≤
                                                        1 -
                                                          ‖Shao.OQ3.jointCharacteristic
                                                                (mu.tilted fun x => p.1 * x + p.2 * x ^ 2) xi eta‖ ^
                                                            2) ∧
                                                    ∀ (xi eta : ℝ),
                                                      R < √(xi ^ 2 + eta ^ 2) →
                                                        ‖Shao.OQ3.jointCharacteristic
                                                              (mu.tilted fun x => p.1 * x + p.2 * x ^ 2) xi eta‖ ≤
                                                          1 - m ^ 2 * c / 2 * (√(xi ^ 2 + eta ^ 2)).rpow (-3 / 8)) ∧
                        (∀ (R eps : ℝ),
                            0 < eps →
                              ∃ xi,
                                R < xi ∧
                                  ‖Shao.OQ3.jointCharacteristic mu xi 0 - 1‖ < eps ∧
                                    1 - eps < ‖Shao.OQ3.jointCharacteristic mu xi 0‖) ∧
                          (¬∃ delta,
                                0 < delta ∧
                                  ∃ R,
                                    0 ≤ R ∧
                                      ∀ (xi eta : ℝ),
                                        R < √(xi ^ 2 + eta ^ 2) →
                                          ‖Shao.OQ3.jointCharacteristic mu xi eta‖ ≤ 1 - delta) ∧
                            0 < (Shao.OQ3.parabolaCovariance mu).det ∧
                              ∃ epsilon,
                                0 < epsilon ∧
                                  epsilon ≤ 1 ∧
                                    ∀ (level : ℝ),
                                      0 < level →
                                        level < epsilon →
                                          ∃ a0 theta L,
                                            have D :=
                                              Shao.OQ3.parabolaCovariance
                                                (mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2);
                                            0 < a0 ∧
                                              theta ∈
                                                  interior
                                                    (Shao.Probability.bivariateMGFDomain mu fun x => (x, x ^ 2)) ∧
                                                0 < D.det ∧
                                                  (∫ (x : ℝ), x ∂mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2,
                                                        ∫ (x : ℝ),
                                                          x ^ 2 ∂mu.tilted fun x => theta.1 * x + theta.2 * x ^ 2) =
                                                      (a0, a0 ^ 2 / level ^ 2) ∧
                                                    theta.1 = -2 * a0 * theta.2 / level ^ 2 ∧
                                                      0 < theta.1 ∧
                                                        theta.2 < 0 ∧
                                                          ↑L =
                                                              (D 0 0 • ContinuousLinearMap.fst ℝ ℝ ℝ +
                                                                    D 0 1 • ContinuousLinearMap.snd ℝ ℝ ℝ).prod
                                                                (D 0 1 • ContinuousLinearMap.fst ℝ ℝ ℝ +
                                                                  D 1 1 • ContinuousLinearMap.snd ℝ ℝ ℝ) ∧
                                                            0 <
                                                                2 * theta.2 / level ^ 2 +
                                                                    (L.symm (1, 2 * a0 / level ^ 2)).1 +
                                                                  2 * a0 / level ^ 2 *
                                                                    (L.symm (1, 2 * a0 / level ^ 2)).2 ∧
                                                              0 <
                                                                  Shao.Probability.bivariateLegendreRate mu

                                                                      (fun x => (x, x ^ 2)) (a0, a0 ^ 2 / level ^ 2) <
                                                                    ⊤ ∧

                                                                                Shao.Probability.bivariateLegendreRate

                                                                                          mu (fun x => (x, x ^ 2)) y
