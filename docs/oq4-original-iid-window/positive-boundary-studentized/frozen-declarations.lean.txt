  fun t ω => (Real.sqrt (c : ℝ))⁻¹ * B (c * t) ω

  Measure.infinitePi (fun _ : ℕ => ν)

private def finiteProjection (n : ℕ) (ω : ℕ → ℝ) : Shao.Path n :=

  (∑ i ∈ Finset.range n, (ω i) ^ 2) / (n : ℝ)

def energyBad (n : ℕ) (δ : ℝ) : Set (Shao.Path n) :=
  {a | δ * (n : ℝ) < |Shao.energy a - (n : ℝ)|}
