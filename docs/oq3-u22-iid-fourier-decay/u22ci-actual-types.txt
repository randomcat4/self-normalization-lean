DECL Shao.OQ3.weak_cramer_pair_compact_frequency_gap
TYPE ∀ (law : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure law] (beta c R : ℝ),
  0 ≤ beta →
    0 < c →
      0 ≤ R →
        (∀ (u v : ℝ),
            R < √(u ^ 2 + v ^ 2) →
              ‖∫ (w : ℝ × ℝ), Complex.exp (↑(u * w.1 + v * w.2) * Complex.I) ∂law‖ ≤
                1 - c * (√(u ^ 2 + v ^ 2)).rpow (-beta)) →
          ∀ (K : Set (ℝ × ℝ)),
            IsCompact K →
              (∀ a ∈ K, a.1 ≠ 0 ∨ a.2 ≠ 0) →
                ∃ gamma,
                  0 < gamma ∧
                    gamma < 1 ∧
                      ∀ a ∈ K, ‖∫ (w : ℝ × ℝ), Complex.exp (↑(a.1 * w.1 + a.2 * w.2) * Complex.I) ∂law‖ ≤ gamma
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.continuous_pair_characteristic
TYPE ∀ (law : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure law],
  Continuous fun a => ∫ (w : ℝ × ℝ), Complex.exp (↑(a.1 * w.1 + a.2 * w.2) * Complex.I) ∂law
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
