DECL Shao.OQ3.pair_characteristic_modulus_one_ae_constant
TYPE ∀ (law : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure law] (u v : ℝ),
  ‖∫ (w : ℝ × ℝ), Complex.exp (↑(u * w.1 + v * w.2) * Complex.I) ∂law‖ = 1 →
    MeasureTheory.Integrable (fun w => Complex.exp (↑(u * w.1 + v * w.2) * Complex.I)) law ∧
      ∀ᵐ (w : ℝ × ℝ) ∂law,
        Complex.exp (↑(u * w.1 + v * w.2) * Complex.I) =
          ∫ (z : ℝ × ℝ), Complex.exp (↑(u * z.1 + v * z.2) * Complex.I) ∂law
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
DECL Shao.OQ3.weak_cramer_pair_characteristic_norm_lt_one
TYPE ∀ (law : MeasureTheory.Measure (ℝ × ℝ)) [MeasureTheory.IsProbabilityMeasure law] (beta c R : ℝ),
  0 ≤ beta →
    0 < c →
      0 ≤ R →
        (∀ (u v : ℝ),
            R < √(u ^ 2 + v ^ 2) →
              ‖∫ (w : ℝ × ℝ), Complex.exp (↑(u * w.1 + v * w.2) * Complex.I) ∂law‖ ≤
                1 - c * (√(u ^ 2 + v ^ 2)).rpow (-beta)) →
          ∀ (u v : ℝ), u ≠ 0 ∨ v ≠ 0 → ‖∫ (w : ℝ × ℝ), Complex.exp (↑(u * w.1 + v * w.2) * Complex.I) ∂law‖ < 1
AXIOMS [propext, Classical.choice, Quot.sound]
KIND theorem
