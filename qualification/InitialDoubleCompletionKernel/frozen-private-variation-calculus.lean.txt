private theorem variation_add_le (f g : ℝ → ℝ) (S : Set ℝ) :
    eVariationOn (fun w => f w + g w) S ≤ eVariationOn f S + eVariationOn g S

private theorem variation_neg_eq (f : ℝ → ℝ) (S : Set ℝ) :
    eVariationOn (fun w => -f w) S = eVariationOn f S

private theorem variation_le_of_pointwise_limit (F : ℕ → ℝ → ℝ) (f : ℝ → ℝ)
    (S : Set ℝ) (C : ENNReal)
    (hconv : ∀ w ∈ S, Tendsto (fun N : ℕ => F N w) atTop (nhds (f w)))
    (hbound : ∀ N : ℕ, eVariationOn (F N) S ≤ C) :
    eVariationOn f S ≤ C
