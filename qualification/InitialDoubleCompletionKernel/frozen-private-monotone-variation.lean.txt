private theorem antitoneOn_variation_le (f : ℝ → ℝ) (S : Set ℝ) (C : ℝ)
    (hf : AntitoneOn f S) (hf0 : ∀ w ∈ S, 0 ≤ f w) (hfC : ∀ w ∈ S, f w ≤ C) :
    eVariationOn f S ≤ ENNReal.ofReal C

    (hf : MonotoneOn f S) (hf0 : ∀ w ∈ S, 0 ≤ f w) (hfC : ∀ w ∈ S, f w ≤ C) :
    eVariationOn f S ≤ ENNReal.ofReal C
