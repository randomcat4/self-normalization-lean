private theorem initial_image_geometric_majorants (x v k : ℝ)
    (hx : 0 ≤ x) (hk : 1 ≤ k) (hv : |v| ≤ x) :
    Real.exp (-(8 * k ^ 2 * x ^ 2 + 4 * k * x * v)) ≤ Real.exp (-4 * k * x ^ 2) ∧
      Real.exp (-(8 * k ^ 2 * x ^ 2 - 4 * k * x * v)) ≤ Real.exp (-4 * k * x ^ 2) ∧
      Real.exp (-(2 * (2 * k + 1) ^ 2 * x ^ 2 + 2 * (2 * k + 1) * x * v)) ≤
        Real.exp (-4 * k * x ^ 2) ∧
      Real.exp (-(2 * (2 * k + 1) ^ 2 * x ^ 2 - 2 * (2 * k + 1) * x * v)) ≤
        Real.exp (-4 * k * x ^ 2)

private theorem initial_geometric_ratio_bound (x : ℝ) (hx : 1 ≤ x) :
    0 < Real.exp (-4 * x ^ 2) ∧ Real.exp (-4 * x ^ 2) ≤ 1 / 2
