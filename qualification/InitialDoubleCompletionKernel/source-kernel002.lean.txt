def doubleCompletion (nu : Measure BridgeNoise) (x w : ℝ) : ℝ :=
  nu.real ((strictUpper x 0 1 w)ᶜ ∩ (strictLower x 0 1 w)ᶜ)

private theorem doubleCompletion_eq (nu : Measure BridgeNoise) [IsProbabilityMeasure nu]
    (x w : ℝ) :
    doubleCompletion nu x w =
      1 - upperCompletion nu x 0 1 w - lowerCompletion nu x 0 1 w +
        intervalCompletion nu x 0 1 w := by
  classical
  let A : Set BridgeNoise := strictUpper x 0 1 w
  let B : Set BridgeNoise := strictLower x 0 1 w
  have hA : MeasurableSet A := measurableSet_strictUpper x 0 1 w
  have hB : MeasurableSet B := measurableSet_strictLower x 0 1 w
  have h0 := measureReal_inter_add_sdiff (μ := nu) (s := Set.univ) hA
  have h0' : nu.real A + nu.real Aᶜ = 1 := by
    simpa only [Set.univ_inter, ← Set.compl_eq_univ_sdiff, probReal_univ] using h0
  have h1 := measureReal_inter_add_sdiff (μ := nu) (s := Aᶜ) hB
  have h2 := measureReal_inter_add_sdiff (μ := nu) (s := B) hA
  have hBA : B \ A = Aᶜ ∩ B := by
    ext g
    simp only [Set.mem_sdiff, Set.mem_inter_iff, Set.mem_compl_iff]
    exact and_comm
  rw [show B ∩ A = A ∩ B from Set.inter_comm B A, hBA] at h2
  change nu.real (Aᶜ ∩ B) + nu.real (Aᶜ ∩ Bᶜ) = nu.real Aᶜ at h1
  change nu.real (Aᶜ ∩ Bᶜ) = 1 - nu.real A - nu.real B + nu.real (A ∩ B)
  linarith

private theorem doubleCompletion_bounds (nu : Measure BridgeNoise) [IsProbabilityMeasure nu]
    (x w : ℝ) :
    0 ≤ doubleCompletion nu x w ∧ doubleCompletion nu x w ≤ 1 := by
  refine ⟨measureReal_nonneg, ?_⟩
  unfold doubleCompletion
  calc
    _ ≤ nu.real Set.univ := measureReal_mono (Set.subset_univ _) (by finiteness)
    _ = 1 := probReal_univ

private theorem measurable_doubleCompletion (nu : Measure BridgeNoise) [IsProbabilityMeasure nu]
    (x : ℝ) : Measurable (doubleCompletion nu x) := by
  let p : ℝ → Shao.C2.CompletionMeasurability.CompletionParameters :=
    fun w => (x, 0, 1, w)
  have hp : Measurable p := by
    dsimp only [p]
    fun_prop
  have hU : Measurable (fun w : ℝ => upperCompletion nu x 0 1 w) := by
    simpa only [Function.comp_def, p] using
      (Shao.C2.CompletionMeasurability.measurable_upperCompletion_parameters nu).comp hp
  have hL : Measurable (fun w : ℝ => lowerCompletion nu x 0 1 w) := by
    simpa only [Function.comp_def, p] using
      (Shao.C2.CompletionMeasurability.measurable_lowerCompletion_parameters nu).comp hp
  have hI : Measurable (fun w : ℝ => intervalCompletion nu x 0 1 w) := by
    simpa only [Function.comp_def, p] using
      (Shao.C2.CompletionMeasurability.measurable_intervalCompletion_parameters nu).comp hp
  have heq : doubleCompletion nu x = fun w : ℝ =>
      1 - upperCompletion nu x 0 1 w - lowerCompletion nu x 0 1 w +
        intervalCompletion nu x 0 1 w := by
    funext w
    exact doubleCompletion_eq nu x w
  rw [heq]
  exact ((measurable_const.sub hU).sub hL).add hI

private theorem doubleCompletion_exterior
    {Ω : Type*} [MeasurableSpace Ω]
    (B : ℝ≥0 → Ω → ℝ) (P : Measure Ω) [IsProbabilityMeasure P]
    (hB : IsBrownianReal B P) (x : ℝ) (hx : 0 < x)
    (w : ℝ) (hw : x ≤ |w|) :
    doubleCompletion (bridgeLaw B P) x w = Real.exp (-2 * x * (x + |w|)) := by
  classical
  letI : IsProbabilityMeasure (bridgeLaw B P) := isProbabilityMeasure_bridgeLaw B P hB
  by_cases hw0 : 0 ≤ w
  · have hxw : x ≤ w := by simpa only [abs_of_nonneg hw0] using hw
    have hdead : strictUpper x 0 1 w = ∅ := by
      apply Set.eq_empty_iff_forall_notMem.mpr
      intro g hg
      have ht := hg (1 : BridgeTime)
      rw [(bridgePath_endpoints 0 1 w g).2] at ht
      linarith
    have hU : upperCompletion (bridgeLaw B P) x 0 1 w = 0 := by
      simp only [upperCompletion, hdead, measureReal_empty]
    have hI : intervalCompletion (bridgeLaw B P) x 0 1 w = 0 := by
      simp [intervalCompletion, strictInterval, hdead]
    have hL := Shao.C2.BridgeLowerIdentification.bridge_lower_formula B P hB x 0 1 w
      (by norm_num) (by linarith) (by linarith)
    have hj : doubleCompletion (bridgeLaw B P) x w = Real.exp (-2 * x * (x + w)) := by
      rw [doubleCompletion_eq, hU, hI, hL]
      simp only [add_zero, div_one]
      ring
    simpa only [abs_of_nonneg hw0] using hj
  · have hwneg : w < 0 := lt_of_not_ge hw0
    have hwx : w ≤ -x := by
      have hh : x ≤ -w := by simpa only [abs_of_neg hwneg] using hw
      linarith
    have hdead : strictLower x 0 1 w = ∅ := by
      apply Set.eq_empty_iff_forall_notMem.mpr
      intro g hg
      have ht := hg (1 : BridgeTime)
      rw [(bridgePath_endpoints 0 1 w g).2] at ht
      linarith
    have hL : lowerCompletion (bridgeLaw B P) x 0 1 w = 0 := by
      simp only [lowerCompletion, hdead, measureReal_empty]
    have hI : intervalCompletion (bridgeLaw B P) x 0 1 w = 0 := by
      simp [intervalCompletion, strictInterval, hdead]
    have hU := Shao.C2.BridgeUpperAffineIdentification.bridge_upper_formula B P hB x 0 1 w
      (by norm_num) hx (by linarith)
    have hj : doubleCompletion (bridgeLaw B P) x w = Real.exp (-2 * x * (x - w)) := by
      rw [doubleCompletion_eq, hL, hI, hU]
      simp only [sub_zero, div_one]
      ring
    simpa only [abs_of_neg hwneg, sub_eq_add_neg] using hj


private theorem doubleCompletion_interior_images
    {Ω : Type*} [MeasurableSpace Ω]
    (B : ℝ≥0 → Ω → ℝ) (P : Measure Ω) [IsProbabilityMeasure P]
    (hB : IsBrownianReal B P) (x : ℝ) (hx : 0 < x)
    (w : ℝ) (hw : |w| < x) :
    doubleCompletion (bridgeLaw B P) x w =
      Real.exp (-Shao.C2.IntervalImageExponents.imageA x 0 w 1) +
        Real.exp (-Shao.C2.IntervalImageExponents.imageA x 0 w (-1)) +
          ∑' n : ℕ, Shao.C2.IntervalImageSpatialConvergence.spatialTailBlock 0 x w 1 0 n := by
  classical
  letI : IsProbabilityMeasure (bridgeLaw B P) := isProbabilityMeasure_bridgeLaw B P hB
  have hU := Shao.C2.BridgeUpperAffineIdentification.bridge_upper_formula B P hB x 0 1 w
    (by norm_num) hx (abs_lt.mp hw).2
  have hL := Shao.C2.BridgeLowerIdentification.bridge_lower_formula B P hB x 0 1 w
    (by norm_num) (by linarith) (abs_lt.mp hw).1
  have hI := Shao.C2.BrownianKilledSineIdentification.bridge_interval_eq_image B P hB x 0 1 w hx
    (by simpa only [abs_zero] using hx) (by norm_num) hw
  have hB0 : Shao.C2.IntervalImageExponents.imageB x 0 w 0 = 2 * x * (x + w) := by
    unfold Shao.C2.IntervalImageExponents.imageB
    ring
  have hBm : Shao.C2.IntervalImageExponents.imageB x 0 w (-1) = 2 * x * (x - w) := by
    unfold Shao.C2.IntervalImageExponents.imageB
    ring
  have he0 : -(2 * x * (x + w)) = -2 * x * (x + w) := by ring
  have hem : -(2 * x * (x - w)) = -2 * x * (x - w) := by ring
  rw [doubleCompletion_eq, hU, hL, hI,
    Shao.C2.IntervalImageCandidateSpatial.intervalImageCandidate_eq_images]
  simp only [sub_zero, add_zero, div_one]
  rw [hB0, hBm, he0, hem]
  ring


private theorem initial_image_tail_eq (x w : ℝ) (n : ℕ) :
    Shao.C2.IntervalImageSpatialConvergence.spatialTailBlock 0 x w 1 0 n =
      Real.exp (-(8 * ((n : ℝ) + 2) ^ 2 * x ^ 2 + 4 * ((n : ℝ) + 2) * x * w)) +
        Real.exp (-(8 * ((n : ℝ) + 2) ^ 2 * x ^ 2 - 4 * ((n : ℝ) + 2) * x * w)) -
        Real.exp (-(2 * (2 * (n : ℝ) + 3) ^ 2 * x ^ 2 + 2 * (2 * (n : ℝ) + 3) * x * w)) -
        Real.exp (-(2 * (2 * (n : ℝ) + 3) ^ 2 * x ^ 2 - 2 * (2 * (n : ℝ) + 3) * x * w)) := by
  have ha : Shao.C2.IntervalImageExponents.imageA x 0 w ((n : ℤ) + 2 : ℤ) =
      8 * ((n : ℝ) + 2) ^ 2 * x ^ 2 + 4 * ((n : ℝ) + 2) * x * w := by
    unfold Shao.C2.IntervalImageExponents.imageA
    push_cast
    ring
  have ham : Shao.C2.IntervalImageExponents.imageA x 0 w (-((n : ℤ) + 2) : ℤ) =
      8 * ((n : ℝ) + 2) ^ 2 * x ^ 2 - 4 * ((n : ℝ) + 2) * x * w := by
    unfold Shao.C2.IntervalImageExponents.imageA
    push_cast
    ring
  have hb : Shao.C2.IntervalImageExponents.imageB x 0 w ((n : ℤ) + 1 : ℤ) =
      2 * (2 * (n : ℝ) + 3) ^ 2 * x ^ 2 + 2 * (2 * (n : ℝ) + 3) * x * w := by
    unfold Shao.C2.IntervalImageExponents.imageB
    push_cast
    ring
  have hbm : Shao.C2.IntervalImageExponents.imageB x 0 w (-((n : ℤ) + 2) : ℤ) =
      2 * (2 * (n : ℝ) + 3) ^ 2 * x ^ 2 - 2 * (2 * (n : ℝ) + 3) * x * w := by
    unfold Shao.C2.IntervalImageExponents.imageB
    push_cast
    ring
  simp only [Shao.C2.IntervalImageSpatialConvergence.spatialTailBlock,
    iteratedDeriv_zero, div_one]
  rw [ha, ham, hb, hbm]

private theorem antitoneOn_variation_le (f : ℝ → ℝ) (S : Set ℝ) (C : ℝ)
    (hf : AntitoneOn f S) (hf0 : ∀ w ∈ S, 0 ≤ f w) (hfC : ∀ w ∈ S, f w ≤ C) :
    eVariationOn f S ≤ ENNReal.ofReal C := by
  apply iSup_le
  rintro ⟨n, w, hw, hmem⟩
  have ht (m : ℕ) :
      (∑ i ∈ Finset.range m, (f (w i) - f (w (i + 1)))) = f (w 0) - f (w m) := by
    induction m with
    | zero => simp
    | succ m ih => rw [Finset.sum_range_succ, ih]; ring
  calc
    _ = ∑ i ∈ Finset.range n, ENNReal.ofReal (f (w i) - f (w (i + 1))) := by
      apply Finset.sum_congr rfl
      intro i hi
      have hh := hf (hmem i) (hmem (i + 1)) (hw (Nat.le_succ i))
      rw [edist_dist, Real.dist_eq, abs_of_nonpos (sub_nonpos.mpr hh)]
      congr 1
      ring
    _ = ENNReal.ofReal (∑ i ∈ Finset.range n, (f (w i) - f (w (i + 1)))) := by
      rw [ENNReal.ofReal_sum_of_nonneg]
      intro i hi
      exact sub_nonneg.mpr (hf (hmem i) (hmem (i + 1)) (hw (Nat.le_succ i)))
    _ = ENNReal.ofReal (f (w 0) - f (w n)) := by rw [ht]
    _ ≤ ENNReal.ofReal C := by
      apply ENNReal.ofReal_le_ofReal
      linarith [hfC (w 0) (hmem 0), hf0 (w n) (hmem n)]


private theorem monotoneOn_variation_le (f : ℝ → ℝ) (S : Set ℝ) (C : ℝ)
    (hf : MonotoneOn f S) (hf0 : ∀ w ∈ S, 0 ≤ f w) (hfC : ∀ w ∈ S, f w ≤ C) :
    eVariationOn f S ≤ ENNReal.ofReal C := by
  apply iSup_le
  rintro ⟨n, w, hw, hmem⟩
  have ht (m : ℕ) :
      (∑ i ∈ Finset.range m, (f (w (i + 1)) - f (w i))) = f (w m) - f (w 0) := by
    induction m with
    | zero => simp
    | succ m ih => rw [Finset.sum_range_succ, ih]; ring
  calc
    _ = ∑ i ∈ Finset.range n, ENNReal.ofReal (f (w (i + 1)) - f (w i)) := by
      apply Finset.sum_congr rfl
      intro i hi
      have hh := hf (hmem i) (hmem (i + 1)) (hw (Nat.le_succ i))
      rw [edist_dist, Real.dist_eq, abs_of_nonneg (sub_nonneg.mpr hh)]
    _ = ENNReal.ofReal (∑ i ∈ Finset.range n, (f (w (i + 1)) - f (w i))) := by
      rw [ENNReal.ofReal_sum_of_nonneg]
      intro i hi
      exact sub_nonneg.mpr (hf (hmem i) (hmem (i + 1)) (hw (Nat.le_succ i)))
    _ = ENNReal.ofReal (f (w n) - f (w 0)) := by rw [ht]
    _ ≤ ENNReal.ofReal C := by
      apply ENNReal.ofReal_le_ofReal
      linarith [hfC (w n) (hmem n), hf0 (w 0) (hmem 0)]

private theorem variation_add_le (f g : ℝ → ℝ) (S : Set ℝ) :
    eVariationOn (fun w => f w + g w) S ≤ eVariationOn f S + eVariationOn g S := by
  apply iSup_le
  rintro ⟨n, w, hw, hmem⟩
  calc
    _ ≤ ∑ i ∈ Finset.range n,
        (edist (f (w (i + 1))) (f (w i)) + edist (g (w (i + 1))) (g (w i))) := by
      apply Finset.sum_le_sum
      intro i hi
      simp only [edist_dist, Real.dist_eq]
      rw [← ENNReal.ofReal_add (abs_nonneg _) (abs_nonneg _)]
      apply ENNReal.ofReal_le_ofReal
      have he : (f (w (i + 1)) + g (w (i + 1))) - (f (w i) + g (w i)) =
          (f (w (i + 1)) - f (w i)) + (g (w (i + 1)) - g (w i)) := by ring
      rw [he]
      exact abs_add_le _ _
    _ = (∑ i ∈ Finset.range n, edist (f (w (i + 1))) (f (w i))) +
        ∑ i ∈ Finset.range n, edist (g (w (i + 1))) (g (w i)) := by
      rw [Finset.sum_add_distrib]
    _ ≤ eVariationOn f S + eVariationOn g S :=
      add_le_add (eVariationOn.sum_le (f := f) (n := n) hw hmem)
        (eVariationOn.sum_le (f := g) (n := n) hw hmem)


private theorem variation_neg_eq (f : ℝ → ℝ) (S : Set ℝ) :
    eVariationOn (fun w => -f w) S = eVariationOn f S := by
  unfold eVariationOn
  simp only [edist_neg_neg]


private theorem variation_le_of_pointwise_limit (F : ℕ → ℝ → ℝ) (f : ℝ → ℝ)
    (S : Set ℝ) (C : ENNReal)
    (hconv : ∀ w ∈ S, Tendsto (fun N : ℕ => F N w) atTop (nhds (f w)))
    (hbound : ∀ N : ℕ, eVariationOn (F N) S ≤ C) :
    eVariationOn f S ≤ C := by
  apply iSup_le
  rintro ⟨n, w, hw, hmem⟩
  have hlim : Tendsto
      (fun N : ℕ => ∑ i ∈ Finset.range n, edist (F N (w (i + 1))) (F N (w i)))
      atTop (nhds (∑ i ∈ Finset.range n, edist (f (w (i + 1))) (f (w i)))) := by
    apply tendsto_finsetSum (Finset.range n)
    intro i hi
    exact (hconv (w (i + 1)) (hmem (i + 1))).edist (hconv (w i) (hmem i))
  apply le_of_tendsto hlim
  exact Filter.Eventually.of_forall fun N =>
    (eVariationOn.sum_le (f := F N) (n := n) hw hmem).trans (hbound N)

private theorem initial_image_geometric_majorants (x v k : ℝ)
    (hx : 0 ≤ x) (hk : 1 ≤ k) (hv : |v| ≤ x) :
    Real.exp (-(8 * k ^ 2 * x ^ 2 + 4 * k * x * v)) ≤ Real.exp (-4 * k * x ^ 2) ∧
      Real.exp (-(8 * k ^ 2 * x ^ 2 - 4 * k * x * v)) ≤ Real.exp (-4 * k * x ^ 2) ∧
      Real.exp (-(2 * (2 * k + 1) ^ 2 * x ^ 2 + 2 * (2 * k + 1) * x * v)) ≤
        Real.exp (-4 * k * x ^ 2) ∧
      Real.exp (-(2 * (2 * k + 1) ^ 2 * x ^ 2 - 2 * (2 * k + 1) * x * v)) ≤
        Real.exp (-4 * k * x ^ 2) := by
  have hk0 : 0 ≤ k := by linarith
  have hkm : 0 ≤ k - 1 := by linarith
  have hcA : 0 ≤ 4 * k * x := by positivity
  have hAL := mul_le_mul_of_nonneg_left (abs_le.mp hv).1 hcA
  have hAR := mul_le_mul_of_nonneg_left (abs_le.mp hv).2 hcA
  have hgapA : 0 ≤ 8 * k * (k - 1) * x ^ 2 := by positivity
  have hq : 0 ≤ 2 * k + 1 := by linarith
  have hcB : 0 ≤ 2 * (2 * k + 1) * x := by positivity
  have hBL := mul_le_mul_of_nonneg_left (abs_le.mp hv).1 hcB
  have hBR := mul_le_mul_of_nonneg_left (abs_le.mp hv).2 hcB
  have hgapB : 0 ≤ 8 * (k * x) ^ 2 := by positivity
  refine ⟨?_, ?_, ?_, ?_⟩ <;> apply Real.exp_le_exp.mpr
  all_goals nlinarith only [hAL, hAR, hgapA, hBL, hBR, hgapB]


private theorem initial_geometric_ratio_bound (x : ℝ) (hx : 1 ≤ x) :
    0 < Real.exp (-4 * x ^ 2) ∧ Real.exp (-4 * x ^ 2) ≤ 1 / 2 := by
  have hx0 : 0 ≤ x := by linarith
  have hxx : 1 ≤ x ^ 2 := by nlinarith [sq_nonneg (x - 1)]
  have he : 2 ≤ Real.exp (4 * x ^ 2) := by
    nlinarith [Real.add_one_le_exp (4 * x ^ 2)]
  refine ⟨Real.exp_pos _, ?_⟩
  have hinv : 1 / Real.exp (4 * x ^ 2) ≤ (1 / 2 : ℝ) :=
    one_div_le_one_div_of_le (by norm_num) he
  have hneg : -4 * x ^ 2 = -(4 * x ^ 2) := by ring
  rw [hneg, Real.exp_neg]
  simpa only [one_div] using hinv

private theorem initial_geometric_series_budget (r : ℝ) (hr0 : 0 < r) (hr : r ≤ 1 / 2) :
    Summable (fun n : ℕ => 2 * r ^ (n + 2) + 2 * r ^ (n + 1)) ∧
      (∑' n : ℕ, (2 * r ^ (n + 2) + 2 * r ^ (n + 1))) ≤ 6 * r ∧
      ∀ N : ℕ, 2 * r + (∑ n ∈ Finset.range N, (2 * r ^ (n + 2) + 2 * r ^ (n + 1))) ≤ 8 * r := by
  have habs : |r| < 1 := by rw [abs_of_pos hr0]; linarith
  have hg : Summable (fun n : ℕ => r ^ n) := summable_geometric_of_abs_lt_one habs
  have hfun : (fun n : ℕ => 2 * r ^ (n + 2) + 2 * r ^ (n + 1)) =
      (fun n : ℕ => (2 * r ^ 2 + 2 * r) * r ^ n) := by
    funext n
    simp only [pow_add, pow_one]
    ring
  have hu : Summable (fun n : ℕ => 2 * r ^ (n + 2) + 2 * r ^ (n + 1)) := by
    rw [hfun]
    exact hg.mul_left _
  have hsum : (∑' n : ℕ, (2 * r ^ (n + 2) + 2 * r ^ (n + 1))) =
      (2 * r ^ 2 + 2 * r) / (1 - r) := by
    rw [hfun, tsum_mul_left, tsum_geometric_of_abs_lt_one habs, div_eq_mul_inv]
  have hden : 0 < 1 - r := by linarith
  have hmain : (2 * r ^ 2 + 2 * r) / (1 - r) ≤ 6 * r := by
    apply (div_le_iff₀ hden).2
    have hp : 0 ≤ r * (1 - 2 * r) := mul_nonneg hr0.le (by linarith)
    nlinarith only [hp]
  refine ⟨hu, ?_, ?_⟩
  · rw [hsum]
    exact hmain
  · intro N
    have hfinite := Summable.sum_le_tsum (Finset.range N)
      (fun n _ => add_nonneg (mul_nonneg (by norm_num) (pow_nonneg hr0.le _))
        (mul_nonneg (by norm_num) (pow_nonneg hr0.le _))) hu
    calc
      _ ≤ 2 * r + ∑' n : ℕ, (2 * r ^ (n + 2) + 2 * r ^ (n + 1)) :=
        add_le_add le_rfl hfinite
      _ = 2 * r + (2 * r ^ 2 + 2 * r) / (1 - r) := by rw [hsum]
      _ ≤ 8 * r := by linarith only [hmain]

theorem actual_initial_double_completion_kernel
    {Ω : Type*} [MeasurableSpace Ω]
    (B : ℝ≥0 → Ω → ℝ) (P : Measure Ω) [IsProbabilityMeasure P]
    (hB : IsBrownianReal B P) (x : ℝ) (hx : 1 ≤ x) :
    let f : ℝ → ℝ := doubleCompletion (bridgeLaw B P) x
    let r : ℝ := Real.exp (-4 * x ^ 2)
    Continuous f ∧
      (∀ w : ℝ, 0 ≤ f w ∧ f w ≤ 8 * r) ∧
      eVariationOn f Set.univ ≤ ENNReal.ofReal (10 * r) ∧
      Tendsto f atTop (nhds 0) ∧ Tendsto f atBot (nhds 0) := by
  classical
  letI : IsProbabilityMeasure (bridgeLaw B P) := isProbabilityMeasure_bridgeLaw B P hB
  let f : ℝ → ℝ := doubleCompletion (bridgeLaw B P) x
  let r : ℝ := Real.exp (-4 * x ^ 2)
  change Continuous f ∧ (∀ w, 0 ≤ f w ∧ f w ≤ 8 * r) ∧
    eVariationOn f Set.univ ≤ ENNReal.ofReal (10 * r) ∧
    Tendsto f atTop (nhds 0) ∧ Tendsto f atBot (nhds 0)
  have hx0 : 0 ≤ x := by linarith
  have hxp : 0 < x := by linarith
  have hr0 : 0 < r := (initial_geometric_ratio_bound x hx).1
  have hrhalf : r ≤ 1 / 2 := (initial_geometric_ratio_bound x hx).2
  have hr1 : r < 1 := by linarith
  let c : ℝ → ℝ := fun w => max (-x) (min x w)
  have hc : Continuous c := by dsimp [c]; fun_prop
  have hcmono : Monotone c := by
    intro v w hvw
    exact max_le_max_left (-x) (min_le_min_left x hvw)
  have hcrange (w : ℝ) : |c w| ≤ x := by
    apply abs_le.mpr
    constructor
    · exact le_max_left _ _
    · exact max_le (by linarith) (min_le_left _ _)
  have hcx : c x = x := by simp [c, max_eq_right (by linarith : -x ≤ x)]
  have hcmx : c (-x) = -x := by simp [c, min_eq_right (by linarith : -x ≤ x)]
  let I : ℝ → ℝ → ℝ → ℝ := fun a b w => Real.exp (-(a + b * c w))
  have hIpos (a b w : ℝ) : 0 < I a b w := Real.exp_pos _
  have hIcont (a b : ℝ) : Continuous (I a b) := by dsimp [I]; fun_prop
  have hIvar (a b C : ℝ) (hb : ∀ w, I a b w ≤ C) :
      eVariationOn (I a b) Set.univ ≤ ENNReal.ofReal C := by
    rcases le_total 0 b with hb0 | hb0
    · apply antitoneOn_variation_le (I a b) Set.univ C
