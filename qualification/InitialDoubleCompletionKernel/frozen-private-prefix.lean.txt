private theorem doubleCompletion_eq (nu : Measure BridgeNoise) [IsProbabilityMeasure nu]
    (x w : ℝ) :
    doubleCompletion nu x w =
      1 - upperCompletion nu x 0 1 w - lowerCompletion nu x 0 1 w +
        intervalCompletion nu x 0 1 w

private theorem doubleCompletion_bounds (nu : Measure BridgeNoise) [IsProbabilityMeasure nu]
    (x w : ℝ) :
    0 ≤ doubleCompletion nu x w ∧ doubleCompletion nu x w ≤ 1

private theorem measurable_doubleCompletion (nu : Measure BridgeNoise) [IsProbabilityMeasure nu]
    (x : ℝ) : Measurable (doubleCompletion nu x)
