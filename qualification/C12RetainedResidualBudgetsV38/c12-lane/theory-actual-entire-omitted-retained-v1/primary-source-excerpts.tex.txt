\subsection{Adaptive actual-energy cutoff and its entire omitted tail}
Fix a sufficiently large numerical $K$ and, in the residual regime
\begin{equation}\label{d:resregime}
 x\ge X_0,\quad |X|\le\eta\sqrt n/x,\quad
 u=\delta(1+x)^3<c_*e^{-3x^2},\qquad c_*\le1,
\end{equation}
put $H=K\beta^2(1+x)^6$, $m=\lceil H\rceil$ and
\[
 \kappa_H=\inf\{r:T_r<H\},\qquad
 \tau=\kappa_H\wedge(n-m),\qquad
 \kappa_0=\inf\{r:T_r<h_0\}.
\]
Choose $X_0$ after $K$ so $2Ke^{-6X_0^2}\le1/4$.
Then $m\le2H\le n/4$, $m\ge h_0$, and exactly
\begin{equation}\label{d:cutoffevents}
 \{r<\tau\}=\{r<n-m,T_r\ge H\},\quad
 \{\tau\le r<\kappa_0\}
 =\{T_r\ge h_0\}\cap(\{T_r<H\}\cup\{n-r\le m\}).
\end{equation}
They are stopping ranks only after conditioning on the full labeled
sample and revealing a uniform permutation; the compensation in
Lemma~\ref{d:telescope} remains exact.

\begin{lemma}[Terminal and omitted absolute budgets]\label{d:cutoff}
For either monitored half-line or the interval,
\begin{align}\label{d:cutoffbounds}
 E\max_{\tau\le j\le n}|V_j^I-1_{A_n^I}|&\le CH/n,\nonumber\\
 E\sum_{\tau\le r<\kappa_0}O_r^\pm&\le CH/n,\nonumber\\
 E\sum_{\tau\le r<\kappa_0}1_{A_r^I}1_{\{z\in I\}}
                                      |\mathfrak r_b^I|&\le CH/n.
\end{align}
Each cost is at most $C\delta\Psi(x)$ under
\eqref{d:resregime}.
\end{lemma}
\begin{proof}
Apply Lemma~\ref{cn:terminal} at the present $H$ and $m$:
$h\le h_0\le H$, $m\le2H$, $m\le n/4$, and
\eqref{d:wallstrip} supplies \eqref{cn:terminalstrip}.
This proves the first assertion and, since $\tau\le\kappa_0$,
$E|V_{\kappa_0}-V_\tau|\le2CH/n$.

For the omitted retained and interval-negative terms, early ranks
force $T<H$ in \eqref{d:cutoffevents}. The shifted damped bound costs
$C\delta e^{-c'n/H}\le C\beta\sqrt H/n$.
At late deterministic ranks the same local corridor calculation in
Lemma~\ref{d:OW} bounds them by
\[
 \frac Cn E[1_{\{T\ge h_0\}}|b|^3T^{-1/2}D^5
                       (1_{\{T<H\}}+1_{\{t\le m\}})].
\]
The two sums are at most
\[
 \frac{C\beta}n\int_0^\infty\frac{s^{-1/2}}{s+1/H}\,ds
 +\frac{C\beta}n\int_0^{1/h_0}s^{-1/2}\min(m,s^{-1})\,ds
 \le C\beta\sqrt H/n.
\]
The first integral is $\pi\sqrt H$ and the second is at most
$4\sqrt m$. The exact one-wall telescope between $\tau$ and
$\kappa_0$ now pays its nonnegative overshoot by this retained bound
and the first assertion. For interval positive overshoots use
\textup{(4.4)}: their coefficients are the same nonnegative $ec$,
their survival/endpoint events are subsets of the corresponding
one-wall events, and their raw $T,d,b,u$ and cutoffs coincide.
The negative part was just paid cubically. This proves the full
absolute assertion. The scale conversion $H/n\le C\delta\Psi(x)$ is
\textup{(5.10)} in the main article.
\end{proof}
