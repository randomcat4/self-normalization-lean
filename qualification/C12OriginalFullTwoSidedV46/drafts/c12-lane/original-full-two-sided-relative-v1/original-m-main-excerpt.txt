150:Let $X,X_1,X_2,\ldots$ be iid real random variables with
151:$\E X=0$ and $0<\sigma^2=\E X^2<\infty$. For $n\ge1$, put
152:\begin{equation}\label{h:eq:statistics}
153: S_k=S_k^X=\sum_{i=1}^kX_i,\quad Q_k=\sum_{i=1}^kX_i^2,\quad
154: V_n=\sqrt{Q_n},\quad S_0=Q_0=0,
155:\end{equation}
156:and define
157:\[
158: M_n=M_n^+=\max_{1\le k\le n}S_k,\quad
159: M_n^-=\max_{1\le k\le n}(-S_k),\quad
160: M_n^a=\max_{1\le k\le n}|S_k|.
161:\]
162:Write $Q=Q_n$, $M=M_n^a$, and $W=S_n/\sqrt Q$ on $Q>0$, with
163:$W=0$ on $Q=0$. Every crossing is a weak multiplication event, such
164:as $\{M_n\ge xV_n\}$, including the all-zero sample. Strict survival
165:is its exact complement. The one-sided maximum excludes time zero;
166:this distinction matters at $x=0$. The absolute maximum is unchanged
167:if time zero is included.
168:
169:Let $B$ be standard Brownian motion, $M_B=\sup_{0\le t\le1}|B_t|$,
170:$H(x)=\PP(M_B\ge x)$, and $\Psi=\Phibar=1-\Phi$, where $\Phi$
171:and $\phi$ are the standard normal distribution function and density.
172:With $F(z)=f(z)=z^2\min(|z|,1)$, define, for each integer $j\ge1$,
173:\begin{equation}\label{m:R}
174: R_j(x)=j\E F\left(\frac{(1+x)X}{\sigma\sqrt j}\right),\quad x\ge0,
175: \qquad R=R_n(0).
176:\end{equation}
177:For $j=n$ this equals
178:\[
179: \frac{(1+x)^2}{\sigma^2}\left\{
180: \E[X^2;|X|>t_{n,x}]+t_{n,x}^{-1}\E[|X|^3;|X|\le t_{n,x}]
181: \right\},\qquad t_{n,x}=\frac{\sigma\sqrt n}{1+x}.
182:\]
183:When a higher moment is required, the additional assumption is
184:\begin{equation}\label{h:eq:assumptions}
185: \mu_p=\E|X|^p<\infty,\qquad 2<p\le3,
186:\end{equation}
187:and we write
188:\begin{equation}\label{h:eq:beta}
189: L_n=\beta_{n,p}=\frac{\mu_p}{\sigma^p n^{(p-2)/2}}.
190:\end{equation}
191:Fix the numerical $\rho_*>0$ in
192:Proposition~\ref{h:prop:preliminary} and set
193:\begin{equation}\label{h:eq:comparisoninterval}
194: I_n=\mathcal I_n=\{x\ge0:x\le L_n^{-1/p}\ \text{or}\ R_n(x)\le\rho_*\}.
195:\end{equation}
196:The enlarged domain is useful because small-budget auxiliary laws need
197:not be evaluated inside the original law's moment range.
198:
199:\begin{theorem}[One-sided relative error]\label{h:thm:main}
200:Under \eqref{h:eq:assumptions}, there is a numerical constant $C$ such
201:that
202:\begin{equation}\label{h:eq:main}
203: \left|\frac{\PP(M_n\ge xV_n)}{\Psi(x)}-2\right|
204: \le C R_n(x),\qquad x\in I_n.
205:\end{equation}
206:The same assertion holds for $M_n^-$. The constant is uniform in
207:$p\in(2,3]$, the law, $n$ and $x$.
208:\end{theorem}
209:
210:\begin{theorem}[Two-sided relative error]\label{m:main}
211:Under \eqref{h:eq:assumptions}, with the same $I_n$, a numerical $C$
212:satisfies
213:\begin{equation}\label{m:B}
214: \left|\frac{\PP(M_n^a\ge xV_n)}{H(x)}-1\right|
215: \le C R_n(x),\qquad x\in[0,1]\cup I_n.
216:\end{equation}
217:In particular,
218:\begin{equation}\label{m:A}
219: \left|\frac{\PP(M_n^a\ge xV_n)}{H(x)}-1\right|
220: \le C(1+x)^pL_n,\qquad 0\le x\le L_n^{-1/p}.
221:\end{equation}
222:For a fixed admissible law and fixed $p$, if
223:$b_n=o(n^{(p-2)/(2p)})$, $b_n\ge0$, then
224:\begin{equation}\label{m:C}
225: \sup_{0\le x\le b_n}
226: \left|\frac{\PP(M_n^a\ge xV_n)}{H(x)}-1\right|\longrightarrow0.
227:\end{equation}
228:\end{theorem}
