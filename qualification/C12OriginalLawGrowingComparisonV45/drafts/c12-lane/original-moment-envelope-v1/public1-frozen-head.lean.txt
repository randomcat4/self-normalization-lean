    (p moment sigma : ℝ) (hp : 2 < p) (hMoment : 0 ≤ moment) (hSigma : 0 < sigma)
    (b : ℕ → ℝ) (hb : ∀ n : ℕ, 0 ≤ b n)
    (hRate : Tendsto
      (fun n : ℕ => b n / Real.rpow (n : ℝ) ((p - 2) / (2 * p)))
      atTop (𝓝 0)) :
    Tendsto
      (fun n : ℕ => Real.rpow (1 + b n) p *
        (moment / (Real.rpow sigma p * Real.rpow (n : ℝ) ((p - 2) / 2))))
      atTop (𝓝 0)
